1//! Fail-closed pre-trade risk evaluation.
2//!
3//! Derived from the internal Nirnay `crates/nirnay-core/src/risk.rs` at source snapshot
4//! `cc3a34c317a86a9984401efe90fa3f43d67a1787`.
5//!
6//! The policy is evaluated against the canonical India instrument model:
7//! listing activity comes from `india-domain`, quantities are non-negative
8//! lots, and every cash limit is exact paise. A policy that fails validation
9//! is never evaluated; an evaluation returns every check so a rejection can
10//! be explained rather than inferred.

65/// Effective risk policy. All limits are mandatory; validation rejects an
66/// empty allowlist, a zero limit or an impossible collar. The sector, stress,
67/// liquidity, Greeks and borrow controls are opt-in: `None` means the control
68/// has no configured cap or reference data, and the corresponding check is
69/// inert. Once configured, each control fails closed on missing evidence.
70#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
71pub struct RiskPolicy {
72    pub version: String,
73    pub allowed_accounts: BTreeSet<AccountId>,
74    pub allowed_strategies: BTreeSet<StrategyId>,
75    pub allowed_algos: BTreeSet<AlgoId>,
76    pub allowed_instruments: BTreeSet<InstrumentId>,
77    pub max_order_quantity: Quantity,
78    pub max_order_notional_paise: Paise,
79    pub max_position_quantity: Quantity,
80    pub max_gross_exposure_paise: Paise,
81    pub max_daily_loss_paise: Paise,
82    pub max_open_orders: u32,
83    pub max_orders_per_second: u32,
84    pub max_price_deviation_ppm: u32,
85    pub max_quote_age_ns: u64,
86    pub allow_short: bool,
87    pub max_strategy_exposure_paise: Paise,
88    pub max_sector_exposure_paise: Option<Paise>,
89    pub stress_scenario: Option<StressScenario>,
90    pub max_stress_loss_paise: Option<Paise>,
91    pub max_participation_ppm: Option<u32>,
92    pub greek_limits: Option<GreekLimits>,
93    pub max_borrow_quantity: Option<Quantity>,
94    pub max_financing_cost_paise: Option<Paise>,
95}

97impl RiskPolicy {
98    pub fn validate(&self) -> Result<(), DomainError> {
99        if self.version.trim().is_empty()
100            || self.allowed_accounts.is_empty()
101            || self.allowed_strategies.is_empty()
102            || self.allowed_algos.is_empty()
103            || self.allowed_instruments.is_empty()
104            || self.max_order_quantity.is_zero()
105            || self.max_order_notional_paise.0 <= 0
106            || self.max_position_quantity.is_zero()
107            || self.max_gross_exposure_paise.0 <= 0
108            || self.max_daily_loss_paise.0 <= 0
109            || self.max_open_orders == 0
110            || self.max_orders_per_second == 0
111            || self.max_price_deviation_ppm > 1_000_000
112            || self.max_quote_age_ns == 0
113            || self.max_strategy_exposure_paise.0 <= 0
114        {
115            return Err(DomainError::InvalidInstrument(
116                "risk policy has invalid or empty limits".to_owned(),
117            ));
118        }
119        if self.max_sector_exposure_paise.is_some_and(|cap| cap.0 <= 0)
120            || self.max_stress_loss_paise.is_some_and(|cap| cap.0 <= 0)
121            || self
122                .max_participation_ppm
123                .is_some_and(|ppm| ppm == 0 || ppm > 1_000_000)
124            || self
125                .max_borrow_quantity
126                .is_some_and(|quantity| quantity.is_zero())
127            || self.max_financing_cost_paise.is_some_and(|cost| cost.0 < 0)
128        {
129            return Err(DomainError::InvalidInstrument(
130                "risk policy has an invalid optional limit".to_owned(),
131            ));
132        }
133        // A scenario without a loss cap (or a cap without a scenario) would
134        // silently disable the stress check: both must be configured together.
135        if self.stress_scenario.is_some() != self.max_stress_loss_paise.is_some() {
136            return Err(DomainError::InvalidInstrument(
137                "stress scenario and stress loss cap must be configured together".to_owned(),
138            ));
139        }
140        if let Some(scenario) = self.stress_scenario
141            && (scenario.price_shift_ppm.unsigned_abs() > 1_000_000
142                || scenario.liquidity_haircut_ppm > 1_000_000
143                || scenario.correlation_multiplier_ppm == 0
144                || scenario.correlation_multiplier_ppm > 10_000_000)
145        {
146            return Err(DomainError::InvalidInstrument(
147                "stress scenario ratios are outside the supported range".to_owned(),
148            ));
149        }
150        if let Some(limits) = self.greek_limits
151            && (limits.max_abs_delta_notional_paise < 0
152                || limits.max_abs_gamma < 0
153                || limits.max_abs_vega < 0
154                || limits.max_abs_theta < 0)
155        {
156            return Err(DomainError::InvalidInstrument(
157                "Greek limits cannot be negative".to_owned(),
158            ));
159        }
160        Ok(())
161    }
162}

164/// A risk policy validated for evaluation. There is exactly one source of
165/// a `ValidatedRiskPolicy`: [`ValidatedRiskPolicy::from_policy`], which
166/// runs [`RiskPolicy::validate`] first. An unvalidated policy can never
167/// reach [`evaluate_risk`]; deserialization is equally unable to skip the
168/// check because construction, not serde, is the only entry point.
169#[derive(Clone, Debug, PartialEq, Eq)]
170pub struct ValidatedRiskPolicy {
171    policy: RiskPolicy,
172}
173
174impl ValidatedRiskPolicy {
175    /// Validate the policy and construct a value that permits evaluation. Rejects
176    /// empty allowlists, zero limits, and impossible collars with the same
177    /// typed error as [`RiskPolicy::validate`].
178    pub fn from_policy(policy: &RiskPolicy) -> Result<Self, DomainError> {
179        policy.validate()?;
180        Ok(Self {
181            policy: policy.clone(),
182        })
183    }
184}

245/// Evaluate every check, including checks after the first failure, so the
246/// caller can report the complete reason for a rejection.
247///
248/// The policy must be a validated value; the context is validated at
249/// the boundary (negative exposure values deny), never trusted blindly:
250/// every caller-supplied financial state is treated as a claim, not a
251/// fact. Overflow flows to `None` and fails rather than producing an
252/// accepted decision.
253pub fn evaluate_risk(
254    intent: &OrderIntent,
255    policy: &ValidatedRiskPolicy,
256    context: &RiskContext,
257) -> RiskAssessment {
258    let instrument = context.instrument.as_ref();
259    let quote = context.quote.as_ref();
260    // A quote is evidence only for the venue it was observed on: the same
261    // instrument can trade on NSE and BSE, where lot, tick, settlement and
262    // price differ. Feed health and the collar must both use the intent's
263    // venue, never "a quote for this instrument".
264    let venue_quote = quote.filter(|value| {
265        value.instrument_id == intent.instrument_id
266            && value.venue == intent.venue
267            && context.venue == intent.venue
268    });
269    // Context boundary: gross-exposure claims must be valid before any
270    // projection arithmetic uses them (a negative current exposure would
271    // understate the projected total and pass the limit dishonestly).
272    let exposures_valid = context.gross_exposure_paise.0 >= 0
273        && context.pending_gross_exposure_paise.0 >= 0
274        && context.strategy_exposure_paise.0 >= 0
275        && context.sector_exposure_paise.0 >= 0;
276    let order_notional = notional(intent);
277    let projected_exposure = order_notional
278        .and_then(|order_value| order_value.checked_add(i128::from(context.gross_exposure_paise.0)))
279        .and_then(|value| value.checked_add(i128::from(context.pending_gross_exposure_paise.0)));
280    let projected_strategy = order_notional
281        .and_then(|value| value.checked_add(i128::from(context.strategy_exposure_paise.0)));
282    let projected_sector = order_notional
283        .and_then(|value| value.checked_add(i128::from(context.sector_exposure_paise.0)));
284    let checks = vec![
285        check(
286            RiskCheck::AccountAllowlist,
287            policy.allowed_accounts.contains(&context.account_id)
288                && intent.account_id == context.account_id,
289            "account must match the active account and its allowlist",
290        ),
291        check(
292            RiskCheck::StrategyAllowlist,
293            policy.allowed_strategies.contains(&intent.strategy_id)
294                && intent.strategy_id == context.strategy_id,
295            "strategy must match an approved strategy identity",
296        ),
297        check(
298            RiskCheck::AlgoAllowlist,
299            policy.allowed_algos.contains(&intent.algo_id)
300                && !intent.strategy_version.trim().is_empty()
301                && !intent.reason.trim().is_empty(),
302            "algo identity must be approved and the intent must carry a version and reason",
303        ),
304        check(
305            RiskCheck::EvidenceReferences,
306            !intent.evidence.is_empty(),
307            "order intent must cite at least one recorded input event",
308        ),
309        check(
310            RiskCheck::InstrumentAllowlist,
311            policy.allowed_instruments.contains(&intent.instrument_id),
312            "instrument must belong to the configured universe",
313        ),
314        check(
315            RiskCheck::InstrumentReferenceData,
316            instrument.is_some_and(|value| {
317                value.instrument_id == intent.instrument_id
318                    && context.venue == intent.venue
319                    && value
320                        .effective_listing_on(intent.venue, context.trade_date)
321                        .is_some()
322            }),
323            "effective-dated instrument metadata must exist and be active on the intent's venue",
324        ),
325        check(
326            RiskCheck::ProductEligibility,
327            instrument.is_some_and(|value| {
328                context.venue == intent.venue
329                    && product_eligible(value, intent.venue, context.trade_date, intent.product)
330            }),
331            "trade-for-trade and Z-category listings are delivery-only; intraday is not eligible",
332        ),
333        check(
334            RiskCheck::SessionOpen,
335            context.session_open,
336            "trading session must be open",
337        ),
338        check(
339            RiskCheck::MarketFeedHealth,
340            matches!(context.feed_health, FeedHealth::Healthy { age_ns } if age_ns <= policy.max_quote_age_ns)
341                && venue_quote.is_some(),
342            "quote must be two-sided, current, continuous and observed on the intent's venue",
343        ),
344        check(
345            RiskCheck::AccountSnapshotAvailable,
346            context.account_snapshot_available,
347            "positions and account state must be available",
348        ),
349        check(
350            RiskCheck::QuantityLimit,
351            !intent.quantity.is_zero() && intent.quantity.get() <= policy.max_order_quantity.get(),
352            "order quantity must fit the configured range",
353        ),
354        check(
355            RiskCheck::LotAlignment,
356            instrument
357                .and_then(|value| value.effective_terms_on(intent.venue, context.trade_date))
358                .is_some_and(|terms| {
359                    terms.market_lot > 0 && intent.quantity.get() % terms.market_lot == 0
360                }),
361            "order quantity must align to the intended venue's effective lot size",
362        ),
363        check(
364            RiskCheck::TickAlignment,
365            instrument
366                .and_then(|value| value.effective_terms_on(intent.venue, context.trade_date))
367                .is_some_and(|terms| {
368                    intent.limit_price.get().0 % terms.tick_assignment.tick_paise == 0
369                }),
370            "limit price must align to the intended venue's effective tick size",
371        ),
372        check(
373            RiskCheck::OrderNotional,
374            notional(intent)
375                .is_some_and(|value| value <= i128::from(policy.max_order_notional_paise.0)),
376            "limit-price notional must not exceed the configured order limit",
377        ),
378        check(
379            RiskCheck::PriceCollar,
380            price_within_collar(intent, venue_quote, policy.max_price_deviation_ppm),
381            "limit price must remain inside the collar of a quote from the intent's venue",
382        ),
383        check(
384            RiskCheck::PositionLimit,
385            position_allowed(
386                intent,
387                context.position_quantity,
388                context.pending_buy_quantity.get(),
389                context.pending_sell_quantity.get(),
390                policy.max_position_quantity.get(),
391                policy.allow_short,
392            ),
393            "projected net position must fit the position policy",
394        ),
395        check(
396            RiskCheck::GrossExposure,
397            exposures_valid
398                && projected_exposure
399                    .is_some_and(|value| value <= i128::from(policy.max_gross_exposure_paise.0)),
400            if exposures_valid {
401                "current gross exposure plus order notional must fit the exposure policy"
402            } else {
403                "gross exposure values must be non-negative"
404            },
405        ),
406        check(
407            RiskCheck::StrategyExposure,
408            exposures_valid
409                && projected_strategy
410                    .is_some_and(|value| value <= i128::from(policy.max_strategy_exposure_paise.0)),
411            "projected strategy notional must fit the strategy cap",
412        ),
413        check(
414            RiskCheck::SectorExposure,
415            sector_allowed(policy, context, projected_sector),
416            "a configured sector cap requires a label and a projected notional within it",
417        ),
418        check(
419            RiskCheck::StressLoss,
420            stress_allowed(policy, context, order_notional),
421            "deterministic projected stress loss must fit the configured cap",
422        ),
423        check(
424            RiskCheck::LiquidityParticipation,
425            participation_allowed(policy, context, intent),
426            "order notional must fit the configured fraction of synthetic displayed depth",
427        ),
428        check(
429            RiskCheck::Greeks,
430            greeks_allowed(policy, context),
431            "a derivative-flagged instrument needs greeks within every configured limit",
432        ),
433        check(
434            RiskCheck::BorrowFinancing,
435            borrow_allowed(policy, context, intent),
436            "a short needs synthetic borrow evidence within the configured cost cap",
437        ),
438        check(
439            RiskCheck::DailyLossLimit,
440            i128::from(context.daily_pnl.0) > -i128::from(policy.max_daily_loss_paise.0),
441            "daily loss must remain inside the configured limit",
442        ),
443        check(
444            RiskCheck::OpenOrderLimit,
445            context.open_orders < policy.max_open_orders,
446            "open-order count must remain below the configured limit",
447        ),
448        check(
449            RiskCheck::OrderRateLimit,
450            context.orders_last_second < policy.max_orders_per_second,
451            "order rate must remain below the configured account limit",
452        ),
453        check(
454            RiskCheck::KillSwitch,
455            !context.kill_switch_engaged,
456            "the kill switch must be released before exposure can increase",
457        ),
458        check(
459            RiskCheck::Reconciliation,
460            !context.reconciliation_blocked,
461            "reconciliation must be complete and matched",
462        ),
463        check(
464            RiskCheck::Idempotency,
465            !context.duplicate_intent,
466            "intent identity must not conflict with a prior request",
467        ),
468        check(
469            RiskCheck::IntentExpiry,
470            context.now < intent.expires_at,
471            "intent must remain valid at the decision time",
472        ),
473    ];
474    let accepted = checks.iter().all(|result| result.passed);
475    RiskAssessment {
476        intent_id: intent.intent_id,
477        policy_version: policy.version.clone(),
478        accepted,
479        checks,
480    }
481}

483/// Product eligibility from the listing state in force on `on_date`.
484///
485/// Tradability and product eligibility are different questions: a
486/// trade-for-trade or Z-category security is `Active` and can be delivered,
487/// but its settlement regime (no netting) makes intraday ineligible. An
488/// absent or inactive listing is not eligible for anything — callers must
489/// fail closed rather than infer from a symbol.
490fn product_eligible(
491    instrument: &trading_core::Instrument,
492    venue: trading_core::Venue,
493    on_date: trading_core::TradeDate,
494    product: trading_core::Product,
495) -> bool {
496    let Some(listing) = instrument.effective_listing_on(venue, on_date) else {
497        return false;
498    };
499    let intraday = product == trading_core::Product::Intraday;
500    if intraday && listing.z_category {
501        return false;
502    }
503    match listing.settlement_regime {
504        trading_core::SettlementRegime::Standard => true,
505        trading_core::SettlementRegime::TradeForTrade => !intraday,
506    }
507}

517fn notional(intent: &OrderIntent) -> Option<i128> {
518    i128::from(intent.limit_price.get().0).checked_mul(i128::from(intent.quantity.get()))
519}
520
521fn price_within_collar(
522    intent: &OrderIntent,
523    quote: Option<&QuoteSnapshot>,
524    maximum_deviation_ppm: u32,
525) -> bool {
526    // Exact: with M₂ = bid + ask and order price p (exact integers), the
527    // deviation test |2p − M₂| · 10⁶ ≤ M₂ · deviation never rounds the
528    // reference price first, so a limit exactly at the half-paise midpoint
529    // is evaluated against the true midpoint.
530    let Some(midpoint) = quote.and_then(QuoteSnapshot::midpoint_doubled) else {
531        return false;
532    };
533    let doubled_limit = i128::from(intent.limit_price.get().0) * 2;
534    let doubled_delta = (doubled_limit - midpoint.value()).unsigned_abs();
535    let lhs = doubled_delta * 1_000_000_u128;
536    // Reference magnitude, matching the pre-change formula: the collar
537    // bound is the absolute reference times the deviation tolerance.
538    let rhs = midpoint.value().unsigned_abs() * u128::from(maximum_deviation_ppm);
539    lhs <= rhs
540}

542/// Sector check. With no configured cap the control is inert; once configured,
543/// an unlabelled instrument cannot pass by default.
544fn sector_allowed(
545    policy: &ValidatedRiskPolicy,
546    context: &RiskContext,
547    projected: Option<i128>,
548) -> bool {
549    let Some(cap) = policy.max_sector_exposure_paise else {
550        return true;
551    };
552    context
553        .sector_label
554        .as_ref()
555        .is_some_and(|label| !label.trim().is_empty())
556        && projected.is_some_and(|value| value <= i128::from(cap.0))
557}

559/// Deterministic projected loss: parallel price shift over the exact notional
560/// of every marked position plus this order, after a liquidity haircut and a
561/// linear correlation multiplier. All ratios are integer ppm and overflow
562/// denies. A position without a mark fails closed.
563fn stress_allowed(
564    policy: &ValidatedRiskPolicy,
565    context: &RiskContext,
566    order_notional: Option<i128>,
567) -> bool {
568    let (Some(scenario), Some(cap)) = (policy.stress_scenario, policy.max_stress_loss_paise) else {
569        return true;
570    };
571    let Some(order) = order_notional.and_then(|value| u128::try_from(value.max(0)).ok()) else {
572        return false;
573    };
574    let mut exposure = order;
575    for (instrument_id, quantity) in &context.positions {
576        let Some(mark) = context.marks_paise.get(instrument_id) else {
577            return false;
578        };
579        let magnitude = u128::from(quantity.unsigned_abs());
580        let Some(value) = magnitude.checked_mul(u128::from(mark.0.unsigned_abs())) else {
581            return false;
582        };
583        let Some(next) = exposure.checked_add(value) else {
584            return false;
585        };
586        exposure = next;
587    }
588    let haircut = 1_000_000_u128.saturating_sub(u128::from(scenario.liquidity_haircut_ppm));
589    let loss = exposure
590        .checked_mul(haircut)
591        .and_then(|value| value.checked_div(1_000_000))
592        .and_then(|value| value.checked_mul(u128::from(scenario.price_shift_ppm.unsigned_abs())))
593        .and_then(|value| value.checked_div(1_000_000))
594        .and_then(|value| value.checked_mul(u128::from(scenario.correlation_multiplier_ppm)))
595        .and_then(|value| value.checked_div(1_000_000));
596    loss.is_some_and(|value| i128::try_from(value).is_ok_and(|loss| loss <= i128::from(cap.0)))
597}

599/// Synthetic participation check. The reference quantity is the smaller of
600/// displayed depth and ADV, valued at the intent's exact limit price; the
601/// order notional may consume at most `max_participation_ppm` of it.
602fn participation_allowed(
603    policy: &ValidatedRiskPolicy,
604    context: &RiskContext,
605    intent: &OrderIntent,
606) -> bool {
607    let Some(max_ppm) = policy.max_participation_ppm else {
608        return true;
609    };
610    let Some(depth) = context.depth else {
611        return false;
612    };
613    let quantity = depth.displayed_quantity.get().min(depth.adv_quantity.get());
614    let reference = i128::from(quantity).checked_mul(i128::from(intent.limit_price.get().0));
615    let allowance = reference
616        .and_then(|value| value.checked_mul(i128::from(max_ppm)))
617        .and_then(|value| value.checked_div(1_000_000));
618    notional(intent)
619        .is_some_and(|order| order >= 0 && allowance.is_some_and(|limit| order <= limit))
620}

622/// Synthetic Greeks check. A derivative-flagged instrument without greeks is
623/// denied even when no limits are configured; configured limits bound each
624/// absolute sensitivity.
625fn greeks_allowed(policy: &ValidatedRiskPolicy, context: &RiskContext) -> bool {
626    if !context.is_derivative {
627        return true;
628    }
629    let Some(greeks) = context.greeks else {
630        return false;
631    };
632    let Some(limits) = policy.greek_limits else {
633        return true;
634    };
635    let within = |observed: i64, limit: i64| {
636        i128::from(observed.unsigned_abs()) <= i128::from(limit.unsigned_abs())
637    };
638    within(
639        greeks.delta_notional_paise,
640        limits.max_abs_delta_notional_paise,
641    ) && within(greeks.gamma_scaled, limits.max_abs_gamma)
642        && within(greeks.vega_scaled, limits.max_abs_vega)
643        && within(greeks.theta_scaled, limits.max_abs_theta)
644}

646/// Synthetic borrow check. With no configured quantity cap the control is
647/// inert; a configured short must carry available borrow within the cap and,
648/// when a cost cap is configured, within that cost too.
649fn borrow_allowed(
650    policy: &ValidatedRiskPolicy,
651    context: &RiskContext,
652    intent: &OrderIntent,
653) -> bool {
654    if intent.side != Side::Sell {
655        return true;
656    }
657    let Some(max_quantity) = policy.max_borrow_quantity else {
658        return true;
659    };
660    let Some(borrow) = context.borrow else {
661        return false;
662    };
663    let quantity_ok = borrow.available_quantity.get() >= intent.quantity.get()
664        && borrow.available_quantity.get() <= max_quantity.get();
665    let cost_ok = policy
666        .max_financing_cost_paise
667        .is_none_or(|cap| borrow.financing_cost_paise.0 <= cap.0);
668    quantity_ok && cost_ok
669}
670
671fn position_allowed(
672    intent: &OrderIntent,
673    current: i64,
674    pending_buys: i64,
675    pending_sells: i64,
676    maximum: i64,
677    allow_short: bool,
678) -> bool {
679    let current = i128::from(current);
680    let pending_buys = i128::from(pending_buys);
681    let pending_sells = i128::from(pending_sells);
682    let new_quantity = i128::from(intent.quantity.get());
683    let upper = current
684        + pending_buys
685        + if intent.side == Side::Buy {
686            new_quantity
687        } else {
688            0
689        };
690    let lower = current
691        - pending_sells
692        - if intent.side == Side::Sell {
693            new_quantity
694        } else {
695            0
696        };
697    let maximum = u128::from(maximum.unsigned_abs());
698    (allow_short || lower >= 0)
699        && upper.unsigned_abs() <= maximum
700        && lower.unsigned_abs() <= maximum
701}
