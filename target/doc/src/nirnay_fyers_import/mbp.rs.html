1//! Typed five-level MBP projection from FYERS Data Socket SDK callbacks.
2//!
3//! The captured item is a Python SDK callback observation, not an original
4//! exchange packet. That origin is carried by the kernel envelope; this module
5//! only converts the decoded depth fields into typed MBP levels.

15/// Five levels for the standard Data Socket product. The separate 50-level TBT
16/// product is not accepted here.
17pub const DATA_SOCKET_DEPTH_LEVELS: u32 = 5;
18
19#[derive(Debug, Clone, PartialEq, Eq, Error)]
20pub enum FyersMbpError {
21    #[error("callback payload is not an object")]
22    NotObject,
23    #[error("depth level {0} has an invalid price")]
24    InvalidPrice(usize),
25    #[error("depth level {0} has an invalid quantity")]
26    InvalidQuantity(usize),
27    #[error("depth level {0} has an invalid order count")]
28    InvalidOrderCount(usize),
29    #[error("price exceeds the paise range")]
30    PriceOverflow,
31    #[error("snapshot validation failed: {0}")]
32    InvalidSnapshot(String),
33}
34
35fn decimal_string_paise(text: &str) -> Result<i64, FyersMbpError> {
36    let (whole, fraction) = text.split_once('.').unwrap_or((text, ""));
37    if fraction.len() > 2 || fraction.bytes().any(|byte| !byte.is_ascii_digit()) {
38        return Err(FyersMbpError::PriceOverflow);
39    }
40    if whole.is_empty() || whole.bytes().any(|byte| !byte.is_ascii_digit()) {
41        return Err(FyersMbpError::PriceOverflow);
42    }
43    let whole: i64 = whole.parse().map_err(|_| FyersMbpError::PriceOverflow)?;
44    let fraction: i64 = if fraction.is_empty() {
45        0
46    } else {
47        format!("{fraction:0<2}")
48            .parse()
49            .map_err(|_| FyersMbpError::PriceOverflow)?
50    };
51    whole
52        .checked_mul(100)
53        .and_then(|value| value.checked_add(fraction))
54        .ok_or(FyersMbpError::PriceOverflow)
55}
56
57fn price_paise(value: &Value) -> Result<i64, FyersMbpError> {
58    if let Some(text) = value.as_str() {
59        return decimal_string_paise(text.trim());
60    }
61    let number = value.as_f64().ok_or(FyersMbpError::PriceOverflow)?;
62    if !number.is_finite() || number < 0.0 || number * 100.0 >= i64::MAX as f64 {
63        return Err(FyersMbpError::PriceOverflow);
64    }
65    Ok((number * 100.0).round() as i64)
66}
67
68fn quantity(value: Option<&Value>, index: usize) -> Result<i64, FyersMbpError> {
69    let Some(value) = value else { return Ok(0) };
70    let number = value
71        .as_i64()
72        .ok_or(FyersMbpError::InvalidQuantity(index))?;
73    if number < 0 {
74        return Err(FyersMbpError::InvalidQuantity(index));
75    }
76    Ok(number)
77}
78
79fn order_count(value: Option<&Value>, index: usize) -> Result<Option<u32>, FyersMbpError> {
80    let Some(value) = value else { return Ok(None) };
81    if value.is_null() {
82        return Ok(None);
83    }
84    let number = value
85        .as_u64()
86        .ok_or(FyersMbpError::InvalidOrderCount(index))?;
87    u32::try_from(number)
88        .map(Some)
89        .map_err(|_| FyersMbpError::InvalidOrderCount(index))
90}
91
92fn levels(payload: &Value, side: &str, index: usize) -> Result<MbpLevel, FyersMbpError> {
93    let price = payload
94        .get(format!("{side}_price{index}"))
95        .ok_or(FyersMbpError::InvalidPrice(index))?;
96    let price_paise = price_paise(price).map_err(|_| FyersMbpError::InvalidPrice(index))?;
97    Ok(MbpLevel {
98        price_paise,
99        quantity: quantity(payload.get(format!("{side}_size{index}")), index)?,
100        order_count: order_count(payload.get(format!("{side}_order{index}")), index)?,
101    })
102}

104/// Project a `dp` callback into a typed five-level MBP snapshot. Non-depth
105/// callback types return `None` rather than fabricating a book.
106pub fn callback_to_mbp(
107    payload: &Value,
108    instrument_id: impl Into<String>,
109    venue: impl Into<String>,
110    product: impl Into<String>,
111    received_utc_ns: i64,
112) -> Result<Option<MarketDataEvent>, FyersMbpError> {
113    if !payload.is_object() {
114        return Err(FyersMbpError::NotObject);
115    }
116    if payload.get("type").and_then(Value::as_str) != Some("dp") {
117        return Ok(None);
118    }
119    let mut bids = Vec::new();
120    let mut asks = Vec::new();
121    for index in 1..=DATA_SOCKET_DEPTH_LEVELS {
122        let bid = levels(payload, "bid", index as usize)?;
123        let ask = levels(payload, "ask", index as usize)?;
124        if bid.quantity > 0 {
125            bids.push(bid);
126        }
127        if ask.quantity > 0 {
128            asks.push(ask);
129        }
130    }
131    if bids.is_empty() || asks.is_empty() {
132        return Ok(None);
133    }
134    let source_time_ns = match payload.get("exch_feed_time") {
135        Some(value) if !value.is_null() => {
136            let seconds = value.as_f64().ok_or(FyersMbpError::InvalidPrice(0))?;
137            if !seconds.is_finite() || seconds < 0.0 {
138                return Err(FyersMbpError::InvalidPrice(0));
139            }
140            let nanos = seconds * 1_000_000_000.0;
141            if nanos >= i64::MAX as f64 {
142                return Err(FyersMbpError::PriceOverflow);
143            }
144            Some(nanos as i64)
145        }
146        _ => None,
147    };
148    let snapshot = MarketByPriceSnapshot {
149        instrument_id: instrument_id.into(),
150        venue: venue.into(),
151        source_id: "fyers.data_socket".to_owned(),
152        product: product.into(),
153        depth_levels: DATA_SOCKET_DEPTH_LEVELS,
154        bids,
155        asks,
156        source_time_ns,
157        last_trade_time_ns: None,
158        received_utc_ns,
159    };
160    snapshot
161        .validate()
162        .map_err(|error| FyersMbpError::InvalidSnapshot(error.to_string()))?;
163    Ok(Some(MarketDataEvent::MarketByPriceSnapshot(snapshot)))
164}

166#[cfg(all(test, feature = "market-shaped-tests"))]
167mod tests {
168    use super::*;
169    use serde_json::json;
170
171    fn depth_payload() -> Value {
172        let mut payload = json!({"type": "dp", "exch_feed_time": 1_700_000_000.0});
173        for index in 1..=5 {
174            payload[format!("bid_price{index}")] = json!(format!("812.{}", 50 - index));
175            payload[format!("bid_size{index}")] = json!(100 + index);
176            payload[format!("bid_order{index}")] = json!(index);
177            payload[format!("ask_price{index}")] = json!(format!("812.{}", 60 + index));
178            payload[format!("ask_size{index}")] = json!(200 + index);
179            payload[format!("ask_order{index}")] = json!(index + 10);
180        }
181        payload
182    }
183
184    #[test]
185    fn dp_callback_projects_five_levels_with_order_counts() {
186        let event = callback_to_mbp(&depth_payload(), "SBIN-EQ", "NSE", "data_socket", 42)
187            .expect("projection")
188            .expect("depth event");
189        let MarketDataEvent::MarketByPriceSnapshot(snapshot) = event else {
190            panic!("expected snapshot");
191        };
192        snapshot.validate().expect("valid snapshot");
193        assert_eq!(snapshot.bids.len(), 5, "five bid levels");
194        assert_eq!(snapshot.asks.len(), 5, "five ask levels");
195        assert_eq!(
196            snapshot.source_id, "fyers.data_socket",
197            "SDK callback origin"
198        );
199        assert_eq!(
200            snapshot.bids[0].order_count,
201            Some(1),
202            "displayed order count is carried"
203        );
204        assert_eq!(
205            snapshot.bids[0].price_paise, 81_249,
206            "decimal string converts exactly"
207        );
208        assert_eq!(
209            snapshot.source_time_ns,
210            Some(1_700_000_000_000_000_000),
211            "exchange seconds convert to ns"
212        );
213    }
214
215    #[test]
216    fn non_depth_callback_does_not_fabricate_a_book() {
217        assert_eq!(
218            callback_to_mbp(
219                &json!({"type": "sf", "ltp": 812.5}),
220                "SBIN-EQ",
221                "NSE",
222                "data_socket",
223                1
224            )
225            .expect("projection"),
226            None,
227            "last-trade callbacks are not depth"
228        );
229    }
230
231    #[test]
232    fn zero_quantity_levels_are_skipped() {
233        let mut payload = depth_payload();
234        payload["bid_size1"] = json!(0);
235        let event = callback_to_mbp(&payload, "SBIN-EQ", "NSE", "data_socket", 1)
236            .expect("projection")
237            .expect("depth event");
238        let MarketDataEvent::MarketByPriceSnapshot(snapshot) = event else {
239            panic!("expected snapshot")
240        };
241        assert_eq!(snapshot.bids.len(), 4, "empty level is omitted");
242    }
243
244    #[test]
245    fn invalid_depth_values_are_refused() {
246        let mut negative = depth_payload();
247        negative["bid_size2"] = json!(-1);
248        assert!(
249            callback_to_mbp(&negative, "SBIN-EQ", "NSE", "data_socket", 1).is_err(),
250            "negative depth quantity is refused"
251        );
252
253        let mut bad_price = depth_payload();
254        bad_price["ask_price3"] = json!("12.345");
255        assert!(
256            callback_to_mbp(&bad_price, "SBIN-EQ", "NSE", "data_socket", 1).is_err(),
257            "a price with more than two decimal places is refused"
258        );
259    }
260
261    #[test]
262    fn empty_depth_returns_no_event() {
263        let mut payload = depth_payload();
264        for index in 1..=5 {
265            payload[format!("bid_size{index}")] = json!(0);
266            payload[format!("ask_size{index}")] = json!(0);
267        }
268        assert_eq!(
269            callback_to_mbp(&payload, "SBIN-EQ", "NSE", "data_socket", 1).expect("projection"),
270            None,
271            "no populated levels means no book"
272        );
273        assert!(
274            callback_to_mbp(&json!({"type": "dp"}), "SBIN-EQ", "NSE", "data_socket", 1).is_err(),
275            "a depth callback missing price fields fails closed"
276        );
277    }
