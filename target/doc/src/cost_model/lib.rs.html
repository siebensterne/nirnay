1//! Effective-dated Indian transaction cost schedules.
2//!
3//! Derived from the internal Nirnay `crates/nirnay-engine/src/fees.rs` at source snapshot
4//! `cc3a34c317a86a9984401efe90fa3f43d67a1787`.
5//!
6//! Every rate is basis points scaled by one billion (parts per billion of
7//! turnover) and every charge is rounded to the nearest paise, half up, in
8//! integer arithmetic. The schedule is effective-dated and selected by
9//! product and side, so a historical calculation never uses a future rate.
10
11use serde::{Deserialize, Serialize};
12use thiserror::Error;
13use trading_core::{ChargeComponents, CheckedPaise, DomainError, Paise, Product, Side, TradeDate};
14
15const RATE_SCALE: i128 = 1_000_000_000;
16
17#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
18pub struct ChargeRate {
19    pub effective_from: TradeDate,
20    pub product: Product,
21    pub side: Side,
22    pub brokerage_ppb: u64,
23    pub stt_ppb: u64,
24    pub exchange_ppb: u64,
25    pub sebi_ppb: u64,
26    pub stamp_ppb: u64,
27    pub gst_ppb: u64,
28}
29
30impl ChargeRate {
31    fn validate(&self) -> Result<(), FeeError> {
32        let rates = [
33            self.brokerage_ppb,
34            self.stt_ppb,
35            self.exchange_ppb,
36            self.sebi_ppb,
37            self.stamp_ppb,
38            self.gst_ppb,
39        ];
40        if rates.into_iter().any(|rate| rate > RATE_SCALE as u64) {
41            return Err(FeeError::InvalidRate);
42        }
43        Ok(())
44    }
45}
46
47/// Validated, ordered set of charge schedules.
48#[derive(Clone, Debug, PartialEq, Eq, Deserialize)]
49#[serde(try_from = "Vec<ChargeRate>")]
50pub struct ChargeBook {
51    rates: Vec<ChargeRate>,
52}
53
54#[derive(Debug, Error)]
55pub enum FeeError {
56    #[error("charge rate is outside the accepted range")]
57    InvalidRate,
58    #[error("two charge schedules share the same effective date, product and side")]
59    DuplicateEffectiveDate,
60    #[error("no charge schedule covers the trade date, product and side")]
61    MissingSchedule,
62    #[error("charge arithmetic failed: {0}")]
63    Arithmetic(#[from] DomainError),
64}
65
66impl ChargeBook {
67    pub fn new(mut rates: Vec<ChargeRate>) -> Result<Self, FeeError> {
68        for rate in &rates {
69            rate.validate()?;
70        }
71        rates.sort_by_key(|rate| (rate.product, side_rank(rate.side), rate.effective_from));
72        for pair in rates.windows(2) {
73            if let [previous, next] = pair
74                && previous.product == next.product
75                && previous.side == next.side
76                && previous.effective_from == next.effective_from
77            {
78                return Err(FeeError::DuplicateEffectiveDate);
79            }
80        }
81        Ok(Self { rates })
82    }
83
84    /// Select the latest schedule effective on or before `trade_date`, then
85    /// compute every charge component for `turnover`.
86    pub fn calculate(
87        &self,
88        trade_date: TradeDate,
89        product: Product,
90        side: Side,
91        turnover: Paise,
92    ) -> Result<ChargeComponents, FeeError> {
93        let rate = self
94            .rates
95            .iter()
96            .rev()
97            .find(|rate| {
98                rate.effective_from <= trade_date && rate.product == product && rate.side == side
99            })
100            .ok_or(FeeError::MissingSchedule)?;
101        let brokerage = levy(turnover, rate.brokerage_ppb)?;
102        let stt = levy(turnover, rate.stt_ppb)?;
103        let exchange = levy(turnover, rate.exchange_ppb)?;
104        let sebi = levy(turnover, rate.sebi_ppb)?;
105        let stamp = levy(turnover, rate.stamp_ppb)?;
106        let taxable = brokerage.checked_add(exchange)?.checked_add(sebi)?;
107        let gst = levy(taxable, rate.gst_ppb)?;
108        Ok(ChargeComponents {
109            brokerage,
110            securities_transaction_tax: stt,
111            exchange_transaction: exchange,
112            sebi_turnover: sebi,
113            stamp_duty: stamp,
114            gst,
115            other: Paise(0),
116        })
117    }
118
119    #[must_use]
120    pub fn schedule_count(&self) -> usize {
121        self.rates.len()
122    }
123}
124
125fn side_rank(side: Side) -> u8 {
126    match side {
127        Side::Buy => 0,
128        Side::Sell => 1,
129    }
130}
131
132fn levy(turnover: Paise, rate_ppb: u64) -> Result<Paise, DomainError> {
133    let scaled = i128::from(turnover.0)
134        .checked_mul(i128::from(rate_ppb))
135        .and_then(|value| value.checked_add(RATE_SCALE / 2))
136        .ok_or(DomainError::Overflow)?;
137    let paise = scaled.div_euclid(RATE_SCALE);
138    i64::try_from(paise)
139        .map(Paise)
140        .map_err(|_| DomainError::Overflow)
141}

143impl TryFrom<Vec<ChargeRate>> for ChargeBook {
144    type Error = FeeError;
145
146    fn try_from(rates: Vec<ChargeRate>) -> Result<Self, Self::Error> {
147        Self::new(rates)
148    }
149}
