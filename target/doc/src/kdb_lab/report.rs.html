1//! Machine-readable analysis report with input/output digests.
2//!
3//! The report is validated by `schemas/analytics/kdb-metrics-report.v1.schema.json`
4//! in the repository's contract gate. It never contains credentials or
5//! provider account identifiers.
6
7use serde::Serialize;
8use serde_json::Value;
9
10use crate::metrics::to_json_value;
11use crate::runner::Analysis;
12
13/// Schema reference carried by every report.
14pub const REPORT_SCHEMA: &str = "schemas/analytics/kdb-metrics-report.v1.schema.json";
15
16/// Identity of the analytics engine used for one run.
17#[derive(Debug, Clone, Serialize)]
18pub struct EngineInfo {
19    pub name: &'static str,
20    pub binary: String,
21}
22
23/// Input lineage recorded for one run.
24#[derive(Debug, Clone, Serialize)]
25pub struct InputInfo {
26    pub path: String,
27    pub sha256: String,
28    pub rows: u64,
29}
30
31/// Output lineage recorded for one run.
32#[derive(Debug, Clone, Serialize)]
33pub struct OutputInfo {
34    pub sha256: String,
35    pub bytes: u64,
36}
37
38/// Complete report for one KDB-X analytics run.
39#[derive(Debug, Clone, Serialize)]
40pub struct AnalysisReport {
41    pub schema: &'static str,
42    pub schema_version: u32,
43    pub source_id: String,
44    pub engine: EngineInfo,
45    pub input: InputInfo,
46    pub output: OutputInfo,
47    pub metrics: Value,
48}
49
50/// Build a report from a successful analysis invocation.
51pub fn build_report(analysis: &Analysis, input_path: &str) -> AnalysisReport {
52    AnalysisReport {
53        schema: REPORT_SCHEMA,
54        schema_version: 1,
55        source_id: analysis.metrics.source_id.clone(),
56        engine: EngineInfo {
57            name: "kdb-x",
58            binary: analysis.binary.clone(),
59        },
60        input: InputInfo {
61            path: input_path.to_owned(),
62            sha256: analysis.input_sha256.clone(),
63            rows: analysis.input_rows,
64        },
65        output: OutputInfo {
66            sha256: analysis.output_sha256.clone(),
67            bytes: analysis.output_bytes,
68        },
69        metrics: to_json_value(&analysis.metrics),
70    }
71}
