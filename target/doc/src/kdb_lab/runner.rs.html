1//! Locate and drive the locally installed KDB-X engine, then pin its output.
2//!
3//! The runner always passes absolute paths and never mutates the input. Every
4//! invocation is recorded by input and output digests so a result can be
5//! reproduced or compared across machines.

29/// Default local KDB-X binary used by the owner environment.
30pub const DEFAULT_Q_BIN: &str = "/Users/tourist/.kx/bin/q";
31/// Override for the q binary, useful on a different owner machine or in tests.
32pub const Q_BIN_ENV: &str = "NIRNAY_Q_BIN";
33
34/// One reproducible analytics invocation.
35#[derive(Debug, Clone)]
36pub struct Analysis {
37    pub metrics: MbpMetrics,
38    pub stdout: String,
39    pub input_sha256: String,
40    pub output_sha256: String,
41    pub input_rows: u64,
42    pub output_bytes: u64,
43    pub binary: String,
44}

46/// Resolve the q binary without printing or exporting any credential.
47pub fn find_q() -> Result<PathBuf> {
48    let candidate = std::env::var_os(Q_BIN_ENV)
49        .map(PathBuf::from)
50        .unwrap_or_else(|| PathBuf::from(DEFAULT_Q_BIN));
51    if candidate.is_file() {
52        Ok(candidate)
53    } else {
54        bail!("KDB-X q binary not found at {}", candidate.display())
55    }
56}

58/// Absolute path of the checked-in q analytics contract.
59pub fn q_script_path() -> Result<PathBuf> {
60    let path = Path::new(env!("CARGO_MANIFEST_DIR")).join("../../analytics/kdb/mbp_metrics.q");
61    path.canonicalize()
62        .with_context(|| format!("cannot resolve q script at {}", path.display()))
63}
64
65fn digest(bytes: &[u8]) -> String {
66    let mut hasher = Sha256::new();
67    hasher.update(bytes);
68    format!("{:x}", hasher.finalize())
69}

71/// Validate the input contract and run q once, returning typed metrics and
72/// digests. Fails closed on a bad input, a bad exit status or a drifted output.
73pub fn analyze(input: &Path, source_id: &str, q_bin: &Path) -> Result<Analysis> {
74    let input_bytes =
75        std::fs::read(input).with_context(|| format!("cannot read input {}", input.display()))?;
76    let input_text = std::str::from_utf8(&input_bytes).context("input is not UTF-8")?;
77    let rows = parse_observations_csv(input_text)
78        .context("input does not satisfy the KDB-X analytics contract")?;
79    let input_abs = input
80        .canonicalize()
81        .with_context(|| format!("cannot resolve input {}", input.display()))?;
82    let script = q_script_path()?;
83
84    let output = Command::new(q_bin)
85        .arg(&script)
86        .arg(&input_abs)
87        .arg(source_id)
88        .output()
89        .with_context(|| format!("cannot launch q at {}", q_bin.display()))?;
90    if !output.status.success() {
91        let stderr = String::from_utf8_lossy(&output.stderr);
92        bail!(
93            "q exited with {:?}: {}",
94            output.status.code(),
95            stderr.trim()
96        );
97    }
98    let stdout = String::from_utf8(output.stdout).context("q stdout is not UTF-8")?;
99    let metrics = parse_metrics_csv(&stdout).context("q output failed the metrics contract")?;
100    if metrics.source_id != source_id {
101        bail!(
102            "q echoed source_id {:?}, expected {source_id:?}",
103            metrics.source_id
104        );
105    }
106    if metrics.rows != rows.len() as u64 {
107        bail!(
108            "q reported {} rows but the input contains {}",
109            metrics.rows,
110            rows.len()
111        );
112    }
113
114    Ok(Analysis {
115        metrics,
116        output_sha256: digest(stdout.as_bytes()),
117        output_bytes: stdout.len() as u64,
118        input_sha256: digest(&input_bytes),
119        input_rows: rows.len() as u64,
120        binary: q_bin.to_string_lossy().into_owned(),
121        stdout,
122    })
123}
