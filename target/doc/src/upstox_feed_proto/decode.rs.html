1//! Structural decoding helpers for `FeedResponse` frames.
2
3use crate::proto::{FeedResponse, feed::FeedUnion};
4use prost::Message;
5
6#[derive(Debug, thiserror::Error)]
7pub enum DecodeError {
8    #[error("protobuf decode failed: {0}")]
9    Decode(String),
10}
11
12pub type DecodeResult<T> = Result<T, DecodeError>;
13
14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub enum FeedResponseType {
16    InitialFeed,
17    LiveFeed,
18    MarketInfo,
19    Unknown(i32),
20}
21
22pub fn response_type_from_raw(raw: i32) -> FeedResponseType {
23    match raw {
24        0 => FeedResponseType::InitialFeed,
25        1 => FeedResponseType::LiveFeed,
26        2 => FeedResponseType::MarketInfo,
27        value => FeedResponseType::Unknown(value),
28    }
29}
30
31#[derive(Debug, Clone, Copy, PartialEq, Eq)]
32pub enum FeedKind {
33    Ltpc,
34    FullFeed,
35    FirstLevelWithGreeks,
36    Unset,
37}
38
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub struct FeedSummary {
41    pub key: String,
42    pub kind: FeedKind,
43    pub request_mode_raw: i32,
44}
45
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct FeedResponseDiagnostic {
48    pub response_type_raw: i32,
49    pub response_type: FeedResponseType,
50    pub current_ts: i64,
51    pub has_market_info: bool,
52    pub feed_keys: Vec<String>,
53    pub feeds: Vec<FeedSummary>,
54}
55
56/// Decode raw bytes into the generated `FeedResponse`.
57///
58/// Unknown fields are skipped by prost as usual; unknown enum values are
59/// preserved in the raw `i32` fields.
60pub fn decode_feed_response(bytes: &[u8]) -> DecodeResult<FeedResponse> {
61    FeedResponse::decode(bytes).map_err(|error| DecodeError::Decode(error.to_string()))
62}
63
64/// Build a deterministic diagnostic projection: keys sorted, no ordering
65/// implied by arrival position.
66pub fn decode_feed_response_diagnostic(bytes: &[u8]) -> DecodeResult<FeedResponseDiagnostic> {
67    let response = decode_feed_response(bytes)?;
68    let mut feeds: Vec<FeedSummary> = Vec::new();
69    let mut keys: Vec<String> = Vec::new();
70    for (key, feed) in &response.feeds {
71        keys.push(key.clone());
72        let kind = match &feed.feed_union {
73            Some(FeedUnion::Ltpc(_)) => FeedKind::Ltpc,
74            Some(FeedUnion::FullFeed(_)) => FeedKind::FullFeed,
75            Some(FeedUnion::FirstLevelWithGreeks(_)) => FeedKind::FirstLevelWithGreeks,
76            None => FeedKind::Unset,
77        };
78        feeds.push(FeedSummary {
79            key: key.clone(),
80            kind,
81            request_mode_raw: feed.request_mode,
82        });
83    }
84    keys.sort();
85    feeds.sort_by(|a, b| a.key.cmp(&b.key));
86    Ok(FeedResponseDiagnostic {
87        response_type_raw: response.r#type,
88        response_type: response_type_from_raw(response.r#type),
89        current_ts: response.current_ts,
90        has_market_info: response.market_info.is_some(),
91        feed_keys: keys,
92        feeds,
93    })
94}
