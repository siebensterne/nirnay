13macro_rules! positive_id {
14    ($name:ident) => {
15        #[derive(
16            Copy, Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize,
17        )]
18        #[serde(transparent)]
19        pub struct $name(NonZeroU64);
20
21        impl $name {
22            pub fn new(value: u64) -> Result<Self, DomainError> {
23                NonZeroU64::new(value)
24                    .map(Self)
25                    .ok_or(DomainError::ZeroIdentifier)
26            }
27
28            #[must_use]
29            pub const fn get(self) -> u64 {
30                self.0.get()
31            }
32        }
33
34        impl TryFrom<u64> for $name {
35            type Error = DomainError;
36
37            fn try_from(value: u64) -> Result<Self, Self::Error> {
38                Self::new(value)
39            }
40        }
41
42        impl fmt::Display for $name {
43            fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
44                write!(formatter, "{}", self.get())
45            }
46        }
47    };
48}

57/// Identity of one simulated or live run. A replay must supply the recorded
58/// run identity so event IDs reproduce exactly.
59#[derive(Copy, Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
60#[serde(try_from = "String", into = "String")]
61pub struct RunId([u8; 16]);
62
63impl RunId {
64    #[must_use]
65    pub const fn from_bytes(bytes: [u8; 16]) -> Self {
66        Self(bytes)
67    }
68
69    #[must_use]
70    pub const fn to_bytes(self) -> [u8; 16] {
71        self.0
72    }
73
74    #[must_use]
75    pub fn to_hex(self) -> String {
76        self.0.iter().map(|byte| format!("{byte:02x}")).collect()
77    }
78
79    pub fn parse(value: impl AsRef<str>) -> Result<Self, DomainError> {
80        let value = value.as_ref();
81        if value.len() != 32 || !value.bytes().all(is_lower_hex) {
82            return Err(DomainError::InvalidIdentifier(format!(
83                "run id `{value}` must be 32 lowercase hexadecimal characters"
84            )));
85        }
86        let mut bytes = [0_u8; 16];
87        for (index, byte) in bytes.iter_mut().enumerate() {
88            let pair = value
89                .get(index * 2..index * 2 + 2)
90                .ok_or_else(|| DomainError::InvalidIdentifier("run id is truncated".to_owned()))?;
91            *byte = u8::from_str_radix(pair, 16).map_err(|_| {
92                DomainError::InvalidIdentifier("run id is not hexadecimal".to_owned())
93            })?;
94        }
95        Ok(Self(bytes))
96    }
97}

100    type Error = DomainError;
101
102    fn try_from(value: String) -> Result<Self, Self::Error> {
103        Self::parse(value)
104    }
105}

127/// Deterministic event identity: run identity plus a monotonic ordinal.
128///
129/// Serialized as 48 lowercase hexadecimal characters. Parsing validates the
130/// shape, so a journal or broker payload cannot inject an arbitrary opaque
131/// event key.
132#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
133#[serde(try_from = "String", into = "String")]
134pub struct EventId(String);
135
136impl EventId {
137    pub const HEX_LEN: usize = 48;
138
139    #[must_use]
140    pub fn for_run(run: RunId, ordinal: u64) -> Self {
141        let mut value = run.to_hex();
142        value.push_str(&format!("{ordinal:016x}"));
143        Self(value)
144    }
145
146    pub fn parse(value: impl AsRef<str>) -> Result<Self, DomainError> {
147        let value = value.as_ref();
148        if value.len() != Self::HEX_LEN || !value.bytes().all(is_lower_hex) {
149            return Err(DomainError::InvalidIdentifier(format!(
150                "event id `{value}` must be 48 lowercase hexadecimal characters"
151            )));
152        }
153        Ok(Self(value.to_owned()))
154    }
155
156    #[must_use]
157    pub fn as_str(&self) -> &str {
158        &self.0
159    }
160}

167    }

190fn is_lower_hex(byte: u8) -> bool {
191    byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte)
