1//! Typed risk placeholders and the deterministic permit-to-intent binding.
2//!
3//! These values are synthetic placeholders by design: they let the risk
4//! engine and authority exercise fail-closed control logic with exact domain
5//! types while real sector, depth, volatility, Greeks and borrow masters stay
6//! out of scope. The digest is an unkeyed integrity check over order terms; it
7//! is not a signature and grants no authenticity (see capability records).

50/// Canonical SHA-256 binding of a permit identity to the exact order terms it
51/// authorized. Both the authority (at issuance) and the OMS (at `create`)
52/// compute this same value; a tampered intent or permit field cannot match.
53///
54/// The preimage names the reservation identity fields that the OMS carries in
55/// `PermitRef` plus the complete `OrderIntent`. The risk assessment is not
56/// included because the OMS does not depend on the authority by design: the
57/// policy version is the typed proxy for the assessment context. This is an
58/// integrity digest only — no key, signature, KMS identity or authenticity is
59/// claimed.
60#[must_use]
61pub fn permit_digest(
62    permit_id: u64,
63    policy_version: &str,
64    snapshot_revision: u64,
65    intent: &OrderIntent,
66) -> String {
67    let notional = i128::from(intent.limit_price.get().0) * i128::from(intent.quantity.get());
68    let preimage = serde_json::json!({
69        "domain": "nirnay.permit.intent.v1",
70        "permit_id": permit_id,
71        "policy_version": policy_version,
72        "reservation": {
73            "permit_id": permit_id,
74            "intent_id": intent.intent_id,
75            "account_id": intent.account_id,
76            "strategy_id": intent.strategy_id,
77            "instrument_id": intent.instrument_id,
78            "venue": intent.venue,
79            "side": intent.side,
80            "quantity": intent.quantity,
81            "reserved_notional_paise": notional,
82            "snapshot_revision": snapshot_revision,
83        },
84        "intent": intent,
85    })
86    .to_string();
87    let raw = Sha256::digest(preimage.as_bytes());
88    raw.iter().map(|byte| format!("{byte:02x}")).collect()
89}

91#[cfg(all(test, feature = "market-shaped-tests"))]
92mod tests {
93    use super::permit_digest;
94    use crate::{
95        AccountId, AlgoId, InstrumentId, IntentId, JournalRef, LogicalNanos, OrderIntent, Paise,
96        PricePaise, Product, Quantity, Side, StrategyId, Venue,
97    };
98
99    fn intent() -> OrderIntent {
100        OrderIntent {
101            intent_id: IntentId::new(1).expect("fixture intent id"),
102            account_id: AccountId::new(1).expect("fixture account id"),
103            strategy_id: StrategyId::new(2).expect("fixture strategy id"),
104            strategy_version: "strategy-v1".to_owned(),
105            algo_id: AlgoId::new(3).expect("fixture algo id"),
106            instrument_id: InstrumentId::new("nse.cash.synidx").expect("fixture instrument id"),
107            venue: Venue::Nse,
108            side: Side::Buy,
109            product: Product::Intraday,
110            quantity: Quantity::new(5).expect("fixture quantity"),
111            limit_price: PricePaise::new(Paise(1_000)).expect("fixture price"),
112            reason: "deterministic digest fixture".to_owned(),
113            evidence: vec![JournalRef {
114                sequence: 1,
115                hash_hex: "ab".repeat(32),
116            }],
117            expires_at: LogicalNanos::new(100_000),
118        }
119    }
120
121    #[test]
122    fn permit_digest_is_deterministic_and_binds_every_intent_field() {
123        let baseline = permit_digest(7, "risk-v1", 3, &intent());
124        assert_eq!(
125            baseline,
126            permit_digest(7, "risk-v1", 3, &intent()),
127            "recomputation must be byte-identical"
128        );
129        assert_eq!(baseline.len(), 64, "digest must be SHA-256 hex");
130
131        let mut changed = intent();
132        changed.limit_price = PricePaise::new(Paise(1_001)).expect("changed price");
133        assert_ne!(
134            baseline,
135            permit_digest(7, "risk-v1", 3, &changed),
136            "a tampered price must change the digest"
137        );
138
139        let mut changed = intent();
140        changed.reason.push('!');
141        assert_ne!(
142            baseline,
143            permit_digest(7, "risk-v1", 3, &changed),
144            "a tampered reason must change the digest"
145        );
146
147        assert_ne!(
148            baseline,
149            permit_digest(8, "risk-v1", 3, &intent()),
150            "a different permit id must change the digest"
151        );
152        assert_ne!(
153            baseline,
154            permit_digest(7, "risk-v2", 3, &intent()),
155            "a different policy version must change the digest"
156        );
157    }
158}
