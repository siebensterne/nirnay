1//! Instrument lookup over the canonical India domain.
2//!
3//! There is exactly one instrument identity: `india_domain::InstrumentId`.
4//! Symbols, tokens, ISINs and broker ids are lookup attributes, never
5//! identity. A catalog rejects duplicate identities and invalid instruments
6//! at construction.
7
8use std::collections::BTreeMap;
9
10use jiff::civil::Date;
11
12use crate::DomainError;
13
14pub use india_domain::{
15    CalendarDay, Instrument, InstrumentId, InstrumentType, Isin, Listing, ListingStatus,
16    ListingTerms, OptionKind, Segment, SessionDay, SettlementRegime, TickAssignment,
17    TradingCalendar, Venue,
18};
19
20/// Whether at least one listing is active on `date`.
21#[must_use]
22pub fn is_active_on(instrument: &Instrument, date: Date) -> bool {
23    instrument.listings.iter().any(|listing| {
24        listing.status == ListingStatus::Active
25            && listing.effective_from <= date
26            && listing.effective_to.is_none_or(|end| date <= end)
27    })
28}
29
30/// Point-in-time instrument catalog keyed by canonical identity.
31#[derive(Debug, Clone, Default)]
32pub struct InstrumentCatalog {
33    instruments: BTreeMap<InstrumentId, Instrument>,
34}
35
36impl InstrumentCatalog {
37    pub fn new(instruments: Vec<Instrument>) -> Result<Self, DomainError> {
38        let mut catalog = BTreeMap::new();
39        for instrument in instruments {
40            instrument
41                .validate()
42                .map_err(|error| DomainError::InvalidInstrument(error.to_string()))?;
43            let identity = instrument.instrument_id.clone();
44            if catalog.insert(identity.clone(), instrument).is_some() {
45                return Err(DomainError::DuplicateInstrument(identity.to_string()));
46            }
47        }
48        Ok(Self {
49            instruments: catalog,
50        })
51    }
52
53    /// The instrument only when it is active on `date`.
54    #[must_use]
55    pub fn at(&self, identity: &InstrumentId, date: Date) -> Option<&Instrument> {
56        self.instruments
57            .get(identity)
58            .filter(|instrument| is_active_on(instrument, date))
59    }
60
61    /// The instrument regardless of listing dates. Use for reference data,
62    /// not for execution eligibility.
63    #[must_use]
64    pub fn get(&self, identity: &InstrumentId) -> Option<&Instrument> {
65        self.instruments.get(identity)
66    }
67
68    pub fn iter(&self) -> impl Iterator<Item = &Instrument> {
69        self.instruments.values()
70    }
71
72    #[must_use]
73    pub fn len(&self) -> usize {
74        self.instruments.len()
75    }
76
77    #[must_use]
78    pub fn is_empty(&self) -> bool {
79        self.instruments.is_empty()
80    }
81}
