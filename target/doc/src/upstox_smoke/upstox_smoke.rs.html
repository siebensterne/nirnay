105impl UpstoxClient {
106    fn new(token: AnalyticsToken) -> anyhow::Result<Self> {
107        let http = Client::builder()
108            .timeout(Duration::from_secs(30))
109            // Fail closed on redirects: never let a credential-bearing request
110            // be re-issued against a different host.
111            .redirect(redirect::Policy::none())
112            .user_agent(concat!(
113                "nirnay/upstox-smoke/",
114                env!("CARGO_PKG_VERSION")
115            ))
116            .build()
117            .context("building Upstox HTTP client")?;
118        Ok(Self {
119            http,
120            token: token.0,
121        })
122    }
123
124    fn endpoint(&self, segments: &[&str]) -> anyhow::Result<Url> {
125        let mut url = Url::parse(API_BASE).context("parsing Upstox API base URL")?;
126        {
127            let mut path = url
128                .path_segments_mut()
129                .map_err(|_| anyhow::anyhow!("Upstox API base URL cannot accept path segments"))?;
130            path.pop_if_empty();
131            for segment in segments {
132                path.push(segment);
133            }
134        }
135        Ok(url)
136    }

138    async fn get_json_bytes(&self, url: Url) -> anyhow::Result<Vec<u8>> {
139        let mut response = self
140            .http
141            .get(url)
142            .bearer_auth(&self.token)
143            .header(header::ACCEPT, "application/json")
144            .send()
145            .await
146            .context("sending authenticated Upstox GET request")?;
147
148        let status = response.status();
149        let request_id = response
150            .headers()
151            .get("x-request-id")
152            .and_then(|value| value.to_str().ok())
153            .unwrap_or("unavailable")
154            .to_owned();
155        let retry_after = response
156            .headers()
157            .get(header::RETRY_AFTER)
158            .and_then(|value| value.to_str().ok())
159            .unwrap_or("unavailable")
160            .to_owned();
161
162        ensure!(
163            response.content_length().unwrap_or(0) <= MAX_HTTP_BODY_BYTES as u64,
164            "Upstox response exceeds the {} MiB body limit",
165            MAX_HTTP_BODY_BYTES / (1024 * 1024)
166        );
167        let mut body = Vec::new();
168        while let Some(chunk) = response
169            .chunk()
170            .await
171            .context("reading Upstox response body")?
172        {
173            ensure!(
174                body.len().saturating_add(chunk.len()) <= MAX_HTTP_BODY_BYTES,
175                "Upstox response exceeds body limit"
176            );
177            body.extend_from_slice(&chunk);
178        }
179
180        if !status.is_success() {
181            let detail = match status {
182                StatusCode::UNAUTHORIZED => {
183                    "token is invalid, expired or not authorized for this API"
184                }
185                StatusCode::FORBIDDEN => "account or token does not have access to this API",
186                StatusCode::TOO_MANY_REQUESTS => {
187                    "rate limit exceeded; honour the Retry-After header"
188                }
189                _ => "provider request failed",
190            };
191            bail!(
192                "Upstox HTTP {status}: {detail}; request_id={request_id}; retry_after={retry_after}"
193            );
194        }
195        Ok(body)
196    }

198    async fn historical_candles(
199        &self,
200        instrument_key: &str,
201        from_date: Date,
202        to_date: Date,
203    ) -> anyhow::Result<Vec<u8>> {
204        let url = self.endpoint(&[
205            "v3",

207            instrument_key,

210            &to_date.strftime("%Y-%m-%d").to_string(),

216    async fn full_market_quote(&self, instrument_key: &str) -> anyhow::Result<Vec<u8>> {
217        let mut url = self.endpoint(&["v3", "market-quote", "quotes"])?;
218        url.query_pairs_mut()
219            .append_pair("instrument_key", instrument_key);
220        self.get_json_bytes(url).await
221    }

223    async fn authorize_market_feed(&self) -> anyhow::Result<String> {
224        let url = self.endpoint(&["v3", "feed", "market-data-feed", "authorize"])?;
225        let bytes = self.get_json_bytes(url).await?;
226        let response: WebsocketAuthorization = serde_json::from_slice(&bytes)
227            .context("decoding Upstox WebSocket authorization response")?;
228        ensure!(
229            response.status == "success",
230            "Upstox WebSocket authorization status was not success"
231        );
232        let authorized_uri = Url::parse(&response.data.authorized_redirect_uri)
233            .context("Upstox returned an invalid WebSocket authorization URI")?;
234        ensure!(
235            authorized_uri.scheme() == "wss",
236            "Upstox returned a non-secure WebSocket URI"
237        );
238        let host = authorized_uri.host_str().unwrap_or_default();
239        ensure!(
240            host == "upstox.com" || host.ends_with(".upstox.com"),
241            "Upstox returned a WebSocket host outside upstox.com"
242        );
243        ensure!(
244            authorized_uri.username().is_empty() && authorized_uri.password().is_none(),
245            "Upstox returned WebSocket URI user information"
246        );
247        Ok(authorized_uri.to_string())
248    }

439fn determine_termination_reason(
440    sequence: usize,
441    max_frames: usize,
442    timed_out: bool,
443    remote_close: bool,
444    stream_end: bool,
445) -> TerminationReason {
446    if sequence >= max_frames {
447        return TerminationReason::FrameLimit;
448    }
449    if timed_out {
450        return TerminationReason::Timeout;
451    }
452    if remote_close {
453        return TerminationReason::RemoteClose;
454    }
455    if stream_end {
456        return TerminationReason::StreamEnd;
457    }
458    TerminationReason::StreamEnd
459}

592        !args.instrument_key.trim().is_empty(),

594    );

597        "instrument key has surrounding whitespace"
598    );
599    ensure!(
600        args.from_date <= args.to_date,
601        "from-date must be <= to-date"
602    );
603    let maximum_to_date = args
604        .from_date
605        .checked_add(jiff::Span::new().months(1))
606        .context("could not calculate one-month request boundary")?;
607    ensure!(
608        args.to_date <= maximum_to_date,
609        "one-minute Historical Candle V3 requests are limited to one calendar month; split the range"
610    );
611    ensure!(
612        args.websocket_frames > 0 && args.websocket_frames <= 10_000,
613        "websocket-frames must be in 1..=10000"
614    );
615    ensure!(
616        args.websocket_timeout_secs > 0 && args.websocket_timeout_secs <= 300,
617        "websocket-timeout-secs must be in 1..=300"
618    );
619    Ok(())
620}

622fn validate_historical_candles(raw: &[u8]) -> anyhow::Result<HistoricalValidation> {
623    let value: Value = serde_json::from_slice(raw).context("historical response is not JSON")?;
624    ensure!(
625        value.get("status").and_then(Value::as_str) == Some("success"),
626        "historical response status is not success"
627    );
628    let candles = value
629        .pointer("/data/candles")
630        .and_then(Value::as_array)
631        .context("historical response has no data.candles array")?;
632
633    if candles.is_empty() {
634        return Ok(HistoricalValidation {
635            candle_count: 0,
636            provider_order: CandleOrder::Empty,
637        });
638    }
639
640    let mut previous_timestamp_nanos: Option<i64> = None;
641    let mut observed_order: Option<CandleOrder> = None;
642    for (index, candle) in candles.iter().enumerate() {
643        let row = candle
644            .as_array()
645            .with_context(|| format!("candle {index} is not an array"))?;
646        ensure!(
647            (6..=7).contains(&row.len()),
648            "candle {index} has {} fields; expected 6 or 7",
649            row.len()
650        );
651        let timestamp = row[0]
652            .as_str()
653            .with_context(|| format!("candle {index} timestamp is not a string"))?;
654        let parsed_timestamp = timestamp
655            .parse::<Timestamp>()
656            .with_context(|| format!("candle {index} timestamp is not RFC3339"))?;
657        let timestamp_nanos = i64::try_from(parsed_timestamp.as_nanosecond())
658            .ok()
659            .context("candle timestamp is outside the supported nanosecond range")?;
660        if let Some(previous) = previous_timestamp_nanos {
661            ensure!(
662                timestamp_nanos != previous,
663                "historical candle has duplicate timestamp at row {index}"
664            );
665            let direction = if timestamp_nanos > previous {
666                CandleOrder::Ascending
667            } else {
668                CandleOrder::Descending
669            };
670            match observed_order {
671                None => observed_order = Some(direction),
672                Some(existing) => ensure!(
673                    existing == direction,
674                    "historical candle has mixed or inconsistent timestamp ordering at row {index}"
675                ),
676            }
677        }
678        previous_timestamp_nanos = Some(timestamp_nanos);
679
680        let open = finite_number(&row[1], index, "open")?;
681        let high = finite_number(&row[2], index, "high")?;
682        let low = finite_number(&row[3], index, "low")?;
683        let close = finite_number(&row[4], index, "close")?;
684        ensure!(low <= high, "candle {index} has low > high");
685        ensure!(
686            low <= open && open <= high,
687            "candle {index} open is outside [low, high]"
688        );
689        ensure!(
690            low <= close && close <= high,
691            "candle {index} close is outside [low, high]"
692        );
693        let volume = row[5]
694            .as_i64()
695            .with_context(|| format!("candle {index} volume is not an integer"))?;
696        ensure!(volume >= 0, "candle {index} volume is negative");
697        if row.len() == 7 {
698            let open_interest = row[6]
699                .as_i64()
700                .with_context(|| format!("candle {index} open interest is not an integer"))?;
701            ensure!(
702                open_interest >= 0,
703                "candle {index} open interest is negative"
704            );
705        }
706    }
707    let provider_order = if candles.len() == 1 {
708        CandleOrder::Single
709    } else {
710        observed_order.context("historical candle ordering could not be determined")?
711    };
712    Ok(HistoricalValidation {
713        candle_count: candles.len(),
714        provider_order,
715    })
716}

726fn validate_quote(raw: &[u8], requested_instrument_key: &str) -> anyhow::Result<()> {
727    let value: Value = serde_json::from_slice(raw).context("quote response is not JSON")?;
728    ensure!(
729        value.get("status").and_then(Value::as_str) == Some("success"),
730        "quote response status is not success"
731    );
732    let data = value
733        .get("data")
734        .and_then(Value::as_object)
735        .context("quote response has no data object")?;
736    ensure!(!data.is_empty(), "quote response data object is empty");
737
738    let mut matching: Vec<&Value> = Vec::new();
739    for record in data.values() {
740        let object = record
741            .as_object()
742            .context("quote record is not an object")?;
743        if object.get("instrument_token").and_then(Value::as_str) == Some(requested_instrument_key)
744        {
745            matching.push(record);
746        }
747    }
748    ensure!(
749        matching.len() == 1,
750        "quote response must contain exactly one record for requested instrument"
751    );
752    let record = matching[0]
753        .as_object()
754        .context("quote matched record is not an object")?;
755
756    let instrument_token = record
757        .get("instrument_token")
758        .and_then(Value::as_str)
759        .context("quote record has no instrument_token")?;
760    ensure!(
761        !instrument_token.is_empty(),
762        "quote instrument_token is empty"
763    );
764    let symbol = record
765        .get("symbol")
766        .and_then(Value::as_str)
767        .context("quote record has no symbol")?;
768    ensure!(!symbol.is_empty(), "quote symbol is empty");
769
770    let timestamp = record
771        .get("timestamp")
772        .and_then(Value::as_str)
773        .context("quote record has no timestamp")?;
774    let parsed_timestamp = timestamp
775        .parse::<Timestamp>()
776        .context("quote timestamp is not RFC3339")?;
777    ensure!(
778        i64::try_from(parsed_timestamp.as_nanosecond()).is_ok(),
779        "quote timestamp is outside the supported range"
780    );
781
782    if let Some(last_trade_raw) = record.get("last_trade_time") {
783        let last_trade_str = last_trade_raw
784            .as_str()
785            .context("quote last_trade_time is not a string")?;
786        let last_trade_ms: i64 = last_trade_str
787            .parse()
788            .context("quote last_trade_time is not decimal milliseconds")?;
789        ensure!(last_trade_ms >= 0, "quote last_trade_time is negative");
790    }
791
792    let last_price_value = record
793        .get("last_price")
794        .context("quote record has no last_price")?;
795    non_negative_finite(last_price_value, "last_price")?;
796    let volume_value = record.get("volume").context("quote record has no volume")?;
797    non_negative_integer(volume_value, "volume")?;
798
799    let ohlc = record
800        .get("ohlc")
801        .and_then(Value::as_object)
802        .context("quote record has no ohlc object")?;
803    let open = non_negative_finite(&ohlc["open"], "ohlc.open")?;
804    let high = non_negative_finite(&ohlc["high"], "ohlc.high")?;
805    let low = non_negative_finite(&ohlc["low"], "ohlc.low")?;
806    let close = non_negative_finite(&ohlc["close"], "ohlc.close")?;
807    ensure!(low <= high, "quote ohlc has low > high");
808    ensure!(
809        low <= open && open <= high,
810        "quote ohlc open is outside [low, high]"
811    );
812    ensure!(
813        low <= close && close <= high,
814        "quote ohlc close is outside [low, high]"
815    );
816    let _ohlc_volume = non_negative_integer(&ohlc["volume"], "ohlc.volume")?;
817    let _ohlc_ts = non_negative_integer(&ohlc["ts"], "ohlc.ts")?;
818
819    let depth = record
820        .get("depth")
821        .and_then(Value::as_object)
822        .context("quote record has no depth object")?;
823    for side in ["buy", "sell"] {
824        let levels = depth
825            .get(side)
826            .and_then(Value::as_array)
827            .with_context(|| format!("quote depth {side} is not an array"))?;
828        ensure!(
829            levels.len() <= 5,
830            "quote depth {side} has more than five levels"
831        );
832        for (index, level) in levels.iter().enumerate() {
833            let level_object = level
834                .as_object()
835                .with_context(|| format!("quote depth {side} level {index} is not an object"))?;
836            let _quantity = non_negative_integer(&level_object["quantity"], "depth.quantity")?;
837            let _price = non_negative_finite(&level_object["price"], "depth.price")?;
838            let _orders = non_negative_integer(&level_object["orders"], "depth.orders")?;
839        }
840    }
841
842    Ok(())
843}

897fn archive_and_diagnose_frame<R: std::io::Write, D: std::io::Write>(
898    payload: &[u8],
899    sequence: usize,
900    received_at_utc: String,
901    instrument_key: &str,
902    raw_writer: &mut R,
903    diagnostic_writer: &mut D,
904) -> anyhow::Result<(FrameDecodeOutcome, ProtobufDecodeCounters)> {
905    let mut counters = ProtobufDecodeCounters::default();
906    ensure!(
907        payload.len() <= MAX_WS_FRAME_BYTES,
908        "Upstox WebSocket frame exceeds 4 MiB limit"
909    );
910    let byte_len = payload.len();
911    let sha256 = sha256_hex(payload);
912    let encoded = BASE64.encode(payload);
913    let row = ArchivedBinaryFrame {
914        schema_version: 1,
915        sequence,
916        received_at_utc: received_at_utc.clone(),
917        byte_len,
918        sha256: sha256.clone(),
919        encoding: "protobuf-v3-base64",
920        payload_base64: &encoded,
921    };
922    serde_json::to_writer(&mut *raw_writer, &row).context("serializing captured feed frame")?;
923    raw_writer
924        .write_all(b"\n")
925        .context("writing captured feed frame")?;
926    raw_writer.flush().context("flushing captured feed frame")?;
927
928    let outcome_record = match decode_feed_response_diagnostic(payload) {
929        Ok(diagnostic) => {
930            counters.decoded_frame_count += 1;
931            match diagnostic.response_type {
932                ProtoFeedResponseType::InitialFeed => counters.initial_feed_count += 1,
933                ProtoFeedResponseType::LiveFeed => counters.live_feed_count += 1,
934                ProtoFeedResponseType::MarketInfo => counters.market_info_count += 1,
935                ProtoFeedResponseType::Unknown(_) => counters.unknown_type_count += 1,
936            }
937            if diagnostic.feed_keys.iter().any(|k| k == instrument_key) {
938                counters.requested_instrument_feed_frames += 1;
939                if diagnostic.response_type == ProtoFeedResponseType::LiveFeed {
940                    counters.requested_instrument_live_feed_frames += 1;
941                }
942            }
943            DiagnosticOutcome::Decoded {
944                response_type_raw: diagnostic.response_type_raw,
945                response_type: response_type_name(diagnostic.response_type).to_owned(),
946                current_ts_raw: diagnostic.current_ts,
947                has_market_info: diagnostic.has_market_info,
948                feed_keys: diagnostic.feed_keys.clone(),
949                feeds: diagnostic
950                    .feeds
951                    .iter()
952                    .map(|f| DiagnosticFeed {
953                        key: f.key.clone(),
954                        kind: frame_kind_name(f.kind).to_owned(),
955                        request_mode_raw: f.request_mode_raw,
956                    })
957                    .collect(),
958            }
959        }
960        Err(error) => {
961            counters.decode_error_count += 1;
962            DiagnosticOutcome::DecodeError {
963                error: error.to_string(),
964            }
965        }
966    };
967    let outcome = match &outcome_record {
968        DiagnosticOutcome::Decoded { .. } => FrameDecodeOutcome::Decoded,
969        DiagnosticOutcome::DecodeError { .. } => FrameDecodeOutcome::DecodeError,
970    };
971    let diagnostic = DiagnosticRecord {
972        schema_version: 1,
973        sequence,
974        received_at_utc,
975        byte_len,
976        sha256,
977        outcome: outcome_record,
978    };
979    serde_json::to_writer(&mut *diagnostic_writer, &diagnostic)
980        .context("serializing diagnostic frame")?;
981    diagnostic_writer
982        .write_all(b"\n")
983        .context("writing diagnostic frame")?;
984    diagnostic_writer
985        .flush()
986        .context("flushing diagnostic frame")?;
987    Ok((outcome, counters))
988}

991    let flush_result = writer.flush().context("flushing capture");
992    let sync_result = writer.get_ref().sync_all().context("syncing capture");
993    flush_result?;
994    sync_result?;
995    Ok(())
996}

998async fn capture_ltpc_frames(
999    authorized_uri: &str,
1000    instrument_key: &str,
1001    output: &Path,
1002    diagnostics_output: &Path,
1003    max_frames: usize,
1004    max_wait: Duration,
1005) -> anyhow::Result<(bool, usize, ProtobufDecodeCounters, CaptureObservability)> {
1006    let (mut socket, _response) = connect_async(authorized_uri).await.map_err(|_| {
1007        anyhow::anyhow!(
1008            "Upstox WebSocket handshake failed; the one-time authorization URI is redacted"
1009        )
1010    })?;
1011    let subscription = subscription_frame(instrument_key)?;
1012    socket
1013        .send(Message::Binary(subscription.into()))
1014        .await
1015        .context("sending binary LTPC subscription")?;
1016    let subscription_sent_at_utc = Timestamp::now()
1017        .strftime("%Y-%m-%dT%H:%M:%S%.3fZ")
1018        .to_string();
1019    let capture_started_at_utc = subscription_sent_at_utc.clone();
1020    let file = OpenOptions::new()
1021        .write(true)
1022        .create_new(true)
1023        .open(output)
1024        .with_context(|| format!("creating WebSocket capture {}", output.display()))?;
1025    let mut writer = BufWriter::new(file);
1026    let diag_file = OpenOptions::new()
1027        .write(true)
1028        .create_new(true)
1029        .open(diagnostics_output)
1030        .with_context(|| {
1031            format!(
1032                "creating diagnostics capture {}",
1033                diagnostics_output.display()
1034            )
1035        })?;
1036    let mut diag_writer = BufWriter::new(diag_file);
1037    let mut counters = ProtobufDecodeCounters::default();
1038    let deadline = Instant::now() + max_wait;
1039    let mut sequence = 0usize;
1040    let mut remote_close = false;
1041    let mut stream_end = false;
1042    let mut timed_out = false;
1043
1044    let capture_result: anyhow::Result<()> = async {
1045        while sequence < max_frames && Instant::now() < deadline {
1046            let next = timeout_at(deadline, socket.next()).await;
1047            match next {
1048                Err(_) => {
1049                    timed_out = true;
1050                    break;
1051                }
1052                Ok(None) => {
1053                    stream_end = true;
1054                    break;
1055                }
1056                Ok(Some(Err(error))) => {
1057                    return Err(error).context("reading Upstox WebSocket frame");
1058                }
1059                Ok(Some(Ok(Message::Binary(payload)))) => {
1060                    sequence += 1;
1061                    let received = Timestamp::now()
1062                        .strftime("%Y-%m-%dT%H:%M:%S%.3fZ")
1063                        .to_string();
1064                    let (_outcome, frame_counters) = archive_and_diagnose_frame(
1065                        payload.as_ref(),
1066                        sequence,
1067                        received,
1068                        instrument_key,
1069                        &mut writer,
1070                        &mut diag_writer,
1071                    )?;
1072                    counters.decoded_frame_count += frame_counters.decoded_frame_count;
1073                    counters.decode_error_count += frame_counters.decode_error_count;
1074                    counters.initial_feed_count += frame_counters.initial_feed_count;
1075                    counters.live_feed_count += frame_counters.live_feed_count;
1076                    counters.market_info_count += frame_counters.market_info_count;
1077                    counters.unknown_type_count += frame_counters.unknown_type_count;
1078                    counters.requested_instrument_feed_frames +=
1079                        frame_counters.requested_instrument_feed_frames;
1080                    counters.requested_instrument_live_feed_frames +=
1081                        frame_counters.requested_instrument_live_feed_frames;
1082                }
1083                Ok(Some(Ok(Message::Ping(payload)))) => {
1084                    socket
1085                        .send(Message::Pong(payload))
1086                        .await
1087                        .context("responding to WebSocket ping")?;
1088                }
1089                Ok(Some(Ok(Message::Close(_)))) => {
1090                    remote_close = true;
1091                    break;
1092                }
1093                Ok(Some(Ok(Message::Text(_))))
1094                | Ok(Some(Ok(Message::Pong(_))))
1095                | Ok(Some(Ok(Message::Frame(_)))) => {}
1096            }
1097        }
1098        if !timed_out && !remote_close && !stream_end && sequence >= max_frames {
1099            // Frame-limit exit is distinguished by the post-loop check below.
1100        }
1101        Ok(())
1102    }
1103    .await;
1104
1105    let raw_finalize = flush_and_sync(&mut writer);
1106    let diagnostic_finalize = flush_and_sync(&mut diag_writer);
1107
1108    capture_result?;
1109    raw_finalize?;
1110    diagnostic_finalize?;
1111    let capture_completed_at_utc = Timestamp::now()
1112        .strftime("%Y-%m-%dT%H:%M:%S%.3fZ")
1113        .to_string();
1114    let termination_reason =
1115        determine_termination_reason(sequence, max_frames, timed_out, remote_close, stream_end);
1116    let _ = socket.close(None).await;
1117    Ok((
1118        true,
1119        sequence,
1120        counters,
1121        CaptureObservability {
1122            subscription_sent_at_utc,
1123            capture_started_at_utc,
1124            capture_completed_at_utc,
1125            termination_reason,
1126        },
1127    ))
1128}
