6//! Repository task runner: a typed, discoverable CLI over the project gates.
7//!
8//! The task set is a clap subcommand enum: one source of truth for names,
9//! help text and delegation. Every gate runs through [`Cmd`] with an explicit
10//! argument vector — never a shell — and its exit status is propagated so CI
11//! sees the gate's own result. The first failing step stops the run; nothing
12//! is skipped silently.

42/// A single process invocation with explicit arguments and optional environment.
43#[derive(Debug, Clone, PartialEq, Eq)]
44struct Cmd {
45    /// Executable name or path.
46    program: &'static str,
47    /// Arguments passed verbatim; no shell parsing.
48    args: Vec<&'static str>,
49    /// Extra environment variables.
50    env: Vec<(&'static str, &'static str)>,
51}
52
53impl Cmd {
54    /// Construct a command with no environment overrides.
55    fn new(program: &'static str, args: &'static [&'static str]) -> Self {
56        Self {
57            program,
58            args: args.to_vec(),
59            env: Vec::new(),
60        }
61    }
62
63    /// Add an environment variable.
64    fn with_env(mut self, key: &'static str, value: &'static str) -> Self {
65        self.env.push((key, value));
66        self
67    }
68
69    /// Human-readable rendering for progress output.
70    fn line(&self) -> String {
71        let mut parts = vec![self.program.to_owned()];
72        parts.extend(self.args.iter().map(|arg| (*arg).to_owned()));
73        parts.join(" ")
74    }
75
76    /// Execute the command from the workspace root.
77    fn run(&self, root: &Path) -> Result<ExitStatus> {
78        let status = Command::new(self.program)
79            .args(&self.args)
80            .envs(self.env.iter().copied())
81            .current_dir(root)
82            .status()
83            .with_context(|| format!("running {}", self.line()))?;
84        Ok(status)
85    }
86}

88/// Repository gates exposed as clap subcommands.

90enum Task {
91    /// Run the full policy gate suite via the repository's single registry (make check).
92    Ci,
93    /// Run the full assurance superset (polish, update and upgrade gates).
94    Assure,
95    /// Run every repository policy gate via check_repo.py.
96    Check,
97    /// Check the architecture dependency graph and manifest hygiene.
98    Arch,
99    /// Check dependency waivers.
100    Waivers,
101    /// Check claims and their falsifiers.
102    Claims,
103    /// Run every source and standards style gate.
104    Style,
105    /// Check Rust formatting.
106    Fmt,
107    /// Run Clippy with warnings denied.
108    Lint,
109    /// Run all-target and doc tests.
110    Test,
111    /// Build rustdoc with warnings and broken links denied.
112    Docs,

117/// Parse errors are handled by clap; this enum is the single source of truth.
118#[derive(Debug, Parser)]
119#[command(name = "xtask", version, about, long_about = None)]
120struct Cli {
121    /// Gate to run.
122    #[command(subcommand)]
123    task: Task,
124}

126impl Task {
127    /// Commands for this task, in execution order. Discovery tasks have none.
128    fn commands(self) -> Vec<Cmd> {
129        match self {
130            Task::Ci => vec![Cmd::new("make", &["check"])],
131            Task::Assure => vec![Cmd::new("scripts/assurance/polish.sh", &[])],
132            Task::Check => vec![script("check")],
133            Task::Arch => vec![script("arch")],
134            Task::Waivers => vec![script("waivers")],
135            Task::Claims => vec![script("claims")],
136            Task::Style => vec![Cmd::new("make", &["style"])],
137            Task::Fmt => vec![Cmd::new("cargo", &["fmt", "--all", "--", "--check"])],
138            Task::Lint => vec![Cmd::new(
139                "cargo",
140                &[
141                    "clippy",
142                    "--workspace",
143                    "--all-targets",
144                    "--locked",
145                    "--",
146                    "-D",
147                    "warnings",
148                ],
149            )],
150            Task::Test => vec![
151                Cmd::new(
152                    "cargo",
153                    &["test", "--workspace", "--all-targets", "--locked"],
154                ),
155                Cmd::new("cargo", &["test", "--workspace", "--doc", "--locked"]),
156            ],
157            Task::Docs => vec![
158                Cmd::new("cargo", &["doc", "--no-deps", "--workspace", "--locked"]).with_env(
159                    "RUSTDOCFLAGS",
160                    "-D warnings -D rustdoc::broken_intra_doc_links",
161                ),
162            ],
163            Task::List => Vec::new(),
164        }
165    }
166}

168/// Build the typed invocation of the repository policy script.
169fn script(argument: &'static str) -> Cmd {

177/// Locate the workspace root from the compiled crate location.
178fn workspace_root() -> Result<PathBuf> {
179    let manifest = Path::new(env!("CARGO_MANIFEST_DIR"));
180    let root = manifest
181        .parent()
182        .context("xtask must live directly under the workspace root")?
183        .to_path_buf();
184    ensure!(
185        root.join("Cargo.toml").is_file(),
186        "workspace manifest not found under {}",
187        root.display()
188    );
189    Ok(root)
190}

192/// Task names taken from the clap command definition, so they cannot drift.
193fn task_names() -> Vec<String> {
194    let mut names: Vec<String> = Cli::command()
195        .get_subcommands()
196        .map(|subcommand| subcommand.get_name().to_owned())
197        .collect();
198    names.sort_unstable();
199    names
200}

202/// Map a child status to this process's exit code.
203///
204/// A successful child stays successful. A failed child keeps its own code when
205/// it fits in `u8`; a signal termination or out-of-range code becomes the
206/// generic failure code. Keeping this mapping pure makes propagation testable.
207fn exit_code_for(status: ExitStatus) -> ExitCode {
208    if status.success() {
209        return ExitCode::SUCCESS;
210    }
211    status
212        .code()
213        .and_then(|value| u8::try_from(value).ok())
214        .map_or(ExitCode::FAILURE, ExitCode::from)
215}
216
217/// Run a task's commands in order, stopping at the first failure.
218fn run_task(root: &Path, task: Task) -> Result<ExitCode> {
219    let commands = task.commands();
220    ensure!(
221        !commands.is_empty(),
222        "task has no executable commands; use `list` for discovery"
223    );
224    for command in commands {
225        let line = command.line();
226        let status = command.run(root)?;
227        let code = exit_code_for(status);
228        println!("[{code:?}] {line}");
229        if code != ExitCode::SUCCESS {
230            return Ok(code);
231        }
232    }
233    Ok(ExitCode::SUCCESS)
234}
235
236fn run() -> Result<ExitCode> {
237    let cli = Cli::parse();
238    let root = workspace_root()?;
239    match cli.task {
240        Task::List => {
241            let output = stdout();
242            let mut writer = BufWriter::new(output.lock());
243            for name in task_names() {
244                writeln!(writer, "{name}").context("writing the task list")?;
245            }
246            writer.flush().context("flushing the task list")?;
247            Ok(ExitCode::SUCCESS)
248        }
249        task => run_task(&root, task),
250    }
251}
252
253fn main() -> ExitCode {
254    match run() {
255        Ok(code) => code,
256        Err(error) => {
257            eprintln!("xtask: {error:#}");
258            ExitCode::FAILURE
259        }
260    }
261}
