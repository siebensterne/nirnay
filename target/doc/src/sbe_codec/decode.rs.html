1//! Challenge SBE wire decoder, encoder and canonical dump formatter.
2//!
3//! All multi-byte integers are little-endian. Every structural read is bounded
4//! and fails closed; no read falls back to a default value.
5
6use std::fmt;

8use thiserror::Error;

10/// Schema id declared by the frozen challenge schema.
11pub const SCHEMA_ID: u16 = 1;
12/// Schema version declared by the frozen challenge schema.
13pub const SCHEMA_VERSION: u16 = 2;
14/// Full order-book snapshot template id.
15pub const SNAPSHOT_TEMPLATE: u16 = 1;
16/// Single price-level update template id.
17pub const UPDATE_TEMPLATE: u16 = 2;
18
19/// Fixed SBE message header size.
20pub const HEADER_BYTES: usize = 8;
21/// Fixed snapshot body block size.
22pub const SNAPSHOT_BLOCK_BYTES: usize = 16;
23/// Fixed update body block size.
24pub const UPDATE_BLOCK_BYTES: usize = 33;
25/// Repeating-group dimension header size.
26pub const GROUP_HEADER_BYTES: usize = 4;
27/// One price/quantity group entry size.
28pub const ENTRY_BYTES: usize = 16;
29/// Smallest aligned capture chunk the retained harness accepts.
30pub const MIN_CHUNK_BYTES: usize = 1024;
31
32/// Fail-closed decode and encode errors.
33#[derive(Debug, Clone, PartialEq, Eq, Error)]
34pub enum SbeError {
35    #[error("truncated {0}")]
36    Truncated(&'static str),
37    #[error("unknown schema id {0}")]
38    UnknownSchemaId(u16),
39    #[error("unknown schema version {0}")]
40    UnknownSchemaVersion(u16),
41    #[error("unknown template id {0}")]
42    UnknownTemplate(u16),
43    #[error("block length mismatch for template {template}: expected {expected} got {actual}")]
44    BlockLengthMismatch {
45        template: u16,
46        expected: usize,
47        actual: usize,
48    },
49    #[error("invalid group entry block length {0}")]
50    InvalidGroupEntryLength(u16),
51    #[error("invalid side encoding {0}")]
52    InvalidSide(u8),
53    #[error("chunk size {0} is below the minimum")]
54    ChunkTooSmall(usize),
55    #[error("message tears across an aligned chunk boundary")]
56    MessageTear,
57    #[error("symbol is not valid UTF-8")]
58    InvalidSymbol,
59    #[error("symbol length exceeds the uint32 field")]
60    SymbolTooLong,
61    #[error("group size exceeds the uint16 field")]
62    GroupTooLarge,
63    #[error("update message is missing {0}")]
64    MissingUpdateField(&'static str),
65}

110/// A decoded message plus the exact number of bytes it consumed.
111#[derive(Debug, Clone, PartialEq, Eq)]
112pub struct DecodeResult {
113    pub message: SbeMessage,
114    pub bytes_consumed: usize,
115}

117fn need(bytes: &[u8], offset: usize, length: usize, what: &'static str) -> Result<(), SbeError> {
118    if offset > bytes.len() || length > bytes.len() - offset {
119        return Err(SbeError::Truncated(what));
120    }
121    Ok(())
122}
123
124fn read_u16(bytes: &[u8], offset: usize) -> Result<u16, SbeError> {
125    need(bytes, offset, 2, "u16")?;
126    let slice: [u8; 2] = bytes[offset..offset + 2]
127        .try_into()
128        .map_err(|_| SbeError::Truncated("u16"))?;
129    Ok(u16::from_le_bytes(slice))
130}
131
132fn read_u32(bytes: &[u8], offset: usize) -> Result<u32, SbeError> {
133    need(bytes, offset, 4, "u32")?;
134    let slice: [u8; 4] = bytes[offset..offset + 4]
135        .try_into()
136        .map_err(|_| SbeError::Truncated("u32"))?;
137    Ok(u32::from_le_bytes(slice))
138}
139
140fn read_u64(bytes: &[u8], offset: usize) -> Result<u64, SbeError> {
141    need(bytes, offset, 8, "u64")?;
142    let slice: [u8; 8] = bytes[offset..offset + 8]
143        .try_into()
144        .map_err(|_| SbeError::Truncated("u64"))?;
145    Ok(u64::from_le_bytes(slice))
146}
147
148fn read_group(bytes: &[u8], pos: &mut usize) -> Result<Vec<SbeLevel>, SbeError> {
149    need(bytes, *pos, GROUP_HEADER_BYTES, "group header")?;
150    let entry_block = read_u16(bytes, *pos)?;
151    let count = read_u16(bytes, *pos + 2)?;
152    if entry_block as usize != ENTRY_BYTES {
153        return Err(SbeError::InvalidGroupEntryLength(entry_block));
154    }
155    *pos += GROUP_HEADER_BYTES;
156    let mut levels = Vec::with_capacity(count as usize);
157    for _ in 0..count {
158        need(bytes, *pos, ENTRY_BYTES, "group entry")?;
159        let price = read_u64(bytes, *pos)? as i64;
160        let quantity = read_u64(bytes, *pos + 8)? as i64;
161        levels.push(SbeLevel::new(price, quantity));
162        *pos += ENTRY_BYTES;
163    }
164    Ok(levels)
165}
166
167fn read_string(bytes: &[u8], pos: &mut usize) -> Result<String, SbeError> {
168    need(bytes, *pos, 4, "var-data length")?;
169    let length = read_u32(bytes, *pos)? as usize;
170    *pos += 4;
171    need(bytes, *pos, length, "var-data")?;
172    let text =
173        std::str::from_utf8(&bytes[*pos..*pos + length]).map_err(|_| SbeError::InvalidSymbol)?;
174    *pos += length;
175    Ok(text.to_owned())
176}

178fn append_u16(out: &mut Vec<u8>, value: u16) {
179    out.extend_from_slice(&value.to_le_bytes());
180}
181
182fn append_u32(out: &mut Vec<u8>, value: u32) {
183    out.extend_from_slice(&value.to_le_bytes());
184}
185
186fn append_u64(out: &mut Vec<u8>, value: u64) {
187    out.extend_from_slice(&value.to_le_bytes());
188}
189
190fn append_group(out: &mut Vec<u8>, levels: &[SbeLevel]) -> Result<(), SbeError> {
191    if levels.len() > u16::MAX as usize {
192        return Err(SbeError::GroupTooLarge);
193    }
194    append_u16(out, ENTRY_BYTES as u16);
195    append_u16(out, levels.len() as u16);
196    for level in levels {
197        append_u64(out, level.price as u64);
198        append_u64(out, level.quantity as u64);
199    }
200    Ok(())
201}
202
203fn append_symbol(out: &mut Vec<u8>, symbol: &str) -> Result<(), SbeError> {
204    let length = u32::try_from(symbol.len()).map_err(|_| SbeError::SymbolTooLong)?;
205    append_u32(out, length);
206    out.extend_from_slice(symbol.as_bytes());
207    Ok(())
208}

210fn check_block(template_id: u16, expected: usize, actual: usize) -> Result<(), SbeError> {
211    if actual == expected {
212        Ok(())
213    } else {
214        Err(SbeError::BlockLengthMismatch {
215            template: template_id,
216            expected,
217            actual,
218        })
219    }
220}

222/// Decode one message at `offset`.
223pub fn decode_message(buffer: &[u8], offset: usize) -> Result<DecodeResult, SbeError> {
224    need(buffer, offset, HEADER_BYTES, "message header")?;
225    let block_length = read_u16(buffer, offset)? as usize;
226    let template_id = read_u16(buffer, offset + 2)?;
227    let schema_id = read_u16(buffer, offset + 4)?;
228    let schema_version = read_u16(buffer, offset + 6)?;
229    if schema_id != SCHEMA_ID {
230        return Err(SbeError::UnknownSchemaId(schema_id));
231    }
232    if schema_version != SCHEMA_VERSION {
233        return Err(SbeError::UnknownSchemaVersion(schema_version));
234    }
235    let body = offset + HEADER_BYTES;
236    let mut message = SbeMessage::new(template_id);
237
238    if template_id == SNAPSHOT_TEMPLATE {
239        check_block(template_id, SNAPSHOT_BLOCK_BYTES, block_length)?;
240        need(buffer, body, block_length, "snapshot block")?;
241        message.received_at_us = read_u64(buffer, body)?;
242        message.exchange_timestamp_us = read_u64(buffer, body + 8)?;
243        let mut pos = body + block_length;
244        message.bids = read_group(buffer, &mut pos)?;
245        message.asks = read_group(buffer, &mut pos)?;
246        message.symbol = read_string(buffer, &mut pos)?;
247        return Ok(DecodeResult {
248            message,
249            bytes_consumed: pos - offset,
250        });
251    }
252
253    if template_id == UPDATE_TEMPLATE {
254        check_block(template_id, UPDATE_BLOCK_BYTES, block_length)?;
255        need(buffer, body, block_length, "update block")?;
256        message.received_at_us = read_u64(buffer, body)?;
257        message.exchange_timestamp_us = read_u64(buffer, body + 8)?;
258        message.price_level = Some(read_u64(buffer, body + 16)? as i64);
259        message.quantity = Some(read_u64(buffer, body + 24)? as i64);
260        let side = buffer[body + 32];
261        if side > 1 {
262            return Err(SbeError::InvalidSide(side));
263        }
264        message.side = Some(side);
265        let mut pos = body + block_length;
266        message.symbol = read_string(buffer, &mut pos)?;
267        return Ok(DecodeResult {
268            message,
269            bytes_consumed: pos - offset,
270        });
271    }
272
273    Err(SbeError::UnknownTemplate(template_id))
274}

276/// Encode one message to its canonical bytes.
277pub fn encode_message(message: &SbeMessage) -> Result<Vec<u8>, SbeError> {
278    let mut out = Vec::new();
279    match message.template_id {
280        SNAPSHOT_TEMPLATE => {
281            append_u16(&mut out, SNAPSHOT_BLOCK_BYTES as u16);
282            append_u16(&mut out, SNAPSHOT_TEMPLATE);
283            append_u16(&mut out, SCHEMA_ID);
284            append_u16(&mut out, SCHEMA_VERSION);
285            append_u64(&mut out, message.received_at_us);
286            append_u64(&mut out, message.exchange_timestamp_us);
287            append_group(&mut out, &message.bids)?;
288            append_group(&mut out, &message.asks)?;
289            append_symbol(&mut out, &message.symbol)?;
290        }
291        UPDATE_TEMPLATE => {
292            let price = message
293                .price_level
294                .ok_or(SbeError::MissingUpdateField("price_level"))?;
295            let quantity = message
296                .quantity
297                .ok_or(SbeError::MissingUpdateField("quantity"))?;
298            let side = message.side.ok_or(SbeError::MissingUpdateField("side"))?;
299            if side > 1 {
300                return Err(SbeError::InvalidSide(side));
301            }
302            append_u16(&mut out, UPDATE_BLOCK_BYTES as u16);
303            append_u16(&mut out, UPDATE_TEMPLATE);
304            append_u16(&mut out, SCHEMA_ID);
305            append_u16(&mut out, SCHEMA_VERSION);
306            append_u64(&mut out, message.received_at_us);
307            append_u64(&mut out, message.exchange_timestamp_us);
308            append_u64(&mut out, price as u64);
309            append_u64(&mut out, quantity as u64);
310            out.push(side);
311            append_symbol(&mut out, &message.symbol)?;
312        }
313        other => return Err(SbeError::UnknownTemplate(other)),
314    }
315    Ok(out)
316}

318/// Decode every message in an aligned capture.
319pub fn decode_capture(bytes: &[u8], chunk_size: usize) -> Result<Vec<SbeMessage>, SbeError> {
320    if chunk_size < MIN_CHUNK_BYTES {
321        return Err(SbeError::ChunkTooSmall(chunk_size));
322    }
323    let mut messages = Vec::new();
324    let mut offset = 0;
325    while offset < bytes.len() {
326        let mut chunk_end = offset + (chunk_size - offset % chunk_size);
327        if chunk_end > bytes.len() {
328            chunk_end = bytes.len();
329        }
330        if offset + HEADER_BYTES > chunk_end {
331            if offset < bytes.len()
332                && offset + HEADER_BYTES <= bytes.len()
333                && chunk_end == bytes.len()
334            {
335                break;
336            }
337            offset = chunk_end;
338            continue;
339        }
340        let block_length = read_u16(bytes, offset)?;
341        let template_id = read_u16(bytes, offset + 2)?;
342        if template_id == 0 && block_length == 0 {
343            offset = chunk_end;
344            continue;
345        }
346        let result = decode_message(bytes, offset)?;
347        if offset + result.bytes_consumed > chunk_end && chunk_end != bytes.len() {
348            return Err(SbeError::MessageTear);
349        }
350        messages.push(result.message);
351        offset += result.bytes_consumed;
352        if offset >= chunk_end {
353            offset = chunk_end;
354        }
355    }
356    Ok(messages)
357}

359/// Format the canonical one-line dump shared with the C++ decoder.
360pub fn format_dump_line(message: &SbeMessage) -> Result<String, SbeError> {
361    match message.template_id {
362        SNAPSHOT_TEMPLATE => {
363            let bids = message
364                .bids
365                .iter()
366                .map(|level| format!("{}:{}", level.price, level.quantity))
367                .collect::<Vec<_>>()
368                .join(",");
369            let asks = message
370                .asks
371                .iter()
372                .map(|level| format!("{}:{}", level.price, level.quantity))
373                .collect::<Vec<_>>()
374                .join(",");
375            Ok(format!(
376                "snapshot {}:{}:{}:bids={}:asks={}",
377                message.symbol, message.received_at_us, message.exchange_timestamp_us, bids, asks
378            ))
379        }
380        UPDATE_TEMPLATE => {
381            let side = message.side.ok_or(SbeError::MissingUpdateField("side"))?;
382            let price = message
383                .price_level
384                .ok_or(SbeError::MissingUpdateField("price_level"))?;
385            let quantity = message
386                .quantity
387                .ok_or(SbeError::MissingUpdateField("quantity"))?;
388            Ok(format!(
389                "update {}:{}:{}:side={}:price={}:qty={}",
390                message.symbol,
391                message.received_at_us,
392                message.exchange_timestamp_us,
393                side,
394                price,
395                quantity
396            ))
397        }
398        other => Err(SbeError::UnknownTemplate(other)),
399    }
400}
