1//! Deterministic rational EMA crossover strategy.
2//!
3//! This is a systems fixture, not validated alpha. It emits intents only and
4//! never accesses cash, OMS, risk, broker state or positions beyond the
5//! caller-supplied context. Arithmetic uses exact rationals with an explicit
6//! rounding rule, so a Python `Fraction` implementation can reproduce it.
7
8use crate::{
9    artifact::FrozenArtifact, //
10    error::StrategyError,
11    rational::Rational,
12};
13
14pub const SUPPORTED_RULE: &str = "rational_ema_cross_v1";
15
16/// Narrow strategy interface. Strategies propose; they never execute.
17pub trait Strategy {
18    type Intent;
19    fn on_event(&mut self, event: &StrategyEvent) -> Result<Option<Self::Intent>, StrategyError>;
20}
21
22/// The only inputs a strategy may observe.
23#[derive(Clone, Debug, PartialEq, Eq, serde::Serialize)]
24pub struct StrategyEvent {
25    pub observation_sequence: u64,
26    pub instrument_id: String,
27    pub midpoint: Rational,
28    pub best_bid: i64,
29    pub best_ask: i64,
30    pub position_quantity: i64,
31    pub has_open_order: bool,
32}
33
34#[derive(Clone, Debug, PartialEq, Eq)]
35pub struct RuntimeIntent {
36    pub intent_id: String,
37    pub candidate_id: String,
38    pub instrument_id: String,
39    pub side: String,
40    pub quantity: i64,
41    pub limit_price_paise: i64,
42}
43
44pub struct RationalEmaCross {
45    candidate_id: String,
46    universe: Vec<String>,
47    fast: i64,
48    slow: i64,
49    quantity: i64,
50    min_observations: u32,
51    aggressive_limit_bps: i64,
52    fast_ema: Option<Rational>,
53    slow_ema: Option<Rational>,
54    previous_fast: Option<Rational>,
55    previous_slow: Option<Rational>,
56    observations: u32,
57    intent_sequence: u32,
58    last_sequence: Option<u64>,
59}
60
61impl RationalEmaCross {
62    /// Construct only from a gated frozen artifact plus bounded parameters.
63    pub fn from_artifact(
64        artifact: &FrozenArtifact,
65        fast: i64,
66        slow: i64,
67        quantity: i64,
68        min_observations: u32,
69        aggressive_limit_bps: i64,
70    ) -> Result<Self, StrategyError> {
71        // Re-gate here: a struct-literal artifact must not bypass load().
72        artifact
73            .validate_runtime(SUPPORTED_RULE)
74            .map_err(|error| StrategyError::ArtifactRejected(error.to_string()))?;
75        if fast < 2
76            || slow <= fast
77            || quantity <= 0
78            || aggressive_limit_bps < 0
79            || min_observations < slow as u32
80        {
81            return Err(StrategyError::InvalidConfiguration);
82        }
83        Ok(Self {
84            candidate_id: artifact.candidate_id.clone(),
85            universe: artifact.universe.clone(),
86            fast,
87            slow,
88            quantity,
89            min_observations,
90            aggressive_limit_bps,
91            fast_ema: None,
92            slow_ema: None,
93            previous_fast: None,
94            previous_slow: None,
95            observations: 0,
96            intent_sequence: 0,
97            last_sequence: None,
98        })
99    }
100
101    fn update(
102        previous: Option<Rational>,
103        value: Rational,
104        span: i64,
105    ) -> Result<Rational, StrategyError> {
106        let Some(previous) = previous else {
107            return Ok(value);
108        };
109        let alpha = Rational::fraction(2, span + 1)?;
110        let one = Rational::integer(1);
111        let complement = one.checked_sub(alpha)?;
112        value
113            .checked_mul(alpha)?
114            .checked_add(previous.checked_mul(complement)?)
115    }
116
117    fn limit_price(side: &str, reference: i64, bps: i64) -> Result<i64, StrategyError> {
118        let basis = Rational::fraction(10_000 + if side == "buy" { bps } else { -bps }, 10_000)?;
119        Rational::integer(reference).checked_mul(basis)?.round_i64()
120    }
121}
122
123impl Strategy for RationalEmaCross {
124    type Intent = RuntimeIntent;
125
126    fn on_event(&mut self, event: &StrategyEvent) -> Result<Option<Self::Intent>, StrategyError> {
127        if self
128            .last_sequence
129            .is_some_and(|last| event.observation_sequence <= last)
130        {
131            return Err(StrategyError::NonMonotonicObservation);
132        }
133        self.last_sequence = Some(event.observation_sequence);
134
135        if !self.universe.contains(&event.instrument_id) {
136            return Ok(None);
137        }
138        if !event.midpoint.is_positive() {
139            return Ok(None);
140        }
141
142        self.previous_fast = self.fast_ema;
143        self.previous_slow = self.slow_ema;
144        self.fast_ema = Some(Self::update(self.fast_ema, event.midpoint, self.fast)?);
145        self.slow_ema = Some(Self::update(self.slow_ema, event.midpoint, self.slow)?);
146        self.observations += 1;
147
148        if self.observations < self.min_observations || event.has_open_order {
149            return Ok(None);
150        }
151        let Some(previous_fast) = self.previous_fast else {
152            return Ok(None);
153        };
154        let Some(previous_slow) = self.previous_slow else {
155            return Ok(None);
156        };
157        let Some(fast) = self.fast_ema else {
158            return Ok(None);
159        };
160        let Some(slow) = self.slow_ema else {
161            return Ok(None);
162        };
163
164        let crossed_up = previous_fast.cmp_value(&previous_slow) != std::cmp::Ordering::Greater
165            && fast.cmp_value(&slow) == std::cmp::Ordering::Greater;
166        let crossed_down = previous_fast.cmp_value(&previous_slow) != std::cmp::Ordering::Less
167            && fast.cmp_value(&slow) == std::cmp::Ordering::Less;
168
169        let (side, quantity, reference) = if crossed_up && event.position_quantity == 0 {
170            ("buy", self.quantity, event.best_ask)
171        } else if crossed_down && event.position_quantity > 0 {
172            ("sell", event.position_quantity, event.best_bid)
173        } else {
174            return Ok(None);
175        };
176
177        self.intent_sequence += 1;
178        Ok(Some(RuntimeIntent {
179            intent_id: format!("{}:{:06}", self.candidate_id, self.intent_sequence),
180            candidate_id: self.candidate_id.clone(),
181            instrument_id: event.instrument_id.clone(),
182            side: side.to_owned(),
183            quantity,
184            limit_price_paise: Self::limit_price(side, reference, self.aggressive_limit_bps)?,
185        }))
186    }
187}
