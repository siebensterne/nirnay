2//! Build a verified kernel session from an offline decoded Upstox capture.
3//!
4//! One-way projection for real-source certification. The raw frame journal
5//! remains the evidence authority; decoded JSON is an input projection. No
6//! network, credential or order capability exists here.
7

36#[derive(Debug, Parser)]
37#[command(
38    name = "upstox-kernel-session",
39    about = "Project a decoded Upstox capture into a verified kernel session"
40)]
41struct Args {
42    #[arg(long)]
43    decoded: PathBuf,
44    #[arg(long)]
45    manifest: PathBuf,
46    #[arg(long)]
47    output: PathBuf,
48    /// Non-secret session identity recorded in the kernel envelope.
49    #[arg(long)]
50    session_id: String,
51    /// Optional JSON map of provider instrument key to canonical instrument id.
52    #[arg(long)]

54}

56fn paise(value: Option<&serde_json::Value>) -> anyhow::Result<i64> {
57    let number = value
58        .and_then(serde_json::Value::as_f64)
59        .context("missing level price")?;
60    rupees_to_paise(number).map_err(|error| anyhow::anyhow!(error.to_string()))
61}
62
63fn quantity(value: Option<&serde_json::Value>) -> anyhow::Result<i64> {
64    let text = value
65        .and_then(serde_json::Value::as_str)
66        .context("level quantity is not a string")?;
67    text.parse::<i64>().context("invalid level quantity")
68}

70fn build(input: &Args) -> anyhow::Result<KernelSession> {
71    let manifest: serde_json::Value =
72        serde_json::from_slice(&std::fs::read(&input.manifest)?).context("reading manifest")?;
73    if manifest.get("provider").and_then(serde_json::Value::as_str) != Some("Upstox") {
74        bail!("manifest is not an Upstox session");
75    }
76    let raw_digest = manifest
77        .get("raw_journal_sha256")
78        .and_then(serde_json::Value::as_str)
79        .context("manifest has no raw journal digest")?;
80    let instrument_map: std::collections::BTreeMap<String, String> = match &input.instrument_map {
81        Some(path) => serde_json::from_slice(&std::fs::read(path)?)?,
82        None => std::collections::BTreeMap::new(),
83    };
84
85    let text = std::fs::read_to_string(&input.decoded).context("reading decoded capture")?;
86    let mut observations = Vec::new();
87    for line in text.lines() {
88        if line.trim().is_empty() {
89            continue;
90        }
91        let row: serde_json::Value = serde_json::from_str(line).context("parsing decoded row")?;
92        let capture_sequence = row
93            .get("capture_sequence")
94            .and_then(serde_json::Value::as_u64)
95            .context("row has no capture sequence")?;
96        let frame_hash = ContentId::parse(
97            row.get("payload_sha256")
98                .and_then(serde_json::Value::as_str)
99                .context("row has no payload digest")?,
100        )
101        .map_err(|error| anyhow::anyhow!(error.to_string()))?;
102        let received_utc_ns = row
103            .get("received_at_utc_ns")
104            .and_then(serde_json::Value::as_i64)
105            .context("row has no receive time")?;
106        let received_monotonic_ns = row
107            .get("received_monotonic_ns")
108            .and_then(serde_json::Value::as_u64)
109            .context("row has no monotonic receive time")?;
110        let current_ms = row
111            .get("message")
112            .and_then(|value| value.get("currentTs"))
113            .and_then(serde_json::Value::as_str)
114            .context("row has no provider currentTs")?
115            .parse::<i64>()
116            .context("invalid provider currentTs")?;
117        let source_time_ns =
118            millis_to_ns(current_ms).map_err(|error| anyhow::anyhow!(error.to_string()))?;
119
120        let Some(feeds) = row
121            .get("message")
122            .and_then(|value| value.get("feeds"))
123            .and_then(serde_json::Value::as_object)
124        else {
125            continue;
126        };
127        for (provider_key, feed) in feeds {
128            let Some(quotes) = feed
129                .get("fullFeed")
130                .and_then(|value| value.get("marketFF"))
131                .and_then(|value| value.get("marketLevel"))
132                .and_then(|value| value.get("bidAskQuote"))
133                .and_then(serde_json::Value::as_array)
134            else {
135                continue;
136            };
137            let mut bids = Vec::new();
138            let mut asks = Vec::new();
139            for quote in quotes {
140                bids.push(MbpLevel {
141                    price_paise: paise(quote.get("bidP"))?,
142                    quantity: quantity(quote.get("bidQ"))?,
143                    order_count: None,
144                });
145                asks.push(MbpLevel {
146                    price_paise: paise(quote.get("askP"))?,
147                    quantity: quantity(quote.get("askQ"))?,
148                    order_count: None,
149                });
150            }
151            let book = MarketByPriceSnapshot {
152                instrument_id: instrument_map
153                    .get(provider_key)
154                    .cloned()
155                    .unwrap_or_else(|| provider_key.clone()),
156                venue: "NSE".to_owned(),
157                source_id: "upstox".to_owned(),
158                product: manifest
159                    .get("mode")
160                    .and_then(serde_json::Value::as_str)
161                    .unwrap_or("full")
162                    .to_owned(),
163                depth_levels: u32::try_from(bids.len().max(asks.len()))?,
164                bids,
165                asks,
166                source_time_ns: Some(source_time_ns),
167                last_trade_time_ns: None,
168                received_utc_ns,
169            };
170            book.validate()
171                .map_err(|error| anyhow::anyhow!(error.to_string()))?;
172            let canonical = format!("{provider_key}:{capture_sequence}").into_bytes();
173            observations.push(SourceObservation {
174                raw_frame_hash: frame_hash.clone(),
175                source_id: "upstox".to_owned(),
176                source_session_id: input.session_id.clone(),
177                provider_instrument_id: provider_key.clone(),
178                canonical_payload: canonical,
179                origin_class: OriginClass::RawWireFrame,
180                receive_utc_ns: received_utc_ns,
181                receive_monotonic_ns: received_monotonic_ns,
182                source_time_ns: Some(source_time_ns),
183                source_clock_domain: Some("provider_current_ts_ms".to_owned()),
184                channel_id: Some("upstox-v3-feed".to_owned()),
185                connection_epoch: 1,
186                receive_sequence: capture_sequence,
187                // Capture-local receive order, not an exchange sequence.
188                sequence: Some(capture_sequence),
189                sequence_scope: SequenceScope::Connection,
190                reference_revision: Some(raw_digest.to_owned()),
191                instrument_id: instrument_map.get(provider_key).cloned(),
192                evidence_token: None,
193                payload: Some(MarketDataEvent::MarketByPriceSnapshot(book)),
194            });
195        }
196    }
197    if observations.is_empty() {
198        bail!("decoded capture has no five-level full-feed observations");
199    }
200    Ok(KernelSession {
201        schema_version: market_kernel::KERNEL_SESSION_SCHEMA_VERSION,
202        source_scope: "NSE".to_owned(),
203        session_reference_revision: Some(raw_digest.to_owned()),
204        observations,
205    })
206}
207
208fn main() -> anyhow::Result<()> {
209    let args = Args::parse();
210    let session = build(&args)?;
211    let encoded = serde_json::to_vec(&session)?;
212    let mut file = std::fs::File::create(&args.output).context("creating kernel session")?;
213    file.write_all(&encoded)?;
214    file.write_all(b"\n")?;
215    file.sync_all()?;
216    eprintln!(
217        "observations={} source=upstox session={}",
218        session.observations.len(),
219        args.session_id
220    );
221    Ok(())
222}

224#[cfg(all(test, feature = "market-shaped-tests"))]
225mod tests {
226    use super::*;
227    use serde_json::json;
228
229    #[test]
230    fn provider_prices_convert_exactly_to_paise() {
231        let level = json!({"bidP": 76.44, "bidQ": "4103", "askP": 76.48, "askQ": "1495"});
232        assert_eq!(
233            paise(level.get("bidP")).expect("bid"),
234            7_644,
235            "rupees convert to paise"
236        );
237        assert_eq!(
238            paise(level.get("askP")).expect("ask"),
239            7_648,
240            "rupees convert to paise"
241        );
242        assert_eq!(
243            quantity(level.get("bidQ")).expect("qty"),
244            4_103,
245            "quantity parses as integer"
246        );
247        assert!(
248            paise(Some(&json!(f64::NAN))).is_err(),
249            "non-finite price is refused"
250        );
251    }
252}
