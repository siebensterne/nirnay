25/// Settlement classification for a BSE group in one source version.
26#[derive(Clone, Copy, Debug, PartialEq, Eq)]
27pub struct BseGroupClass {
28    pub settlement_regime: SettlementRegime,
29    pub z_category: bool,
30}
31
32/// Versioned group classification. Unknown source versions are rejected.
33pub fn bse_group_classes(
34    source_version: &str,
35) -> Result<Vec<(&'static str, BseGroupClass)>, String> {
36    match source_version {
37        "bse-security-master-v1" | "bse-security-master-v2" => Ok(known_group_classes()),
38        other => Err(format!("unsupported BSE source version {other}")),
39    }
40}
41
42/// The group-to-regime mapping shared by both declared source versions.
43/// Groups outside this table are never inferred from a row: the caller records
44/// them as rejections so an unknown settlement regime cannot gain eligibility.
45fn known_group_classes() -> Vec<(&'static str, BseGroupClass)> {
46    let trade = ["T", "MT", "TS", "IT", "Z", "R", "ZP", "P", "Y", "ZY", "XT"];
47    let standard = ["A", "B", "M", "MS", "S", "X", "XC", "XD"];
48    let mut rows = Vec::new();
49    for group in trade {
50        rows.push((
51            group,
52            BseGroupClass {
53                settlement_regime: SettlementRegime::TradeForTrade,
54                z_category: matches!(group, "Z" | "ZP" | "ZY"),
55            },
56        ));
57    }
58    for group in standard {
59        rows.push((
60            group,
61            BseGroupClass {
62                settlement_regime: SettlementRegime::Standard,
63                z_category: false,
64            },
65        ));
66    }
67    rows
68}

70/// One accepted BSE cash-market record.
71#[derive(Clone, Debug, PartialEq, Eq)]
72pub struct BseCashRow {
73    pub line: usize,
74    pub scrip_code: String,
75    pub symbol: String,
76    pub name: String,
77    pub isin: Isin,
78    pub settlement_group: String,
79    pub group_class: BseGroupClass,
80    pub tick_paise: i64,
81    pub market_lot: i64,
82    pub status: ListingStatus,
83}

85/// Column layout for one declared BSE source version.
86///
87/// The standardised file replaces the legacy names with ISO 20022 tags. The
88/// mapping is fixed by BSE's `SCRIP_STANDARDISED_FORMAT.docx` shipped with the
89/// dated source; no column is guessed from its position or sample value.
90struct BseLayout {
91    scrip: &'static str,
92    symbol: &'static str,
93    name: &'static str,
94    isin: &'static [&'static str],
95    tick: &'static [&'static str],
96    lot: &'static str,
97    type_flag: &'static [&'static str],
98    group: &'static str,
99    status: &'static str,
100    /// Only the standardised contract declares suspend/inactive as exclusions.
101    standardised: bool,
102}
103
104fn layout_for_version(source_version: &str) -> Result<BseLayout, String> {
105    match source_version {
106        "bse-security-master-v1" => Ok(BseLayout {
107            scrip: "SCRIP CODE",
108            symbol: "INSTRUMENT CODE",
109            name: "SCRIP NAME",
110            isin: &["ISIN CODE", "ISIN"],
111            tick: &["NEW TICK SIZE"],
112            lot: "MARKET LOT",
113            type_flag: &["SECURITY TYPE FLAG", "SECURITY TYPE", "SECURITY_TYPE_FLAG"],
114            group: "GROUP NAME",
115            status: "STATUS",
116            standardised: false,
117        }),
118        "bse-security-master-v2" => Ok(BseLayout {
119            scrip: "FININSTRMID",
120            symbol: "TCKRSYMB",
121            name: "FININSTRMNM",
122            isin: &["ISIN"],
123            // Field 58 (TickSz) is the authoritative tick size in BSE's
124            // standardised contract. Field 9 (BidIntrvl) is the semantic
125            // successor of v1's NEW TICK SIZE and is kept only as a fallback;
126            // in the dated fixture the two agree on every EQ row.
127            tick: &["TICKSZ", "BIDINTRVL"],
128            lot: "NEWBRDLOTQTY",
129            type_flag: &["SCTYTPFLG"],
130            group: "SCTYSRS",
131            status: "STS",
132            standardised: true,
133        }),
134        other => Err(format!("unsupported BSE source version {other}")),
135    }
136}

138/// Parse BSE security-master bytes under a declared source version.
139pub fn parse_bse_master(
140    bytes: &[u8],
141    source_version: &str,
142    scope: SnapshotScope,
143    limits: Limits,
144) -> Result<SourceBatch<BseCashRow>, String> {
145    let classes = bse_group_classes(source_version)?;
146    let layout = layout_for_version(source_version)?;
147    let document = parse_document(bytes, limits).map_err(|error| error.to_string())?;
148    let scrip_column = column(&document.header, &[layout.scrip])?;
149    let symbol_column = column(&document.header, &[layout.symbol])?;
150    let name_column = column(&document.header, &[layout.name])?;
151    let isin_column = column(&document.header, layout.isin)?;
152    let tick_column = column(&document.header, layout.tick)?;
153    let lot_column = column(&document.header, &[layout.lot])?;
154    let type_column = column(&document.header, layout.type_flag)?;
155    let group_column = column(&document.header, &[layout.group])?;
156    let status_column = column(&document.header, &[layout.status])?;
157
158    let mut batch = SourceBatch {
159        source_id: "bse.cash_security_master".to_owned(),
160        source_version: source_version.to_owned(),
161        scope,
162        accepted: Vec::new(),
163        excluded: Vec::new(),
164        rejected: Vec::new(),
165        physical_rows: document.rows.len(),
166    };
167    let mut scrips = std::collections::BTreeSet::new();
168    let mut isins = std::collections::BTreeSet::new();
169    for row in document.rows {
170        let line = row.line;
171        let scrip_code = row.fields[scrip_column].trim().to_owned();
172        let symbol = row.fields[symbol_column].trim().to_owned();
173        if scrip_code.is_empty() || symbol.is_empty() {
174            batch.rejected.push(Rejection {
175                line,
176                reason: "empty scrip code or security id".to_owned(),
177            });
178            continue;
179        }
180        if !row.fields[type_column].trim().eq_ignore_ascii_case("EQ") {
181            batch.excluded.push(Exclusion {
182                line,
183                rule: format!(
184                    "security type {} outside bse_cash scope",
185                    row.fields[type_column].trim()
186                ),
187            });
188            continue;
189        }
190        let group = row.fields[group_column].trim().to_uppercase();
191        let group_class = match classes.iter().find(|(name, _)| *name == group) {
192            Some((_, class)) => *class,
193            None => {
194                batch.rejected.push(Rejection {
195                    line,
196                    reason: format!("unclassified BSE group {group}; eligibility refused"),
197                });
198                continue;
199            }
200        };
201        let isin_text = row.fields[isin_column].trim();
202        let isin = match Isin::new(isin_text) {
203            Ok(value) => value,
204            Err(error) => {
205                batch.rejected.push(Rejection {
206                    line,
207                    reason: format!("invalid ISIN {isin_text}: {error}"),
208                });
209                continue;
210            }
211        };
212        let name = row.fields[name_column].trim().to_owned();
213        if name.is_empty() {
214            batch.rejected.push(Rejection {
215                line,
216                reason: "empty security name; no fallback identity is inferred".to_owned(),
217            });
218            continue;
219        }
220        let tick_paise = match market_types::fixed::parse_scaled(row.fields[tick_column].trim(), 2)
221        {
222            Ok(value) if value > 0 => value,
223            _ => {
224                batch.rejected.push(Rejection {
225                    line,
226                    reason: "tick size is not a positive paise value".to_owned(),
227                });
228                continue;
229            }
230        };
231        let market_lot = match row.fields[lot_column].trim().parse::<i64>() {
232            Ok(value) if value > 0 => value,
233            _ => {
234                batch.rejected.push(Rejection {
235                    line,
236                    reason: "market lot is not a positive integer".to_owned(),
237                });
238                continue;
239            }
240        };
241        let status_text = row.fields[status_column].trim().to_uppercase();
242        let status = match status_text.as_str() {
243            "A" | "ACTIVE" => ListingStatus::Active,
244            // BSE's standardised contract declares S (suspend) and I (inactive)  # prose-audit-ignore: BSE ticker symbol identifier
245            // as real, non-tradable states. They are excluded by that declared
246            // rule, not silently treated as active and never inferred as errors.
247            "S" | "SUSPEND" | "I" | "INACTIVE" if layout.standardised => {
248                batch.excluded.push(Exclusion {
249                    line,
250                    rule: format!("instrument status {status_text} is not active"),
251                });
252                continue;
253            }
254            other => {
255                batch.rejected.push(Rejection {
256                    line,
257                    reason: format!("unsupported instrument status {other}"),
258                });
259                continue;
260            }
261        };
262        let isin_key = isin.as_str().to_owned();
263        if scrips.contains(&scrip_code) || isins.contains(&isin_key) {
264            batch.rejected.push(Rejection {
265                line,
266                reason: format!("duplicate scrip code or ISIN for {scrip_code} / {isin_key}"),
267            });
268            continue;
269        }
270        scrips.insert(scrip_code.clone());
271        isins.insert(isin_key);
272        batch.accepted.push(BseCashRow {
273            line,
274            scrip_code,
275            symbol,
276            name,
277            isin,
278            settlement_group: group,
279            group_class,
280            tick_paise,
281            market_lot,
282            status,
283        });
284    }
285    Ok(batch)
