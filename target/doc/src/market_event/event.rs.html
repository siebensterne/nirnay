17/// Contract schema version. Bump only with a reviewed decision.
18pub const KERNEL_EVENT_SCHEMA_VERSION: u32 = 1;

20/// Evidence that a recovery boundary was crossed with an authoritative event.
21#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
22pub struct RecoveryEvidence {
23    /// Identity of the authoritative snapshot or reset that cleared the gap.
24    pub authority_id: ContentId,
25    /// The connection epoch at which the authority was accepted.
26    pub connection_epoch: u64,
27}

29/// One kernel market event. Every field required to reconstruct origin and
30/// interpretation is explicit; payload content stays in the research domain
31/// model, not in this envelope.
32#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
33pub struct KernelEvent {
34    pub schema_version: u32,
35    /// Identity of the original captured frame (all instruments in it).
36    pub capture_frame_id: ContentId,
37    /// Stable identity of one source delivery.
38    pub observation_id: ContentId,
39    /// Source-backed business identity, present only when the source supplies
40    /// a distinct business event. `None` for a bare state observation.
41    pub business_event_id: Option<ContentId>,
42    /// Back-reference for a derived event.
43    pub source_observation_id: Option<ContentId>,
44    pub origin_class: OriginClass,
45    pub source_id: String,
46    pub source_session_id: String,
47    pub connection_epoch: u64,
48    pub receive_sequence: u64,
49    pub sequence: Option<u64>,
50    pub sequence_scope: SequenceScope,
51    pub provider_instrument_id: String,
52    /// Nirnay opaque instrument identity. `None` until resolved.
53    pub instrument_id: Option<String>,
54    pub venue_id: Option<String>,
55    /// Source-provided timestamp in its own clock domain.
56    pub source_time_ns: Option<i64>,
57    pub source_clock_domain: Option<String>,
58    /// Local UTC receive time, ns since epoch.
59    pub receive_utc_ns: i64,
60    /// Local monotonic receive time.
61    pub receive_monotonic_ns: u64,
62    pub raw_frame_hash: ContentId,
63    /// Hash of the exact reference revision used for resolution.
64    pub reference_revision: Option<String>,
65    pub quality: super::continuity::StreamState,
66    pub evidence_token: Option<String>,
67    /// Typed market payload. The envelope keeps identity and quality; the
68    /// payload keeps financial meaning. None for bare state observations.
69    pub payload: Option<crate::mbp::MarketDataEvent>,
70}

72impl KernelEvent {
73    /// Build an observation-only event. No business identity is invented.
74    #[allow(clippy::too_many_arguments)] // full identity and time surface
75    pub fn observation_only(
76        capture_frame_id: ContentId,
77        observation_id: ContentId,
78        raw_frame_hash: ContentId,
79        origin_class: OriginClass,
80        source_id: impl Into<String>,
81        source_session_id: impl Into<String>,
82        provider_instrument_id: impl Into<String>,
83        receive_utc_ns: i64,
84        receive_monotonic_ns: u64,
85    ) -> Result<Self, ContractError> {
86        if origin_class.can_be_exchange_event() {
87            return Err(ContractError::ExchangeEventWithoutEvidence);
88        }
89        if receive_utc_ns <= 0 {
90            return Err(ContractError::TimeDomainCollision);
91        }
92        Ok(Self {
93            schema_version: KERNEL_EVENT_SCHEMA_VERSION,
94            capture_frame_id,
95            observation_id,
96            business_event_id: None,
97            source_observation_id: None,
98            origin_class,
99            source_id: source_id.into(),
100            source_session_id: source_session_id.into(),
101            connection_epoch: 0,
102            receive_sequence: 0,
103            sequence: None,
104            sequence_scope: SequenceScope::None,
105            provider_instrument_id: provider_instrument_id.into(),
106            instrument_id: None,
107            venue_id: None,
108            source_time_ns: None,
109            source_clock_domain: None,
110            receive_utc_ns,
111            receive_monotonic_ns,
112            raw_frame_hash,
113            reference_revision: None,
114            quality: super::continuity::StreamState::Clean,
115            evidence_token: None,
116            payload: None,
117        })
118    }
119
120    /// Attach a receive fence. Epoch zero means no fence; sequence must
121    /// advance with the epoch so no half-fence is accepted.
122    pub fn with_receive_fence(
123        mut self,
124        connection_epoch: u64,
125        receive_sequence: u64,
126    ) -> Result<Self, ContractError> {
127        if (connection_epoch == 0) != (receive_sequence == 0) {
128            return Err(ContractError::InvalidReceiveFence);
129        }
130        self.connection_epoch = connection_epoch;
131        self.receive_sequence = receive_sequence;
132        Ok(self)
133    }
134
135    /// Attach a source sequence with its declared scope.
136    pub fn with_sequence(mut self, sequence: u64, scope: SequenceScope) -> Self {
137        self.sequence = Some(sequence);
138        self.sequence_scope = scope;
139        self
140    }
141
142    /// Attach source time with an explicit clock domain. The clock domain must
143    /// not claim receive semantics.
144    pub fn with_source_time(
145        mut self,
146        source_time_ns: i64,
147        source_clock_domain: impl Into<String>,
148    ) -> Result<Self, ContractError> {
149        let domain = source_clock_domain.into();
150        if domain == "receive" || domain == "receive_monotonic" {
151            return Err(ContractError::TimeDomainCollision);
152        }
153        self.source_time_ns = Some(source_time_ns);
154        self.source_clock_domain = Some(domain);
155        Ok(self)
156    }
157
158    /// Resolve the instrument against an exact reference revision.
159    pub fn resolve_instrument(
160        mut self,
161        instrument_id: impl Into<String>,
162        venue_id: Option<String>,
163        reference_revision: impl Into<String>,
164    ) -> Result<Self, ContractError> {
165        let revision = reference_revision.into();
166        if revision.is_empty() {
167            return Err(ContractError::MissingReferenceRevision);
168        }
169        self.instrument_id = Some(instrument_id.into());
170        self.venue_id = venue_id;
171        self.reference_revision = Some(revision);
172        Ok(self)
173    }
174
175    /// Derive a business event from this observation. The discriminator is
176    /// caller supplied and must name a domain event kind.
177    pub fn derived_business_event(
178        &self,
179        event_kind: &str,
180        discriminator: &str,
181    ) -> Result<Self, ContractError> {
182        if discriminator.is_empty() {
183            return Err(ContractError::MissingDomainDiscriminator);
184        }
185        let business =
186            crate::identity::business_event_id(&self.observation_id, event_kind, discriminator)?;
187        let mut derived = self.clone();
188        derived.business_event_id = Some(business);
189        derived.source_observation_id = Some(self.observation_id.clone());
190        derived.origin_class = OriginClass::Reconstructed;
191        Ok(derived)
192    }
193
194    /// Validate structural invariants. Public fields permit construction; this
195    /// is the authoritative check and is invoked on deserialization.
196    pub fn validate_invariants(&self) -> Result<(), ContractError> {
197        if self.schema_version != KERNEL_EVENT_SCHEMA_VERSION {
198            return Err(ContractError::UnsupportedSchemaVersion(self.schema_version));
199        }
200        if self.business_event_id.is_some() && self.source_observation_id.is_none() {
201            return Err(ContractError::MissingSourceObservation);
202        }
203        if self.origin_class.can_be_exchange_event() {
204            let token = self
205                .evidence_token
206                .as_deref()
207                .ok_or(ContractError::ExchangeEventWithoutEvidence)?;
208            ContentId::parse(token).map_err(|_| ContractError::ExchangeEventWithoutEvidence)?;
209        }
210        if (self.connection_epoch == 0) != (self.receive_sequence == 0) {
211            return Err(ContractError::InvalidReceiveFence);
212        }
213        if self.quality == super::continuity::StreamState::Recovered {
214            let token = self
215                .evidence_token
216                .as_deref()
217                .ok_or(ContractError::RecoveryWithoutEvidence)?;
218            ContentId::parse(token).map_err(|_| ContractError::RecoveryWithoutEvidence)?;
219        }
220        Ok(())
221    }
222
223    /// Whether this event may be used for a risk-increasing decision.
224    #[must_use]
225    pub fn is_actionable(&self) -> bool {
226        self.validate_invariants().is_ok()
227            && self.instrument_id.is_some()
228            && self.reference_revision.is_some()
229            && self.quality.is_clean()
230    }
231}

233/// Deserialization validates invariants so a forged wire object cannot pass as
234/// a kernel event.
235impl<'de> Deserialize<'de> for KernelEvent {
236    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
237    where
238        D: serde::Deserializer<'de>,
239    {
240        #[derive(Deserialize)]
241        struct Wire {
242            schema_version: u32,
243            capture_frame_id: ContentId,
244            observation_id: ContentId,
245            business_event_id: Option<ContentId>,
246            source_observation_id: Option<ContentId>,
247            origin_class: OriginClass,
248            source_id: String,
249            source_session_id: String,
250            connection_epoch: u64,
251            receive_sequence: u64,
252            sequence: Option<u64>,
253            sequence_scope: SequenceScope,
254            provider_instrument_id: String,
255            instrument_id: Option<String>,
256            venue_id: Option<String>,
257            source_time_ns: Option<i64>,
258            source_clock_domain: Option<String>,
259            receive_utc_ns: i64,
260            receive_monotonic_ns: u64,
261            raw_frame_hash: ContentId,
262            reference_revision: Option<String>,
263            quality: super::continuity::StreamState,
264            evidence_token: Option<String>,
265            payload: Option<crate::mbp::MarketDataEvent>,
266        }
267        let wire = Wire::deserialize(deserializer)?;
268        let event = Self {
269            schema_version: wire.schema_version,
270            capture_frame_id: wire.capture_frame_id,
271            observation_id: wire.observation_id,
272            business_event_id: wire.business_event_id,
273            source_observation_id: wire.source_observation_id,
274            origin_class: wire.origin_class,
275            source_id: wire.source_id,
276            source_session_id: wire.source_session_id,
277            connection_epoch: wire.connection_epoch,
278            receive_sequence: wire.receive_sequence,
279            sequence: wire.sequence,
280            sequence_scope: wire.sequence_scope,
281            provider_instrument_id: wire.provider_instrument_id,
282            instrument_id: wire.instrument_id,
283            venue_id: wire.venue_id,
284            source_time_ns: wire.source_time_ns,
285            source_clock_domain: wire.source_clock_domain,
286            receive_utc_ns: wire.receive_utc_ns,
287            receive_monotonic_ns: wire.receive_monotonic_ns,
288            raw_frame_hash: wire.raw_frame_hash,
289            reference_revision: wire.reference_revision,
290            quality: wire.quality,
291            evidence_token: wire.evidence_token,
292            payload: wire.payload,
293        };
294        event
295            .validate_invariants()
296            .map_err(serde::de::Error::custom)?;
297        Ok(event)
298    }
