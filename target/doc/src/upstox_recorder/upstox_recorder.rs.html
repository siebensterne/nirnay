45/// A payload the archive cannot accept. This is a terminal capture outcome, not
46/// a network failure: retrying the connection cannot make it valid.
47#[derive(Clone, Copy, Debug, PartialEq, Eq)]
48struct OversizedFrame {
49    size: usize,
50    limit: usize,
51}
52
53impl std::fmt::Display for OversizedFrame {
54    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
55        write!(
56            formatter,
57            "Upstox frame of {} bytes exceeds the archive payload limit of {} bytes",
58            self.size, self.limit
59        )
60    }
61}

63impl std::error::Error for OversizedFrame {}

65/// The shared ingress port refused a frame. Retrying the connection cannot
66/// make a refused observation acceptable, so this is terminal.
67#[derive(Clone, Copy, Debug, PartialEq, Eq)]
68struct IngressRefusal;
69
70impl std::fmt::Display for IngressRefusal {
71    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
72        formatter.write_str("shared ingress refused an Upstox frame")
73    }
74}
75
76impl std::error::Error for IngressRefusal {}

78#[derive(Clone, Copy, Debug, PartialEq, Eq)]
79enum ConnectionFailure {
80    OversizedFrame,
81    IngressRefusal,
82    Retryable,
83}
84
85fn classify_failure(error: &anyhow::Error) -> ConnectionFailure {
86    if error.downcast_ref::<OversizedFrame>().is_some() {
87        ConnectionFailure::OversizedFrame
88    } else if error.downcast_ref::<IngressRefusal>().is_some() {
89        ConnectionFailure::IngressRefusal
90    } else {
91        ConnectionFailure::Retryable
92    }
93}

95fn validate_frame_size(size: usize) -> Result<(), OversizedFrame> {
96    if size > MAX_PAYLOAD_BYTES {
97        return Err(OversizedFrame {
98            size,
99            limit: MAX_PAYLOAD_BYTES,
100        });
101    }
102    Ok(())
103}

226struct Session {
227    run_dir: PathBuf,
228    journal_path: PathBuf,
229    ingestor: DurableIngestor,
230    observations: BufWriter<File>,
231    lifecycle: BufWriter<File>,
232    started_utc: String,
233    started_instant: Instant,
234    instrument_keys: Vec<String>,
235    mode: FeedMode,
236    entitlement_ref: String,
237    connection_count: u64,
238    frame_count: u64,
239    decode_error_count: u64,
240    terminal_error: Option<String>,
241}
242
243impl Session {
244    fn create(args: &Args, instrument_keys: Vec<String>) -> anyhow::Result<Self> {
245        let started = Timestamp::now();
246        let stamp = started.strftime("%Y%m%dT%H%M%S%.fZ");
247        let run_dir = args.output_root.join(format!("upstox-{stamp}"));
248        fs::create_dir_all(&args.output_root)
249            .with_context(|| format!("creating output root {}", args.output_root.display()))?;
250        set_private_permissions(&args.output_root)?;
251        fs::create_dir(&run_dir)
252            .with_context(|| format!("creating unique session directory {}", run_dir.display()))?;
253        set_private_permissions(&run_dir)?;
254        let journal_path = run_dir.join("raw-frames.jsonl");
255        // One authoritative journal, owned by the shared ingress port. The
256        // recorder-local sequence is a capture-session sequence, never a
257        // provider sequence.
258        let ingestor = DurableIngestor::create_new(
259            &journal_path,
260            UPSTOX_SOURCE_ID,
261            SequencePolicy::CaptureSession,
262        )
263        .context("creating shared-ingress raw-frame journal")?;
264        let observations = BufWriter::new(create_private_file(
265            &run_dir.join("decoded-observations.jsonl"),
266        )?);
267        let lifecycle = BufWriter::new(create_private_file(
268            &run_dir.join("connection-events.jsonl"),
269        )?);
270        Ok(Self {
271            run_dir,
272            journal_path,
273            ingestor,
274            observations,
275            lifecycle,
276            started_utc: started.strftime("%Y-%m-%dT%H:%M:%S%.9fZ").to_string(),
277            started_instant: Instant::now(),
278            instrument_keys,
279            mode: args.mode,
280            entitlement_ref: args.entitlement_ref.clone(),
281            connection_count: 0,
282            frame_count: 0,
283            decode_error_count: 0,
284            terminal_error: None,
285        })
286    }
287
288    fn lifecycle(&mut self, event: &str, attempt: u64, detail: Option<&str>) -> anyhow::Result<()> {
289        let row = json!({
290            "schema_version": 1,
291            "event": event,
292            "attempt": attempt,
293            "at_utc": Timestamp::now()
294                .strftime("%Y-%m-%dT%H:%M:%S%.9fZ")
295                .to_string(),
296            "detail": detail,
297        });
298        serde_json::to_writer(&mut self.lifecycle, &row)?;
299        self.lifecycle.write_all(b"\n")?;
300        self.lifecycle.flush()?;
301        Ok(())
302    }
303
304    fn frame(
305        &mut self,
306        payload: &[u8],
307        received_utc_ns: i64,
308        received_monotonic_ns: u64,
309    ) -> anyhow::Result<()> {
310        validate_frame_size(payload.len())?;
311        let capture_sequence = self
312            .frame_count
313            .checked_add(1)
314            .context("Upstox capture sequence exhausted")?;
315        let source_frame = SourceFrame {
316            source_id: UPSTOX_SOURCE_ID.to_owned(),
317            provider_instrument_id: "__multi_instrument_frame__".to_owned(),
318            entitlement_ref: self.entitlement_ref.clone(),
319            received_utc_nanos: received_utc_ns,
320            received_monotonic_nanos: received_monotonic_ns,
321            exchange_utc_nanos: None,
322            sequence: Some(Sequence::capture_session(capture_sequence)),
323            payload: payload.to_vec(),
324        };
325        match self
326            .ingestor
327            .ingest_one(source_frame)
328            .context("durably appending Upstox raw frame through shared ingress")?
329        {
330            IngestOutcome::Committed { .. } => {}
331            IngestOutcome::DuplicateSkipped
332            | IngestOutcome::ConflictRejected
333            | IngestOutcome::MalformedRejected => {
334                return Err(IngressRefusal.into());
335            }
336        }
337        self.frame_count = self.ingestor.checkpoint().committed_frames;
338
339        match decode_feed_response_diagnostic(payload) {
340            Ok(diagnostic) => {
341                let feeds = diagnostic
342                    .feeds
343                    .iter()
344                    .map(|feed| FeedObservation {
345                        key: &feed.key,
346                        kind: match feed.kind {
347                            upstox_feed_proto::decode::FeedKind::Ltpc => "ltpc",
348                            upstox_feed_proto::decode::FeedKind::FullFeed => "full_feed",
349                            upstox_feed_proto::decode::FeedKind::FirstLevelWithGreeks => {
350                                "first_level_with_greeks"
351                            }
352                            upstox_feed_proto::decode::FeedKind::Unset => "unset",
353                        },
354                        request_mode_raw: feed.request_mode_raw,
355                    })
356                    .collect();
357                let row = Observation {
358                    schema_version: 1,
359                    capture_sequence: self.frame_count,
360                    received_at_utc_ns: received_utc_ns,
361                    provider_current_ts_ms: diagnostic.current_ts,
362                    response_type_raw: diagnostic.response_type_raw,
363                    feed_keys: &diagnostic.feed_keys,
364                    feeds,
365                };
366                serde_json::to_writer(&mut self.observations, &row)?;
367                self.observations.write_all(b"\n")?;
368            }
369            Err(error) => {
370                self.decode_error_count += 1;
371                let row = json!({
372                    "schema_version": 1,
373                    "capture_sequence": self.frame_count,
374                    "received_at_utc_ns": received_utc_ns,
375                    "decode_error": error.to_string(),
376                });
377                serde_json::to_writer(&mut self.observations, &row)?;
378                self.observations.write_all(b"\n")?;
379            }
380        }
381        Ok(())
382    }
383
384    fn finish(mut self) -> anyhow::Result<PathBuf> {
385        self.observations.flush()?;
386        self.observations.get_ref().sync_all()?;
387        self.lifecycle.flush()?;
388        self.lifecycle.get_ref().sync_all()?;
389        drop(self.ingestor);
390        let summary = verify_journal(&self.journal_path).context("verifying Upstox raw journal")?;
391        let mut source = File::open(&self.journal_path)?;
392        let mut hasher = Sha256::new();
393        let mut buffer = [0_u8; 1024 * 1024];
394        loop {
395            let count = source.read(&mut buffer)?;
396            if count == 0 {
397                break;
398            }
399            hasher.update(&buffer[..count]);
400        }
401        let digest = hasher.finalize();
402        let raw_journal_sha256 = digest
403            .iter()
404            .map(|byte| format!("{byte:02x}"))
405            .collect::<String>();
406        let manifest = Manifest {
407            schema_version: 1,
408            provider: "Upstox",
409            feed_protocol: "Market Data Feed V3; protobuf binary",
410            mode: self.mode.as_str().to_owned(),
411            instrument_keys: self.instrument_keys,
412            entitlement_ref: self.entitlement_ref,
413            started_at_utc: self.started_utc,
414            ended_at_utc: Timestamp::now()
415                .strftime("%Y-%m-%dT%H:%M:%S%.9fZ")
416                .to_string(),
417            duration_ns: self.started_instant.elapsed().as_nanos(),
418            connection_count: self.connection_count,
419            frame_count: self.frame_count,
420            decode_error_count: self.decode_error_count,
421            raw_journal: "raw-frames.jsonl".to_owned(),
422            raw_journal_sha256,
423            raw_journal_frame_count: summary.frame_count,
424            provider_sequence_available: false,
425            capture_sequence_semantics: "local receive-order counter; not a provider sequence",
426            terminal_error: self.terminal_error,
427            limitations: vec![
428                "A successful connection does not verify data redistribution rights.",
429                "The provider envelope currentTs is retained in decoded diagnostics; per-instrument exchange timestamps are not inferred.",
430                "Unknown protobuf fields remain in raw frames but may not appear in the diagnostic projection.",
431                "The first V4 recorder does not normalize instrument identities, prices or quantities into trading-core events.",
432            ],
433        };
434        let manifest_path = self.run_dir.join("manifest.json");
435        let temp_path = self.run_dir.join("manifest.json.tmp");
436        {
437            let file = create_private_file(&temp_path)?;
438            let mut writer = BufWriter::new(file);
439            serde_json::to_writer_pretty(&mut writer, &manifest)?;
440            writer.write_all(b"\n")?;
441            writer.flush()?;
442            writer.get_ref().sync_all()?;
443        }
444        fs::rename(&temp_path, &manifest_path)?;
445        let dir = File::open(&self.run_dir)?;
446        dir.sync_all()?;
447        Ok(self.run_dir)
448    }
449}

459/// The only production path to a session. Authorization happens before any
460/// directory, journal or manifest file is created; a denial leaves no files.
461fn create_authorized_session(args: &Args, instrument_keys: Vec<String>) -> anyhow::Result<Session> {
462    let registry = SourceProductRegistry::load(&args.registry)
463        .with_context(|| format!("loading rights registry {}", args.registry.display()))?;
464    let key = registry_key(args.mode);
465    registry
466        .authorize(&key, DataOperation::Capture)
467        .context("capture is not authorized by the registry")?;
468    registry
469        .authorize(&key, DataOperation::Store)
470        .context("store is not authorized by the registry")?;
471    Session::create(args, instrument_keys)
472}

474fn read_watchlist(path: &Path, mode: FeedMode) -> anyhow::Result<Vec<String>> {
475    let text = fs::read_to_string(path)
476        .with_context(|| format!("reading watchlist {}", path.display()))?;
477    let mut keys = Vec::new();
478    let mut seen = BTreeSet::new();
479    for (line_number, line) in text.lines().enumerate() {
480        let value = line.trim();
481        if value.is_empty() || value.starts_with('#') {
482            continue;
483        }
484        ensure!(
485            value.contains('|'),
486            "watchlist line {} is not an Upstox instrument_key",
487            line_number + 1
488        );
489        ensure!(
490            seen.insert(value.to_owned()),
491            "duplicate Upstox instrument key at line {}",
492            line_number + 1
493        );
494        keys.push(value.to_owned());
495    }
496    ensure!(!keys.is_empty(), "Upstox watchlist is empty");
497    ensure!(
498        keys.len() <= mode.max_symbols(),
499        "{} mode allows at most {} symbols",
500        mode.as_str(),
501        mode.max_symbols()
502    );
503    Ok(keys)
504}

506async fn authorize(client: &Client, token: &Token) -> anyhow::Result<String> {
507    #[derive(Deserialize)]
508    struct Envelope {
509        data: Option<Data>,
510        message: Option<String>,
511    }
512    #[derive(Deserialize)]
513    struct Data {
514        authorized_redirect_uri: String,
515    }
516    let response = client
517        .get(AUTHORIZE_URL)
518        .bearer_auth(&token.0)
519        .header("Accept", "application/json")
520        .send()
521        .await
522        .context("requesting Upstox authorized WebSocket URL")?;
523    let status = response.status();
524    ensure!(
525        status.is_success(),
526        "Upstox authorize endpoint returned HTTP {status}"
527    );
528    let body: Envelope = response
529        .json()
530        .await
531        .context("decoding Upstox authorize response")?;
532    let Some(data) = body.data else {
533        bail!(
534            "Upstox authorize response omitted data: {}",
535            body.message
536                .unwrap_or_else(|| "unknown response".to_owned())
537        );
538    };
539    let parsed = reqwest::Url::parse(&data.authorized_redirect_uri)
540        .context("parsing Upstox WebSocket URL")?;
541    ensure!(
542        parsed.scheme() == "wss",
543        "Upstox authorized URL must use wss"
544    );
545    let host = parsed.host_str().unwrap_or_default();
546    ensure!(
547        host == "upstox.com" || host.ends_with(".upstox.com"),
548        "Upstox authorize returned an unexpected host"
549    );
550    Ok(data.authorized_redirect_uri)
551}

553async fn run_connection(
554    client: &Client,
555    token: &Token,
556    session: &mut Session,
557    stop: &mut std::pin::Pin<Box<dyn std::future::Future<Output = std::io::Result<()>> + Send>>,
558    attempt: u64,
559) -> anyhow::Result<bool> {
560    let authorized_url = authorize(client, token).await?;
561    let (mut socket, _response) = connect_async(authorized_url.as_str())
562        .await
563        .map_err(|_| anyhow::anyhow!("connecting to Upstox Market Data Feed V3 failed"))?;
564    let subscription = json!({
565        "guid": format!("nirnay-{}-{attempt}", Timestamp::now().as_millisecond()),
566        "method": "sub",
567        "data": {
568            "mode": session.mode.as_str(),
569            "instrumentKeys": &session.instrument_keys,
570        }
571    });
572    let bytes = serde_json::to_vec(&subscription)?;
573    socket
574        .send(Message::Binary(bytes.into()))
575        .await
576        .context("sending Upstox binary subscription")?;
577    session.connection_count += 1;
578    session.lifecycle(
579        "connected_and_subscribed",
580        attempt,
581        Some("subscription sent as binary JSON control frame"),
582    )?;
583    loop {
584        tokio::select! {
585            result = stop.as_mut() => {
586                result.context("waiting for SIGINT")?;
587                session.lifecycle("operator_stop", attempt, None)?;
588                let _ = socket.close(None).await;
589                return Ok(true);
590            }
591            message = socket.next() => {
592                match message {
593                    Some(Ok(Message::Binary(payload))) => {
594                        let utc_ns = i64::try_from(Timestamp::now().as_nanosecond())
595                            .ok()
596                            .context("UTC timestamp outside nanosecond range")?;
597                        let mono_ns = u64::try_from(session.started_instant.elapsed().as_nanos()).context("monotonic duration overflow")?;
598                        session.frame(&payload, utc_ns, mono_ns)?;
599                    }
600                    Some(Ok(Message::Ping(payload))) => socket.send(Message::Pong(payload)).await.context("answering Upstox ping")?,
601                    Some(Ok(Message::Close(frame))) => {
602                        session.lifecycle("server_closed", attempt, Some(&format!("{frame:?}")))?;
603                        return Ok(false);
604                    }
605                    Some(Ok(Message::Text(_))) => {
606                        session.lifecycle("unexpected_text_frame", attempt, Some("text frame preserved only as a lifecycle observation"))?;
607                    }
608                    Some(Ok(Message::Pong(_))) => (),
609                    Some(Ok(Message::Frame(_))) => (),
610                    Some(Err(error)) => {
611                        session.lifecycle("socket_error", attempt, Some("websocket read failed; diagnostic details omitted"))?;
612                        return Err(error).context("Upstox WebSocket read failed");
613                    }
614                    None => {
615                        session.lifecycle("stream_ended", attempt, None)?;
616                        return Ok(false);
617                    }
618                }
619            }
620        }
621    }
622}

624#[tokio::main]
625async fn main() -> anyhow::Result<()> {
626    let args = Args::parse();
627    let token = Token::from_environment()?;
628    ensure!(
629        !args.entitlement_ref.trim().is_empty(),
630        "NIRNAY_UPSTOX_ENTITLEMENT_REF is required"
631    );
632    let instrument_keys = read_watchlist(&args.instrument_file, args.mode)?;
633    let client = Client::builder()
634        .timeout(Duration::from_secs(20))
635        .redirect(redirect::Policy::none())
636        .user_agent(concat!(
637            "nirnay-upstox-recorder/",
638            env!("CARGO_PKG_VERSION")
639        ))
640        .build()?;
641    let mut session = create_authorized_session(&args, instrument_keys)?;
642    let mut stop: std::pin::Pin<Box<dyn std::future::Future<Output = std::io::Result<()>> + Send>> =
643        Box::pin(signal::ctrl_c());
644
645    for attempt in 0..=args.max_reconnects {
646        if attempt > 0 {
647            let backoff = 1_u64.checked_shl(attempt.min(6)).unwrap_or(64);
648            session.lifecycle(
649                "reconnect_wait",
650                u64::from(attempt),
651                Some(&format!("backoff_seconds={backoff}")),
652            )?;
653            tokio::select! {
654                result = stop.as_mut() => { result.context("waiting for SIGINT")?; break; }
655                _ = sleep(Duration::from_secs(backoff)) => (),
656            }
657        }
658        match run_connection(&client, &token, &mut session, &mut stop, u64::from(attempt)).await {
659            Ok(true) => break,
660            Ok(false) if attempt < args.max_reconnects => (),
661            Ok(false) => {
662                session.terminal_error =
663                    Some("reconnect budget exhausted after server closed the stream".to_owned());
664                session.lifecycle("reconnect_budget_exhausted", u64::from(attempt), None)?;
665            }
666            Err(error) => {
667                if matches!(
668                    classify_failure(&error),
669                    ConnectionFailure::OversizedFrame | ConnectionFailure::IngressRefusal
670                ) {
671                    session.terminal_error = Some(format!("terminal capture failure: {error}"));
672                    session.lifecycle(
673                        "terminal_capture_failure",
674                        u64::from(attempt),
675                        Some(&error.to_string()),
676                    )?;
677                    break;
678                }
679                if attempt < args.max_reconnects {
680                    session.lifecycle(
681                        "reconnect_after_error",
682                        u64::from(attempt),
683                        Some("network_or_protocol_error; diagnostic details omitted"),
684                    )?;
685                } else {
686                    session.terminal_error = Some(
687                        "reconnect budget exhausted after network or protocol error".to_owned(),
688                    );
689                    session.lifecycle(
690                        "reconnect_budget_exhausted_after_error",
691                        u64::from(attempt),
692                        Some("network_or_protocol_error; diagnostic details omitted"),
693                    )?;
694                }
695            }
696        }
697    }
698    let terminal_error = session.terminal_error.clone();
699    let path = session.finish()?;
700    writeln!(io::stdout().lock(), "{}", path.display())?;
701    if terminal_error.is_some() {
702        bail!(
703            "Upstox capture ended with a terminal error; inspect the session events at {}",
704            path.display()
705        );
706    }
707    Ok(())
708}
