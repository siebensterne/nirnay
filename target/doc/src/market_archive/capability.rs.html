1//! Machine-readable source-product capability contract.
2//!
3//! A product declares what it supplies. Missing capabilities are
4//! rejections, never inferences. Rights and entitlements are recorded as
5//! verification state only; this module never grants or upgrades them.
6

15const SCHEMA_VERSION: u32 = 1;

17#[derive(Debug, Clone, PartialEq, Eq, Error)]
18pub enum CapabilityError {
19    #[error("registry has schema version {0}, expected 1")]
20    UnsupportedSchema(u32),
21    #[error("registry could not be read")]
22    Unreadable,
23    #[error("registry is invalid: {0}")]
24    InvalidRegistry(String),
25    #[error("no declared product for provider `{0}`, product `{1}`, venue `{2}`, segment `{3}`")]
26    UnknownProduct(String, String, String, String),
27    #[error("product supplies {declared} levels; requirement needs {required}")]
28    InsufficientDepth { declared: u32, required: u32 },
29    #[error("product depth model is {declared:?}; requirement needs {required:?}")]
30    DepthModelMismatch {
31        declared: DepthModel,
32        required: DepthModel,
33    },
34    #[error("product does not supply trade prints")]
35    NoTradePrints,
36    #[error("product does not supply a clean snapshot")]
37    NoSnapshot,
38    #[error("product declares no source sequence continuity")]
39    NoSourceContinuity,
40    #[error("product does not grant the verified right to {0}")]
41    MissingRight(String),
42    #[error("product entitlement `{0}` has not been verified")]
43    EntitlementUnverified(String),
44}

146/// A data-use operation. The vocabulary matches the seven written-evidence
147/// classes in the registry rights review.
148#[derive(Copy, Clone, Debug, PartialEq, Eq)]
149pub enum DataOperation {
150    Capture,
151    Store,
152    Replay,
153    Backup,
154    InternalAnalytics,
155    ExternalRedistribution,
156    Retention,
157}
158
159impl DataOperation {
160    /// All operations, in the canonical registry order. Use this for exhaustive
161    /// fail-closed checks so a newly added operation cannot be skipped.
162    pub const ALL: [Self; 7] = [
163        Self::Capture,
164        Self::Store,
165        Self::Replay,
166        Self::Backup,
167        Self::InternalAnalytics,
168        Self::ExternalRedistribution,
169        Self::Retention,
170    ];
171
172    pub fn as_str(self) -> &'static str {
173        match self {
174            Self::Capture => "capture",
175            Self::Store => "store",
176            Self::Replay => "replay",
177            Self::Backup => "backup",
178            Self::InternalAnalytics => "internal_analytics",
179            Self::ExternalRedistribution => "external_redistribution",
180            Self::Retention => "retention",
181        }
182    }
183
184    /// Persistence operations additionally require a verified entitlement.
185    pub fn is_persistence(self) -> bool {
186        matches!(
187            self,
188            Self::Store | Self::Replay | Self::Backup | Self::Retention
189        )
190    }
191}

303/// What a consumer requires before it may activate on a feed.
304#[derive(Clone, Debug, PartialEq, Eq)]
305pub struct MarketDataRequirement {
306    pub provider: String,
307    pub product: String,
308    pub venue: String,
309    pub segment: String,
310    pub depth_model: DepthModel,
311    pub min_depth: u32,
312    pub requires_trade_prints: bool,
313    pub requires_clean_snapshot: bool,
314    pub requires_source_continuity: bool,
315    pub requires_capture: bool,
316    pub requires_store: bool,
317    pub requires_replay: bool,
318    pub requires_backup: bool,
319    pub requires_internal_analytics: bool,
320    pub requires_external_redistribution: bool,
321    pub requires_retention: bool,
322}
323
324impl SourceProductRegistry {
325    pub fn load(path: impl AsRef<Path>) -> Result<Self, CapabilityError> {
326        let bytes = std::fs::read(path).map_err(|_| CapabilityError::Unreadable)?;
327        Self::from_bytes(&bytes)
328    }
329
330    pub fn from_bytes(bytes: &[u8]) -> Result<Self, CapabilityError> {
331        let registry: Self = serde_json::from_slice(bytes)
332            .map_err(|error| CapabilityError::InvalidRegistry(error.to_string()))?;
333        if registry.schema_version != SCHEMA_VERSION {
334            return Err(CapabilityError::UnsupportedSchema(registry.schema_version));
335        }
336        if registry.registry_id.trim().is_empty() || registry.products.is_empty() {
337            return Err(CapabilityError::InvalidRegistry(
338                "registry id and at least one product are required".to_owned(),
339            ));
340        }
341        Ok(registry)
342    }
343
344    /// Find the exact declared product. No venue or segment inference.
345    pub fn product(
346        &self,
347        provider: &str,
348        product: &str,
349        venue: &str,
350        segment: &str,
351    ) -> Result<&SourceProduct, CapabilityError> {
352        self.products
353            .iter()
354            .find(|candidate| {
355                candidate.provider == provider
356                    && candidate.product == product
357                    && candidate.venue == venue
358                    && candidate.segment == segment
359            })
360            .ok_or_else(|| {
361                CapabilityError::UnknownProduct(
362                    provider.to_owned(),
363                    product.to_owned(),
364                    venue.to_owned(),
365                    segment.to_owned(),
366                )
367            })
368    }
369
370    /// The single fail-closed authorization helper. `UNKNOWN`, `PROHIBITED`
371    /// and undeclared products are all denials; no caller may infer a right
372    /// from a successful connection or from a partial product identity.
373    pub fn authorize(
374        &self,
375        product_key: &ProductKey,
376        operation: DataOperation,
377    ) -> Result<(), CapabilityError> {
378        let declared = self.product(
379            &product_key.provider,
380            &product_key.product,
381            &product_key.venue,
382            &product_key.segment,
383        )?;
384        let right = declared.rights.right_for(operation);
385        if !right.state.is_permitted() {
386            return Err(CapabilityError::MissingRight(operation.as_str().to_owned()));
387        }
388        if operation.is_persistence() && !declared.entitlement.verified {
389            return Err(CapabilityError::EntitlementUnverified(
390                declared.entitlement.reference.clone(),
391            ));
392        }
393        Ok(())
394    }
395
396    /// Mirror the schema's cross-field coherence rules at runtime. A hand-edited
397    /// registry must not become an internally inconsistent contract.
398    fn check_coherence(&self, declared: &SourceProduct) -> Result<(), CapabilityError> {
399        let incoherent = (declared.incremental_updates == UpdateMode::SnapshotOnly
400            && !declared.snapshot)
401            || (declared.recovery.contains(&RecoveryMode::FreshSnapshot) && !declared.snapshot)
402            || (declared.depth_model == DepthModel::Mbo && !declared.order_level_identity);
403        if incoherent {
404            return Err(CapabilityError::InvalidRegistry(format!(
405                "product `{}/{}` violates the schema coherence rules",
406                declared.provider, declared.product
407            )));
408        }
409        Ok(())
410    }
411
412    /// Reject the requirement unless every field is satisfied by the declared
413    /// product. A five-level feed never satisfies a twenty-level requirement.
414    pub fn evaluate(&self, requirement: &MarketDataRequirement) -> Result<(), CapabilityError> {
415        let declared = self.product(
416            &requirement.provider,
417            &requirement.product,
418            &requirement.venue,
419            &requirement.segment,
420        )?;
421        self.check_coherence(declared)?;
422        if declared.depth_model != requirement.depth_model {
423            return Err(CapabilityError::DepthModelMismatch {
424                declared: declared.depth_model,
425                required: requirement.depth_model,
426            });
427        }
428        if declared.depth_levels < requirement.min_depth {
429            return Err(CapabilityError::InsufficientDepth {
430                declared: declared.depth_levels,
431                required: requirement.min_depth,
432            });
433        }
434        if requirement.requires_trade_prints && !declared.trade_prints {
435            return Err(CapabilityError::NoTradePrints);
436        }
437        if requirement.requires_clean_snapshot && !declared.snapshot {
438            return Err(CapabilityError::NoSnapshot);
439        }
440        if requirement.requires_source_continuity
441            && declared.source_sequence == SourceSequence::Unavailable
442        {
443            return Err(CapabilityError::NoSourceContinuity);
444        }
445        let key = declared.key();
446        let required_operations = [
447            (requirement.requires_capture, DataOperation::Capture),
448            (requirement.requires_store, DataOperation::Store),
449            (requirement.requires_replay, DataOperation::Replay),
450            (requirement.requires_backup, DataOperation::Backup),
451            (
452                requirement.requires_internal_analytics,
453                DataOperation::InternalAnalytics,
454            ),
455            (
456                requirement.requires_external_redistribution,
457                DataOperation::ExternalRedistribution,
458            ),
459            (requirement.requires_retention, DataOperation::Retention),
460        ];
461        for (needed, operation) in required_operations {
462            if needed {
463                self.authorize(&key, operation)?;
464            }
465        }
466        Ok(())
467    }
