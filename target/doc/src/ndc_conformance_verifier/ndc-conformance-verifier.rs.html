2//! Conformance executable for the canonical NDC v2 Profile 1 wire mapping.
3//!
4//! The binary reads a release directory containing canonical wire documents,
5//! runs the twelve-stage release verifier against that directory and writes a
6//! stage-by-stage diagnostic. It uses an injected test-only trusted publisher
7//! entry whose public key is derived from the fixed conformance seed
8//! `[0x11; 32]`; no private key material, release key, provider contact or
9//! market observation is present in this executable.
10//!
11//! Rights remain `UNKNOWN` under the current freeze. The injected rights
12//! evaluator therefore returns `Unknown` unconditionally. The expected result
13//! is a first refusal at the rights-admission stage after the structural,
14//! manifest, signature, artifact and byte-derived-ordering stages pass. No
15//! admission receipt is created and no private snapshot survives the refusal.
16

137impl PrivateStaging {
138    fn create() -> Result<Self, Box<dyn Error>> {
139        let nanos = SystemTime::now()
140            .duration_since(UNIX_EPOCH)
141            .map_err(|error| error.to_string())?
142            .as_nanos();
143        let path = env::temp_dir().join(format!(
144            "ndc-conformance-private-{}-{}",
145            std::process::id(),
146            nanos
147        ));
148        fs::create_dir(&path)?;
149        Ok(Self(path))
150    }
151
152    fn path(&self) -> &Path {
153        &self.0
154    }
155
156    fn has_entries(&self) -> Result<bool, Box<dyn Error>> {
157        Ok(fs::read_dir(self.path())?.next().is_some())
158    }
159}

161impl Drop for PrivateStaging {
162    fn drop(&mut self) {
163        fs::remove_dir_all(&self.0).ok();
164    }

167fn parse_arguments() -> Result<Arguments, Box<dyn Error>> {
168    let mut release = None;
169    let mut out = None;
170    let mut arguments = env::args_os().skip(1);
171    while let Some(argument) = arguments.next() {
172        match argument.to_str() {
173            Some("--release") => {
174                if release.is_some() {
175                    return Err("--release may appear once".into());
176                }
177                release = Some(arguments.next().ok_or("--release needs a directory")?);
178            }
179            Some("--out") => {
180                if out.is_some() {
181                    return Err("--out may appear once".into());
182                }
183                out = Some(arguments.next().ok_or("--out needs a file path")?);
184            }
185            Some("--help") => {
186                eprintln!(
187                    "usage: ndc-conformance-verifier --release <dir> --out <diagnostic.json>"
188                );
189                return Err("--help".into());
190            }
191            _ => return Err(format!("unknown argument: {}", argument.to_string_lossy()).into()),
192        }
193    }
194    Ok(Arguments {
195        release: PathBuf::from(release.ok_or("--release is required")?),
196        out: PathBuf::from(out.ok_or("--out is required")?),
197    })
198}

200fn locate_release(path: &Path) -> Result<(PathBuf, ReleaseLocator), Box<dyn Error>> {
201    let mut components: Vec<&OsStr> = path
202        .components()
203        .map(|component| component.as_os_str())
204        .collect();
205    while components
206        .last()
207        .is_some_and(|component| component.is_empty())
208    {
209        components.pop();
210    }
211    if components.len() < 4
212        || components
213            .iter()
214            .any(|component| *component == OsStr::new("."))
215    {
216        return Err("release path must be <root>/<publisher>/<dataset>/<release>".into());
217    }
218    let release_component = components.pop().ok_or("missing release component")?;
219    let dataset_component = components.pop().ok_or("missing dataset component")?;
220    let publisher_component = components.pop().ok_or("missing publisher component")?;
221    let to_text = |component: &OsStr| -> Result<String, Box<dyn Error>> {
222        component
223            .to_str()
224            .map(str::to_owned)
225            .ok_or_else(|| "release path component is not UTF-8".into())
226    };
227    let locator = ReleaseLocator {
228        publisher_id: to_text(publisher_component)?,
229        dataset_id: to_text(dataset_component)?,
230        release_id: to_text(release_component)?,
231    };
232    let root = components
233        .iter()
234        .fold(PathBuf::new(), |mut path, component| {
235            path.push(component);
236            path
237        });
238    Ok((root, locator))
239}

241fn conformance_registry(publisher_id: &str) -> Result<TrustedKeyRegistry, Box<dyn Error>> {
242    TrustedKeyRegistry::new(
243        vec![TrustedKeyEntry {
244            publisher_id: publisher_id.to_owned(),
245            key_id: CONFORMANCE_KEY_ID.to_owned(),
246            public_key_bytes: CONFORMANCE_PUBLIC_KEY,
247            algorithm: "Ed25519".to_owned(),
248            purpose: KeyPurpose::ReleaseSigning,
249            valid_from_ns: 1,
250            valid_until_ns: u64::MAX,
251            state: KeyState::Active,
252            evidence_reference: None,
253        }],
254        CONFORMANCE_REGISTRY_REVISION.to_owned(),
255    )
256    .map_err(|error| error.into())
257}

259/// Locally supplied TEST_ONLY payload registry for the conformance fixture.
260///
261/// No schema byte is read from the release. The digest is computed over this
262/// local binding; the fixture manifest declares the identical schema digest.
263fn conformance_payload_registry() -> Result<PayloadRegistryInput, Box<dyn Error>> {
264    let schema_bytes = release_verifier::payload_registry::structural_schema_bytes(
265        CONFORMANCE_EVENT_TYPE,
266        &[(CONFORMANCE_PAYLOAD_FIELD, "string")],
267    )?;
268    let schema = LocalSchemaInput::seal(
269        CONFORMANCE_EVENT_TYPE.to_owned(),
270        "payload-example".to_owned(),
271        "1".to_owned(),
272        "structural-v1".to_owned(),
273        schema_bytes,
274    );
275    Ok(PayloadRegistryInput::seal(
276        "conformance-payload-registry".to_owned(),
277        "nirnay-payload-registry-v1".to_owned(),
278        vec![schema],
279    ))
280}

299fn stage_diagnostic(passed: &[u8], refusal_stage: Option<u8>, code: Option<&str>) -> Vec<Value> {
300    STAGE_NAMES
301        .iter()
302        .enumerate()
303        .map(|(index, name)| {
304            let number = (index + 1) as u8;
305            let status = if passed.contains(&number) {
306                "PASS"
307            } else if refusal_stage == Some(number) {
308                "REFUSED"
309            } else {
310                "NOT_RUN"
311            };
312            let stage_code = if status == "PASS" {
313                "OK"
314            } else if status == "REFUSED" {
315                code.unwrap_or("REJECT_UNKNOWN")
316            } else {
317                "NOT_APPLICABLE"
318            };
319            json!({
320                "stage": number,
321                "name": name,
322                "status": status,
323                "code": stage_code,
324            })
325        })
326        .collect()
327}

329fn cleanup_value(cleanup: &Option<Box<CleanupDisposition>>) -> Value {
330    match cleanup.as_deref() {
331        None => Value::Null,
332        Some(CleanupDisposition::Completed(report)) => json!({
333            "cleanup_succeeded": report.is_complete(),
334            "cleanup_pending": false,
335            "code": "CLEANUP_COMPLETED",
336            "files_expected": report.files_expected,
337            "files_removed": report.files_removed,
338            "files_already_absent": report.files_already_absent,
339            "bytes_removed": report.bytes_removed,
340            "directory_empty": report.directory_empty,
341            "directory_sync_completed": report.directory_sync_completed,
342        }),
343        Some(CleanupDisposition::Pending(pending)) => json!({
344            "cleanup_succeeded": false,
345            "cleanup_pending": true,
346            "code": "CLEANUP_PENDING",
347            "staging_path": pending.staging_path.to_string_lossy(),
348            "remaining_entries": pending.remaining_entries,
349            "detail": pending.detail,
350        }),
351    }
352}

354fn diagnostic(
355    locator: &ReleaseLocator,
356    stages: Vec<Value>,
357    refusal: Option<Value>,
358    outcome: &str,
359    snapshot_survives: bool,
360    cleanup: Value,
361) -> Value {
362    let status_of = |stage: u8| -> &str {
363        stages
364            .iter()
365            .find(|entry| entry["stage"] == stage)
366            .and_then(|entry| entry["status"].as_str())
367            .unwrap_or("NOT_RUN")
368    };
369    json!({
370        "schema_version": 1,
371        "record_date": "2026-11-24",
372        "tool": "ndc-conformance-verifier",
373        "release": {
374            "publisher_id": locator.publisher_id,
375            "dataset_id": locator.dataset_id,
376            "release_id": locator.release_id,
377        },
378        "outcome": outcome,
379        "stages": stages,
380        "phase_summary": {
381            "structural": status_of(1),
382            "manifest": status_of(2),
383            "signature": status_of(3),
384            "artifact": status_of(5),
385            "ordering": status_of(5),
386            "rights_admission": status_of(6),
387        },
388        "refusal": refusal,
389        "cleanup": cleanup,
390        "admission_receipt": Value::Null,
391        "receipt_written": false,
392        "private_snapshot_survives": snapshot_survives,
393        "rights_state": "UNKNOWN",
394        "dataset_accepted": false,
395        "collection_enabled": false,
396        "live_execution_enabled": false,
397    })
398}

408fn run(arguments: &Arguments) -> Result<Value, Box<dyn Error>> {
409    let (root, locator) = locate_release(&arguments.release)?;
410    let manifest_bytes = fs::read(arguments.release.join("manifest.json"))?;
411    let manifest_value = parse_strict(&manifest_bytes).map_err(|error| error.to_string())?;
412    let parsed =
413        canonical_wire::parse_manifest(&manifest_value).map_err(|error| error.to_string())?;
414    let digest = hex_lower(&sha256(&canonical_bytes(&manifest_value)));
415    let request = canonical_wire::into_internal_request(&parsed, &digest)?;
416
417    let passed = Rc::new(RefCell::new(Vec::new()));
418    let staging = PrivateStaging::create()?;
419    let verifier = ReleaseVerifier::new(VerifierConfig {
420        root,
421        private_staging: staging.path().to_owned(),
422        registry: conformance_registry(&locator.publisher_id)?,
423        rights: Box::new(FreezeUnknownRights),
424        coverage: Box::new(NoPromotionCoverage),
425        lineage: Box::new(NoTrustedPredecessor),
426        policy: conformance_policy(),
427        verifier_commit: "conformance-verifier-no-commit".to_owned(),
428    })
429    .with_canonical_wire()
430    .with_payload_schema_registry(conformance_payload_registry()?)
431    .with_stage_observer({
432        let passed = Rc::clone(&passed);
433        move |stage| passed.borrow_mut().push(stage)
434    });
435
436    let outcome: Result<_, RefusalDiagnostic> =
437        verifier.verify_release_with_diagnostics(&locator, &request);
438    drop(verifier);
439    let observed = Rc::try_unwrap(passed)
440        .map_err(|_| "stage observer outlived verification".to_owned())?
441        .into_inner();
442    let survives = staging.has_entries()?;
443    match outcome {
444        Ok(_) => Ok(diagnostic(
445            &locator,
446            stage_diagnostic(&observed, None, None),
447            None,
448            "UNEXPECTED_RIGHTS_ADMISSION",
449            survives,
450            Value::Null,
451        )),
452        Err(failure) => {
453            let refusal = refusal_value(&failure.refusal);
454            Ok(diagnostic(
455                &locator,
456                stage_diagnostic(
457                    &observed,
458                    Some(failure.refusal.stage()),
459                    Some(failure.refusal.code()),
460                ),
461                Some(refusal),
462                "RIGHTS_ADMISSION_REFUSED",
463                survives,
464                cleanup_value(&failure.cleanup),
465            ))
466        }
467    }
468}

470fn main() -> ExitCode {
471    let arguments = match parse_arguments() {
472        Ok(arguments) => arguments,
473        Err(error) => {
474            eprintln!("conformance verifier arguments refused: {}", error);
475            return ExitCode::FAILURE;
476        }
477    };
478    let result = run(&arguments);
479    let report = match result {
480        Ok(report) => report,
481        Err(error) => {
482            let placeholder = ReleaseLocator {
483                publisher_id: "unresolved".to_owned(),
484                dataset_id: "unresolved".to_owned(),
485                release_id: "unresolved".to_owned(),
486            };
487            diagnostic(
488                &placeholder,
489                stage_diagnostic(&[], Some(1), Some("REJECT_SCHEMA_ENVELOPE")),
490                Some(json!({
491                    "stage": 1,
492                    "code": "REJECT_SCHEMA_ENVELOPE",
493                    "detail": error.to_string(),
494                })),
495                "PREFLIGHT_REFUSED",
496                false,
497                Value::Null,
498            )
499        }
500    };
501    let conforms = report["outcome"] == "RIGHTS_ADMISSION_REFUSED"
502        && report["refusal"]["stage"] == 6
503        && report["refusal"]["code"] == "REJECT_RIGHTS_EVIDENCE"
504        && report["admission_receipt"].is_null()
505        && report["private_snapshot_survives"] == false
506        && report["cleanup"]["cleanup_succeeded"] == true
507        && report["cleanup"]["cleanup_pending"] == false;
508    if let Some(parent) = arguments.out.parent()
509        && !parent.as_os_str().is_empty()
510        && let Err(error) = fs::create_dir_all(parent)
511    {
512        eprintln!("cannot create diagnostic parent directory: {}", error);
513        return ExitCode::FAILURE;
514    }
515    let encoded = match serde_json::to_string_pretty(&report) {
516        Ok(encoded) => encoded,
517        Err(error) => {
518            eprintln!("cannot encode diagnostic: {}", error);
519            return ExitCode::FAILURE;
520        }
521    };
522    if let Err(error) = fs::write(&arguments.out, encoded) {
523        eprintln!("cannot write diagnostic: {}", error);
524        return ExitCode::FAILURE;
525    }
526    if conforms {
527        ExitCode::SUCCESS
528    } else {
529        ExitCode::FAILURE
530    }
