1//! Account projection materialized by replaying the account journal.
2//!
3//! Derived from the internal Nirnay `crates/nirnay-engine/src/state.rs` at source snapshot
4//! `cc3a34c317a86a9984401efe90fa3f43d67a1787`.
5//!
6//! The projection is the accounting authority for the paper engine: cash,
7//! charges, signed positions and order state are derived only from journaled
8//! events. The earlier money and quantity definitions are replaced by the

25/// Run format of a journal, declared once by the genesis `account_opened`
26/// record's schema version and enforced for every fill that follows.
27/// Legacy (1) journals were written before fill provenance existed;
28/// provenance (2) journals require every fill to name its funding quote.
29pub(super) const FORMAT_PROVENANCE_LEGACY: u16 = 1;
30pub(super) const FORMAT_PROVENANCE_REQUIRED: u16 = 2;

33pub(super) enum ProjectionError {

36    #[error("event at sequence {sequence} is inconsistent: {reason}")]

38    #[error("domain arithmetic failed: {0}")]
39    Domain(#[from] DomainError),
40    #[error("market state failed: {0}")]

42}

84/// Gross value of a position against an exact doubled mark, rounded once
85/// to integer paise at the accounting boundary: `roundHalfEven(|q| × M₂ / 2)`
86/// with checked arithmetic. The half is never dropped before the
87/// multiplication, so odd marks and odd quantities stay exact.
88fn rounded_gross_paise(quantity: i64, mark: DoubledPaise) -> Option<Paise> {
89    let scaled = i128::from(quantity.unsigned_abs()).checked_mul(mark.value())?;
90    let quotient = scaled.div_euclid(2);
91    let rounded = if scaled.rem_euclid(2) == 1 && quotient % 2 != 0 {
92        quotient + 1
93    } else {
94        quotient
95    };
96    i64::try_from(rounded).ok().map(Paise)
97}

174impl Projection {
175    pub fn apply(&mut self, record: &StoredRecord) -> Result<(), ProjectionError> {
176        let sequence = record.sequence;
177        let envelope = &record.body.envelope;
178        if let Some(previous) = self.last_available_time
179            && envelope.times.available_time < previous
180        {
181            return Err(inconsistent(
182                sequence,
183                "logical availability time moved backwards",
184            ));
185        }
186        if self.last_sequence == 0
187            && !matches!(&record.body.payload, EventPayload::AccountOpened { .. })
188        {
189            return Err(ProjectionError::MissingAccountOpening);
190        }
191        match &record.body.payload {
192            EventPayload::AccountOpened {
193                account_id,
194                starting_cash,
195                trade_date,
196            } => {
197                if starting_cash.0 <= 0 || self.account_id.is_some() || sequence != 1 {
198                    return Err(inconsistent(
199                        sequence,
200                        "account_opened must occur once at sequence one",
201                    ));
202                }
203                // A v2 genesis is self-causative: its causation names the
204                // run root, i.e. this record's own event identity.
205                if envelope.schema_version == FORMAT_PROVENANCE_REQUIRED
206                    && envelope.causation_id.as_ref() != Some(&envelope.event_id)
207                {
208                    return Err(inconsistent(
209                        sequence,
210                        "schema version 2 account_opened must be self-causative",
211                    ));
212                }
213                // The genesis record declares the run format for the whole
214                // journal: legacy (1) or provenance-required (2). The
215                // declaration is journal-level — provenance acceptance is
216                // never decided per fill.
217                self.format_provenance = envelope.schema_version;
218                self.account_id = Some(*account_id);
219                self.trade_date = Some(*trade_date);
220                self.starting_cash = Some(*starting_cash);
221                self.cash = *starting_cash;
222                self.kill_switch_engaged = true;
223                self.reconciliation_blocked = true;
224                self.account_snapshot_available = false;
225            }
226            EventPayload::SessionStateObserved {
227                trade_date,
228                phase,
229                calendar_version,
230            } => {
231                if Some(*trade_date) != self.trade_date || calendar_version.trim().is_empty() {
232                    return Err(inconsistent(
233                        sequence,
234                        "session state does not match the stream trading date or calendar",
235                    ));
236                }
237                self.session_phase = Some(*phase);
238                self.calendar_version = Some(calendar_version.clone());
239            }
240            EventPayload::QuoteReceived { .. }
241            | EventPayload::LateQuoteObserved { .. }
242            | EventPayload::FeedGapDetected { .. } => self.market.apply_record(record)?,
243            EventPayload::DecisionMade { decision } => {
244                if let Some(intent) = &decision.intent {
245                    if decision.strategy_id != intent.strategy_id
246                        || decision.strategy_version != intent.strategy_version
247                        || decision.input_refs != intent.evidence
248                    {
249                        return Err(inconsistent(
250                            sequence,
251                            "decision and intent identities or evidence differ",
252                        ));
253                    }
254                    if self.intents.contains_key(&intent.intent_id) {
255                        return Err(inconsistent(sequence, "intent identity was reused"));
256                    }
257                    self.intents.insert(
258                        intent.intent_id,
259                        IntentProjection {
260                            decision: decision.clone(),
261                            risk: None,
262                            order_id: None,
263                        },
264                    );
265                }
266            }
267            EventPayload::RiskAssessed { assessment } => {
268                let intent = self.intents.get_mut(&assessment.intent_id).ok_or_else(|| {
269                    inconsistent(sequence, "risk assessment has no prior decision")
270                })?;
271                if intent.risk.is_some() {
272                    return Err(inconsistent(
273                        sequence,
274                        "intent has multiple risk assessments",
275                    ));
276                }
277                intent.risk = Some(assessment.clone());
278            }
279            EventPayload::OrderCreated { order } => {
280                if order.status != OrderStatus::Working
281                    || order.filled_quantity != Quantity::ZERO
282                    || order.average_fill_price.is_some()
283                    || order.fill_count != 0
284                    || order.filled_notional.0 != 0
285                {
286                    return Err(inconsistent(sequence, "new order has non-initial state"));
287                }
288                let intent = self
289                    .intents
290                    .get_mut(&order.intent.intent_id)
291                    .ok_or_else(|| inconsistent(sequence, "order has no prior decision"))?;
292                if !intent
293                    .risk
294                    .as_ref()
295                    .is_some_and(|assessment| assessment.accepted)
296                    || intent.order_id.is_some()
297                    || intent.decision.intent.as_ref() != Some(&order.intent)
298                {
299                    return Err(inconsistent(
300                        sequence,
301                        "order lacks a unique accepted risk decision",
302                    ));
303                }
304                intent.order_id = Some(order.order_id);
305                if self
306                    .orders
307                    .insert(
308                        order.order_id,
309                        OrderView {
310                            order: order.clone(),
311                            created_sequence: sequence,
312                            last_sequence: sequence,
313                        },
314                    )
315                    .is_some()
316                {
317                    return Err(inconsistent(sequence, "order identity was reused"));
318                }
319                self.order_times.push_back(envelope.times.available_time);
320            }
321            EventPayload::OrderTransitioned {
322                order_id, from, to, ..
323            } => {
324                let view = self.orders.get_mut(order_id).ok_or_else(|| {
325                    inconsistent(sequence, "transition references an unknown order")
326                })?;
327                if view.order.status != *from || !from.permits(*to) {
328                    return Err(inconsistent(
329                        sequence,
330                        "order state transition is not permitted",
331                    ));
332                }
333                let total = view.order.intent.quantity;
334                let filled = view.order.filled_quantity;
335                let state_matches_fills = match to {
336                    OrderStatus::Working | OrderStatus::Rejected => filled.is_zero(),
337                    OrderStatus::PartiallyFilled => !filled.is_zero() && filled < total,
338                    OrderStatus::Filled => filled == total,
339                    OrderStatus::CancelPending | OrderStatus::OutcomeUnknown => filled <= total,
340                    OrderStatus::Cancelled => filled < total,
341                };
342                if !state_matches_fills {
343                    return Err(inconsistent(
344                        sequence,
345                        "order status contradicts cumulative fill quantity",
346                    ));
347                }
348                view.order.status = *to;
349                view.last_sequence = sequence;
350                if *to == OrderStatus::OutcomeUnknown {
351                    self.reconciliation_blocked = true;
352                    self.account_snapshot_available = false;
353                }
354            }
355            EventPayload::FillBooked { fill, next_status } => {
356                if self.fills.contains_key(&fill.fill_id) || fill.fill_id.get() != self.next_fill_id
357                {
358                    return Err(inconsistent(
359                        sequence,
360                        "fill identity is repeated or out of sequence",
361                    ));
362                }
363                let view = self
364                    .orders
365                    .get(&fill.order_id)
366                    .ok_or_else(|| inconsistent(sequence, "fill references an unknown order"))?;
367                if fill.side != view.order.intent.side || fill.quantity.is_zero() {
368                    return Err(inconsistent(
369                        sequence,
370                        "fill side or quantity differs from its order",
371                    ));
372                }
373                let instrument_id = view.order.intent.instrument_id.clone();
374                let venue = view.order.intent.venue;
375                // ---- validation phase: no mutations until every check passes ----
376                if self.format_provenance == FORMAT_PROVENANCE_LEGACY {
377                    // Legacy journals: pre-v2 fills have no provenance by
378                    // contract; only budget and accounting still apply.
379                    if record.body.envelope.schema_version == 2 {
380                        return Err(inconsistent(
381                            sequence,
382                            "provenance event in a legacy journal",
383                        ));
384                    }
385                } else if self.format_provenance != FORMAT_PROVENANCE_REQUIRED {
386                    return Err(inconsistent(sequence, "unknown journal run format"));
387                } else {
388                    // Provenance-format journal: every fill must be a v2
389                    // causally enriched record authored by the engine.
390                    if record.body.envelope.schema_version == 1 {
391                        return Err(inconsistent(
392                            sequence,
393                            "legacy fill in a provenance journal",
394                        ));
395                    }
396                    validate_fill_provenance(
397                        &self.market,
398                        &record.body.envelope,
399                        &view.order.intent,
400                        fill,
401                        sequence,
402                        FillProvenanceRule {
403                            slippage_ticks: self.slippage_ticks,
404                            tick: self.fill_price_tick(&instrument_id, venue),
405                            max_quote_age_ns: self.max_quote_age_ns,
406                        },
407                    )?;
408                }
409                // Budget sufficiency pre-check (no mutation here).
410                let market_remaining = match fill.side {
411                    Side::Buy => self.market.ask_remaining(&instrument_id, venue),
412                    Side::Sell => self.market.bid_remaining(&instrument_id, venue),
413                };
414                if fill.quantity.get() > market_remaining {
415                    return Err(inconsistent(
416                        sequence,
417                        "fill exceeds the remaining liquidity budget",
418                    ));
419                }
420                let remaining = view
421                    .order
422                    .intent
423                    .quantity
424                    .checked_sub(view.order.filled_quantity)?;
425                if fill.quantity > remaining {
426                    return Err(inconsistent(
427                        sequence,
428                        "fill exceeds remaining order quantity",
429                    ));
430                }
431                if !view.order.status.permits(*next_status) && view.order.status != *next_status {
432                    return Err(inconsistent(
433                        sequence,
434                        "fill produces an invalid order state transition",
435                    ));
436                }
437                let new_filled = view.order.filled_quantity.checked_add(fill.quantity)?;
438                let expected_status = if new_filled == view.order.intent.quantity {
439                    OrderStatus::Filled
440                } else if view.order.status == OrderStatus::CancelPending {
441                    OrderStatus::CancelPending
442                } else {
443                    OrderStatus::PartiallyFilled
444                };
445                if *next_status != expected_status {
446                    return Err(inconsistent(
447                        sequence,
448                        "fill state does not match cumulative filled quantity",
449                    ));
450                }
451                let turnover = fill.price.notional(fill.quantity)?;
452                let new_notional = view.order.filled_notional.checked_add(turnover)?;
453                let weighted_price =
454                    i64::try_from(i128::from(new_notional.0) / i128::from(new_filled.get()))
455                        .map_err(|_| {
456                            inconsistent(sequence, "average fill price arithmetic failed")
457                        })?;
458                let average_fill_price = PricePaise::new(Paise(weighted_price))?;
459                let charge_total = fill.charges.total()?;
460                // Replay does not trust the recorded fee total. The opening
461                // ChargeBook must reproduce every component for this date,
462                // product, side, and turnover; a total that balances only
463                // because cash was adjusted is still a forged fill.
464                let expected_charges = self.replay_fill_charges(
465                    view.order.intent.product,
466                    fill.side,
467                    turnover,
468                    sequence,
469                )?;
470                if expected_charges != fill.charges {
471                    return Err(inconsistent(
472                        sequence,
473                        "fill fees do not match the charge schedule for its trade date",
474                    ));
475                }
476                let expected_cash_delta = match fill.side {
477                    Side::Buy => turnover.checked_add(charge_total)?.checked_neg()?,
478                    Side::Sell => turnover.checked_sub(charge_total)?,
479                };
480                if expected_cash_delta != fill.cash_delta {
481                    return Err(inconsistent(
482                        sequence,
483                        "fill cash delta differs from turnover and charges",
484                    ));
485                }
486                // Every derived value is computed before mutation. An
487                // arithmetic overflow here therefore rejects the fill
488                // without changing any projection field.
489                let position_delta = signed_quantity(fill.side, fill.quantity.get());
490                let current_position = self.positions.get(&instrument_id).copied().unwrap_or(0);
491                let next_position =
492                    i64::try_from(i128::from(current_position) + i128::from(position_delta))
493                        .map_err(|_| inconsistent(sequence, "position arithmetic overflow"))?;
494                let next_cash = self.cash.checked_add(fill.cash_delta)?;
495                let next_charges = self.charges_paid.checked_add(charge_total)?;
496                let next_fill_count = view
497                    .order
498                    .fill_count
499                    .checked_add(1)
500                    .ok_or_else(|| inconsistent(sequence, "fill counter overflow"))?;
501                let next_fill_id = fill
502                    .fill_id
503                    .get()
504                    .checked_add(1)
505                    .ok_or_else(|| inconsistent(sequence, "fill identifier space exhausted"))?;
506                // ---- mutation phase: no fallible arithmetic or validation ----
507                self.market.debit_validated_liquidity(
508                    &instrument_id,
509                    venue,
510                    fill.side,
511                    fill.quantity.get(),
512                );
513                if let Some(view) = self.orders.get_mut(&fill.order_id) {
514                    view.order.filled_quantity = new_filled;
515                    view.order.filled_notional = new_notional;
516                    view.order.average_fill_price = Some(average_fill_price);
517                    view.order.fill_count = next_fill_count;
518                    view.order.status = *next_status;
519                    view.last_sequence = sequence;
520                }
521                self.positions.insert(instrument_id, next_position);
522                self.cash = next_cash;
523                self.charges_paid = next_charges;
524                self.fills.insert(fill.fill_id, fill.clone());
525                self.next_fill_id = next_fill_id;
526            }
527            EventPayload::ReconciliationObserved { report } => {
528                let unresolved_unknown = self
529                    .orders
530                    .values()
531                    .any(|view| view.order.status == OrderStatus::OutcomeUnknown);
532                let actual_status =
533                    match (&report.observed_positions, &report.observed_active_orders) {
534                        (Some(positions), Some(orders)) => {
535                            if positions == &report.expected_positions
536                                && orders == &report.expected_active_orders
537                                && !unresolved_unknown
538                            {
539                                ReconciliationStatus::Matched
540                            } else {
541                                ReconciliationStatus::Mismatch
542                            }
543                        }
544                        _ => ReconciliationStatus::Unavailable,
545                    };
546                if report.status != actual_status {
547                    return Err(inconsistent(
548                        sequence,
549                        "reconciliation status contradicts its snapshots",
550                    ));
551                }
552                if report.expected_positions != self.position_snapshot()
553                    || report.expected_active_orders != self.active_order_ids()
554                {
555                    return Err(inconsistent(
556                        sequence,
557                        "reconciliation expected state differs from the ledger",
558                    ));
559                }
560                self.reconciliation_blocked = report.status != ReconciliationStatus::Matched;
561                self.account_snapshot_available =
562                    report.status != ReconciliationStatus::Unavailable;
563            }
564            EventPayload::KillSwitchChanged { enabled, .. } => {
565                if !enabled && (self.reconciliation_blocked || !self.account_snapshot_available) {
566                    return Err(inconsistent(
567                        sequence,
568                        "kill switch release requires a matched account snapshot",
569                    ));
570                }
571                self.kill_switch_engaged = *enabled;
572            }
573            EventPayload::RiskStateObserved {
574                daily_pnl,
575                account_snapshot_available,
576            } => {
577                self.daily_pnl = *daily_pnl;
578                self.account_snapshot_available = *account_snapshot_available;
579                if !account_snapshot_available {
580                    self.reconciliation_blocked = true;
581                }
582            }
583            EventPayload::BrokerReportObserved { applied, .. } => {
584                if !applied {
585                    self.reconciliation_blocked = true;
586                }
587            }
588            // The M3.5.2 ledger payloads belong to the account-ledger stream,
589            // not to the paper-engine stream. Refuse them rather than silently
590            // projecting economic facts into the wrong state machine.
591            EventPayload::AccountOpenedEvidenced { .. }
592            | EventPayload::ExecutionProjected { .. } => {
593                return Err(inconsistent(
594                    sequence,
595                    "ledger projection payload is not valid in the paper engine stream",
596                ));
597            }
598        }
599        self.last_sequence = sequence;
600        self.last_available_time = Some(envelope.times.available_time);
601        Ok(())
602    }

604    #[must_use]
605    pub fn position_snapshot(&self) -> BTreeMap<InstrumentId, i64> {
606        self.positions
607            .iter()
608            .filter_map(|(id, quantity)| (*quantity != 0).then_some((id.clone(), *quantity)))
609            .collect()
610    }

617            .collect()

634            .filter(|time| {

636                    .checked_sub(time.get())

644    pub fn trim_order_times(&mut self, now: LogicalNanos) {
645        while self.order_times.front().is_some_and(|time| {
646            now.get()
647                .checked_sub(time.get())
648                .is_some_and(|age| age >= 1_000_000_000)
649        }) {
650            self.order_times.pop_front();
651        }

699    pub fn view(&self) -> Option<AccountView> {
700        let account_id = self.account_id?;
701        let positions = self
702            .positions
703            .iter()
704            .filter_map(|(instrument_id, quantity)| {
705                if *quantity == 0 {
706                    return None;
707                }
708                // Gross value: multiply by the exact doubled mark BEFORE
709                // rounding (round-once at the accounting boundary).
710                let mark_price_doubled = self.market.mark_price_doubled(instrument_id);
711                let gross_paise =
712                    mark_price_doubled.and_then(|exact| rounded_gross_paise(*quantity, exact));
713                Some(PositionView {
714                    instrument_id: instrument_id.clone(),
715                    quantity: *quantity,
716                    mark_price_doubled,
717                    gross_paise,
718                })
719            })
720            .collect();
721        let open_orders = self
722            .orders
723            .values()
724            .filter(|view| view.order.status.active())
725            .cloned()
726            .collect();
727        Some(AccountView {
728            account_id,
729            cash: self.cash,
730            charges_paid: self.charges_paid,
731            daily_pnl: self.daily_pnl,
732            account_snapshot_available: self.account_snapshot_available,

748    pub fn gross_exposure_paise(&self) -> Option<Paise> {
749        let mut total = 0_i128;
750        for (instrument_id, quantity) in &self.positions {
751            if *quantity == 0 {
752                continue;
753            }
754            // Identical multiply-then-round boundary as PositionView.
755            let mark = self.market.mark_price_doubled(instrument_id)?;
756            let exposure = i128::from(rounded_gross_paise(*quantity, mark)?.0);
757            total = total.checked_add(exposure)?;
758        }
759        i64::try_from(total).ok().map(Paise)
760    }

846/// Prove that a (schema version 2) fill names the quote event that funded
847/// it, and that the fill respects the market-intent execution rules: the
848/// referenced quote must be the active book entry at the time of the fill
849/// (not an earlier quote for the same instrument), the book must be
850/// executable under the same feed-health gate as live matching, carry the
851/// displayed side (ask liquidity for buys, bid liquidity for sells), price
852/// exactly at the configured slippage rule applied to that displayed
853/// price, and respect the order limit.
854/// Inputs needed to reproduce the matcher against the opening
855/// configuration: the configured slippage, the instrument's assigned
856/// tick, and the quote-age threshold used by feed-health eligibility.
857struct FillProvenanceRule {
858    slippage_ticks: u32,
859    tick: Option<Paise>,
860    max_quote_age_ns: Option<u64>,
861}
862
863fn validate_fill_provenance(
864    market: &MarketState,
865    envelope: &EventEnvelope,
866    intent: &OrderIntent,
867    fill: &Fill,
868    sequence: u64,
869    rule: FillProvenanceRule,
870) -> Result<(), ProjectionError> {
871    let causation = envelope.causation_id.as_ref().ok_or_else(|| {
872        inconsistent(
873            sequence,
874            "schema version 2 fill carries no quote provenance",
875        )
876    })?;
877    let book = market
878        .book(&intent.instrument_id, intent.venue)
879        .ok_or_else(|| {
880            inconsistent(
881                sequence,
882                "fill references an instrument with no active book",
883            )
884        })?;
885    // The cited event must be the currently active book's own funding
886    // event: citations to any earlier quote for this instrument are
887    // rejected, so a forged fill cannot revive a stale quote level.
888    if book.journal_event_id != *causation {
889        return Err(inconsistent(
890            sequence,
891            "fill quote provenance is not the active book quote",
892        ));
893    }
894    if book.quote.instrument_id != intent.instrument_id {
895        return Err(inconsistent(
896            sequence,
897            "fill quote provenance names another instrument",
898        ));
899    }
900    let maximum_age_ns = rule.max_quote_age_ns.ok_or_else(|| {
901        inconsistent(
902            sequence,
903            "fill feed-health rule cannot be reproduced: quote-age threshold is absent",
904        )
905    })?;
906    if !matches!(
907        market.health(
908            &intent.instrument_id,
909            intent.venue,
910            envelope.times.available_time,
911            maximum_age_ns
912        ),
913        FeedHealth::Healthy { .. }
914    ) {
915        return Err(inconsistent(
916            sequence,
917            "fill cites a book that was not healthy at fill time",
918        ));
919    }
920    let displayed = match fill.side {
921        Side::Buy => book.quote.ask_price,
922        Side::Sell => book.quote.bid_price,
923    }
924    .ok_or_else(|| {
925        inconsistent(
926            sequence,
927            "fill is funded by a side the referenced quote does not display",
928        )
929    })?;
930    // The matcher prices a fill by shifting the displayed price by the
931    // configured slippage in ticks, where the tick is the instrument's
932    // assigned tick for the term interval. Replay must reproduce that
933    // exact price: a fill at any other price — even one inside the order
934    // limit — is a price the engine would never have generated.
935    let tick = rule.tick.ok_or_else(|| {
936        inconsistent(
937            sequence,
938            "fill price rule cannot be reproduced: instrument tick terms are absent",
939        )
940    })?;
941    let signed_slippage = match fill.side {
942        Side::Buy => i64::from(rule.slippage_ticks),
943        Side::Sell => -i64::from(rule.slippage_ticks),
944    };
945    let expected_price = displayed
946        .shifted_ticks(tick, signed_slippage)
947        .map_err(|_| inconsistent(sequence, "fill price rule arithmetic failed"))?;
948    if fill.price != expected_price {
949        return Err(inconsistent(
950            sequence,
951            "fill price does not match the configured slippage rule",
952        ));
953    }
954    let within_limit = match fill.side {
955        Side::Buy => fill.price <= intent.limit_price,
956        Side::Sell => fill.price >= intent.limit_price,
957    };
958    if !within_limit {
959        return Err(inconsistent(
960            sequence,
961            "fill price violates the order limit",
962        ));
963    }
964    Ok(())
965}
