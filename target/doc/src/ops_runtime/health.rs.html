1//! Operational health metrics and the gate they impose on new risk.
2//!
3//! Monitoring is not decorative: when a safety condition fails, new exposure
4//! must be denied. This module produces the verdict; the runtime wires it to
5//! the risk authority at the integration stage.
6
7/// Thresholds for the paper/shadow runtime. Every field is mandatory.
8#[derive(Copy, Clone, Debug, PartialEq, Eq)]
9pub struct RuntimeThresholds {
10    pub max_feed_age_ns: u64,
11    pub max_journal_append_latency_ns: u64,
12    pub max_journal_queue_depth: u64,
13    pub min_disk_available_bytes: u64,
14    pub max_unknown_orders: u32,
15    pub max_reconciliation_differences: u32,
16    pub max_sequence_gaps: u32,
17}
18
19/// A point-in-time operational snapshot.
20#[derive(Clone, Debug, PartialEq, Eq)]
21pub struct HealthSnapshot {
22    pub feed_age_ns: u64,
23    pub sequence_gaps: u32,
24    pub reconnect_count: u32,
25    pub journal_append_latency_ns: u64,
26    pub journal_queue_depth: u64,
27    pub disk_available_bytes: u64,
28    pub open_orders: u32,
29    pub unknown_orders: u32,
30    pub reconciliation_differences: u32,
31    pub kill_switch_engaged: bool,
32    pub account_snapshot_available: bool,
33}
34
35/// Verdict on whether new risk-increasing decisions may proceed.
36#[derive(Clone, Debug, PartialEq, Eq)]
37pub struct HealthVerdict {
38    pub safe_for_new_risk: bool,
39    pub reasons: Vec<String>,
40}
41
42impl HealthSnapshot {
43    /// Evaluate the snapshot against the operating thresholds.
44    pub fn evaluate(&self, thresholds: &RuntimeThresholds) -> HealthVerdict {
45        let mut reasons = Vec::new();
46        if self.feed_age_ns > thresholds.max_feed_age_ns {
47            reasons.push("feed is stale".to_owned());
48        }
49        if self.sequence_gaps > thresholds.max_sequence_gaps {
50            reasons.push("sequence gaps are unresolved".to_owned());
51        }
52        if self.journal_append_latency_ns > thresholds.max_journal_append_latency_ns {
53            reasons.push("journal append latency exceeded".to_owned());
54        }
55        if self.journal_queue_depth > thresholds.max_journal_queue_depth {
56            reasons.push("journal queue backlog exceeded".to_owned());
57        }
58        if self.disk_available_bytes < thresholds.min_disk_available_bytes {
59            reasons.push("disk capacity below minimum".to_owned());
60        }
61        if self.unknown_orders > thresholds.max_unknown_orders {
62            reasons.push("UNKNOWN orders exceed the limit".to_owned());
63        }
64        if self.reconciliation_differences > thresholds.max_reconciliation_differences {
65            reasons.push("account reconciliation differs".to_owned());
66        }
67        if self.kill_switch_engaged {
68            reasons.push("kill switch engaged".to_owned());
69        }
70        if !self.account_snapshot_available {
71            reasons.push("account snapshot unavailable".to_owned());
72        }
73        HealthVerdict {
74            safe_for_new_risk: reasons.is_empty(),
75            reasons,
76        }
77    }
78
79    /// Prometheus-style metric text. No external metrics dependency.
80    #[must_use]
81    pub fn metrics_text(&self) -> String {
82        format!(
83            "nirnay_feed_age_ns {}\n\
84             nirnay_sequence_gaps {}\n\
85             nirnay_reconnect_count {}\n\
86             nirnay_journal_append_latency_ns {}\n\
87             nirnay_journal_queue_depth {}\n\
88             nirnay_disk_available_bytes {}\n\
89             nirnay_open_orders {}\n\
90             nirnay_unknown_orders {}\n\
91             nirnay_reconciliation_differences {}\n\
92             nirnay_kill_switch_engaged {}\n\
93             nirnay_account_snapshot_available {}\n",
94            self.feed_age_ns,
95            self.sequence_gaps,
96            self.reconnect_count,
97            self.journal_append_latency_ns,
98            self.journal_queue_depth,
99            self.disk_available_bytes,
100            self.open_orders,
101            self.unknown_orders,
102            self.reconciliation_differences,
103            u8::from(self.kill_switch_engaged),
104            u8::from(self.account_snapshot_available),
105        )
106    }
107}
