1//! Input contract: a verified source observation supplied to the kernel.
2//!
3//! Raw bytes remain the evidence authority upstream. The kernel receives the
4//! exact verified frame bytes and capture metadata, plus an instrument-local
5//! canonical payload for identity derivation. A multi-instrument frame produces
6//! several observations sharing one raw-frame commitment.
7
8use serde::{
9    Deserialize, //
10    Serialize,
11};
12
13use market_event::{
14    ContentId, //
15    OriginClass,
16    SequenceScope,
17};
18
19/// One verified, instrument-local source observation.
20#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
21pub struct SourceObservation {
22    /// Identity of the original raw frame (shared across instruments).
23    pub raw_frame_hash: ContentId,
24    /// Provider or venue source identity.
25    pub source_id: String,
26    /// Capture session identity from the verified session manifest.
27    pub source_session_id: String,
28    /// Provider-specific instrument identity from the frame.
29    pub provider_instrument_id: String,
30    /// Canonical bytes of this instrument's portion of the payload.
31    pub canonical_payload: Vec<u8>,
32    /// Origin of the observation. Exchange origin requires evidence.
33    pub origin_class: OriginClass,
34    /// Local UTC receive time, ns since epoch.
35    pub receive_utc_ns: i64,
36    /// Local monotonic receive time.
37    pub receive_monotonic_ns: u64,
38    /// Source-provided timestamp and clock domain, if any.
39    pub source_time_ns: Option<i64>,
40    pub source_clock_domain: Option<String>,
41    /// Feed or channel identity when a source multiplexes instruments on one
42    /// ordered channel. Channel sequence belongs to the channel, not to each
43    /// instrument inside a frame.
44    pub channel_id: Option<String>,
45    /// Connection generation and receive fence.
46    pub connection_epoch: u64,
47    pub receive_sequence: u64,
48    /// Source sequence and its declared scope.
49    pub sequence: Option<u64>,
50    pub sequence_scope: SequenceScope,
51    /// Exact reference revision used for resolution, if resolved.
52    pub reference_revision: Option<String>,
53    /// Opaque Nirnay instrument identity, or `None` when unresolved.
54    pub instrument_id: Option<String>,
55    /// Optional evidence token (e.g. exchange-origin evidence).
56    pub evidence_token: Option<String>,
57    /// Typed payload carried through the kernel to the book. The envelope
58    /// still owns identity and quality; this is not a second authority.
59    pub payload: Option<market_event::MarketDataEvent>,
60}
61
62impl SourceObservation {
63    /// Validate the input before any identity is derived.
64    pub fn validate(&self) -> Result<(), crate::KernelError> {
65        if self.source_id.is_empty() {
66            return Err(crate::KernelError::EmptySource);
67        }
68        if self.provider_instrument_id.is_empty() {
69            return Err(crate::KernelError::EmptyProviderInstrument);
70        }
71        if self.canonical_payload.is_empty() {
72            return Err(crate::KernelError::EmptyFrame);
73        }
74        if self.receive_utc_ns <= 0 {
75            return Err(crate::KernelError::InvalidReceiveTime);
76        }
77        // A resolved instrument needs an exact revision. The revision may be
78        // supplied on the observation or inherited from the session, so the
79        // resolution check happens in `project_observation` where both are
80        // visible.
81        Ok(())
82    }
83}
