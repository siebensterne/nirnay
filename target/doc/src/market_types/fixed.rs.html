1//! Exact decimal parsing and rescaling for exchange wire values.
2//!
3//! The SBE challenge schema represents both dollars and contract counts as
4//! signed 64-bit integers scaled by 1,000,000. Exchange APIs publish decimal
5//! strings with venue-specific precision, and non-prediction instruments need
6//! arbitrary scales and signed prices. All conversion therefore goes through a
7//! configured scale here, using `i128` intermediates and checked conversions.
8//! Nothing is rounded silently: an inexact conversion is an error.
9
10use anyhow::{Context, bail};
11
12/// Number of fixed-point units in one canonical SBE dollar or contract/share.
13pub const SCALE: i64 = 1_000_000;
14/// Number of decimal places represented by [`SCALE`].
15pub const DECIMALS: usize = 6;
16/// Maximum supported decimal scale for an `i64` coefficient.
17///
18/// `10^18` is representable in `i64`; `10^19` is not.
19pub const MAX_SCALE: u8 = 18;
20
21/// Return `10^scale` as an `i64`, rejecting unsupported scales.
22pub fn scale_factor(scale: u8) -> anyhow::Result<i64> {
23    anyhow::ensure!(
24        scale <= MAX_SCALE,
25        "decimal scale {scale} exceeds maximum {MAX_SCALE}"
26    );
27    Ok(10i64.pow(u32::from(scale)))
28}
29
30/// Parse an exact signed decimal string into an integer coefficient at
31/// `scale` decimal places.
32///
33/// Accepts plain decimal notation only (no exponent, no thousands separators)
34/// and rejects more fractional digits than `scale` rather than rounding.
35pub fn parse_scaled(input: &str, scale: u8) -> anyhow::Result<i64> {
36    let text = input.trim();
37    if text.is_empty() {
38        bail!("empty fixed-point value");
39    }
40    if scale > MAX_SCALE {
41        bail!("decimal scale {scale} exceeds maximum {MAX_SCALE}");
42    }
43
44    let (negative, unsigned) = if let Some(unsigned) = text.strip_prefix('-') {
45        (true, unsigned)
46    } else {
47        (false, text.strip_prefix('+').unwrap_or(text))
48    };
49    if unsigned.is_empty() {
50        bail!("missing digits in fixed-point value `{input}`");
51    }
52
53    let mut pieces = unsigned.split('.');
54    let whole = pieces.next().unwrap_or_default();
55    let fraction = pieces.next();
56    if pieces.next().is_some() {
57        bail!("multiple decimal points in fixed-point value `{input}`");
58    }
59    if whole.is_empty() && fraction.is_none_or(str::is_empty) {
60        bail!("missing digits in fixed-point value `{input}`");
61    }
62    if !whole.bytes().all(|byte| byte.is_ascii_digit()) {
63        bail!("invalid whole-number digits in fixed-point value `{input}`");
64    }
65    if let Some(fraction) = fraction {
66        if !fraction.bytes().all(|byte| byte.is_ascii_digit()) {
67            bail!("invalid fractional digits in fixed-point value `{input}`");
68        }
69        if fraction.len() > usize::from(scale) {
70            bail!(
71                "fixed-point value `{input}` has {} fractional digits; maximum is {scale}",
72                fraction.len()
73            );
74        }
75    }
76
77    let whole_value: i128 = if whole.is_empty() {
78        0
79    } else {
80        whole
81            .parse()
82            .with_context(|| format!("fixed-point whole part out of range in `{input}`"))?
83    };
84    let mut fractional_value = 0i128;
85    if let Some(fraction) = fraction
86        && !fraction.is_empty()
87    {
88        let exponent = u32::from(scale) - u32::try_from(fraction.len())?;
89        let factor = 10i128.pow(exponent);
90        fractional_value = fraction
91            .parse::<i128>()
92            .with_context(|| format!("fixed-point fraction out of range in `{input}`"))?
93            * factor;
94    }
95
96    let magnitude = whole_value
97        .checked_mul(10i128.pow(u32::from(scale)))
98        .and_then(|value| value.checked_add(fractional_value))
99        .with_context(|| format!("fixed-point value `{input}` overflows"))?;
100    let signed = if negative { -magnitude } else { magnitude };
101    i64::try_from(signed).with_context(|| format!("fixed-point value `{input}` out of range"))
102}

104/// Parse a signed decimal string into millionths.
105pub fn parse_micros(input: &str) -> anyhow::Result<i64> {
106    parse_scaled(input, u8::try_from(DECIMALS)?)
107}

109/// Parse a prediction-market price in dollars and validate the challenge's
110/// `[0, 1]` probability range.
111///
112/// This guard belongs to the prediction-market adapters. The general numeric
113/// model uses [`parse_scaled`] and accepts any signed price.
114pub fn parse_price(input: &str) -> anyhow::Result<i64> {
115    let price = parse_micros(input)?;
116    anyhow::ensure!(
117        (0..=SCALE).contains(&price),
118        "price `{input}` outside [0, 1]"
119    );
120    Ok(price)
121}

123/// Parse a non-negative quantity in canonical six-decimal units.
124pub fn parse_quantity(input: &str) -> anyhow::Result<i64> {
125    parse_quantity_scaled(input, u8::try_from(DECIMALS)?)
126}
127
128/// Parse a non-negative quantity at a configured scale.
129pub fn parse_quantity_scaled(input: &str, scale: u8) -> anyhow::Result<i64> {
130    let quantity = parse_scaled(input, scale)?;
131    anyhow::ensure!(quantity >= 0, "negative quantity `{input}`");
132    Ok(quantity)
133}

135/// Format an integer coefficient at `scale` decimal places as a stable string.
136#[must_use]
137pub fn format_scaled(value: i64, scale: u8) -> String {
138    let Ok(scale_value) = scale_factor(scale) else {
139        return value.to_string();
140    };
141    let sign = if value < 0 { "-" } else { "" };
142    let magnitude = value.unsigned_abs();
143    let divisor = scale_value.unsigned_abs();
144    let whole = magnitude / divisor;
145    let fraction = magnitude % divisor;
146    if fraction == 0 {
147        format!("{sign}{whole}")
148    } else {
149        let width = usize::from(scale);
150        let fraction = format!("{fraction:0width$}");
151        format!("{sign}{whole}.{}", fraction.trim_end_matches('0'))
152    }
153}
154
155/// Format millionths as a stable decimal string, primarily for diagnostics.
156#[must_use]
157pub fn format_micros(value: i64) -> String {
158    format_scaled(value, u8::try_from(DECIMALS).unwrap_or(6))
159}

161/// Rescale an exact integer coefficient, rejecting inexact conversions and
162/// overflow.
163pub fn rescale_exact(value: i64, from_scale: u8, to_scale: u8) -> anyhow::Result<i64> {
164    anyhow::ensure!(
165        from_scale <= MAX_SCALE && to_scale <= MAX_SCALE,
166        "decimal scale exceeds maximum {MAX_SCALE}"
167    );
168    let value = i128::from(value);
169    let rescaled = if to_scale >= from_scale {
170        let factor = 10i128.pow(u32::from(to_scale - from_scale));
171        value
172            .checked_mul(factor)
173            .context("rescaling overflows i128")?
174    } else {
175        let divisor = 10i128.pow(u32::from(from_scale - to_scale));
176        if value % divisor != 0 {
177            bail!(
178                "value {value} at scale {from_scale} is not exactly representable at scale {to_scale}"
179            );
180        }
181        value / divisor
182    };
183    i64::try_from(rescaled).context("rescaled value outside i64 range")
184}

186/// Convert an instrument-scale coefficient to the canonical SBE scale.
187pub fn to_sbe_micros(value: i64, scale: u8) -> anyhow::Result<i64> {
188    rescale_exact(value, scale, u8::try_from(DECIMALS)?)
189}
190
191/// Convert a canonical SBE coefficient to an instrument scale.
192pub fn from_sbe_micros(value: i64, scale: u8) -> anyhow::Result<i64> {
193    rescale_exact(value, u8::try_from(DECIMALS)?, scale)
194}

196#[cfg(test)]
197mod tests {
198    use super::*;
199
200    #[test]
201    fn parses_exact_decimals() {
202        assert_eq!(parse_micros("0").unwrap(), 0);
203        assert_eq!(parse_micros("1").unwrap(), 1_000_000);
204        assert_eq!(parse_micros("0.08").unwrap(), 80_000);
205        assert_eq!(parse_micros("0.0800").unwrap(), 80_000);
206        assert_eq!(parse_micros(".5").unwrap(), 500_000);
207        assert_eq!(parse_micros("1.").unwrap(), 1_000_000);
208        assert_eq!(parse_micros("-54.00").unwrap(), -54_000_000);
209        assert_eq!(parse_micros("300.00").unwrap(), 300_000_000);
210        assert_eq!(parse_micros("219.217767").unwrap(), 219_217_767);
211    }
212
213    #[test]
214    fn parses_at_configured_scale_beyond_six_decimals() {
215        assert_eq!(parse_scaled("0.123456789", 9).unwrap(), 123_456_789);
216        assert_eq!(parse_scaled("-0.000000001", 9).unwrap(), -1);
217        assert_eq!(
218            parse_scaled("123456789012345678", 0).unwrap(),
219            123_456_789_012_345_678
220        );
221        assert_eq!(parse_scaled("1.2300", 4).unwrap(), 12_300);
222        assert!(parse_scaled("1.0000000001", 9).is_err());
223        assert!(parse_scaled("1", 19).is_err());
224    }
225
226    #[test]
227    fn rejects_inexact_or_malformed_values() {
228        for value in [
229            "",
230            ".",
231            "-",
232            "+",
233            "1.0000001",
234            "1e-3",
235            "1,000",
236            "nan",
237            "--1",
238            "−1",
239            "\u{FF11}\u{FF12}",
240        ] {
241            assert!(parse_micros(value).is_err(), "{value} should be rejected");
242        }
243    }
244
245    #[test]
246    fn validates_prediction_market_price_range() {
247        assert_eq!(parse_price("0.0000").unwrap(), 0);
248        assert_eq!(parse_price("1.0000").unwrap(), 1_000_000);
249        assert!(parse_price("1.000001").is_err());
250        assert!(parse_price("-0.1").is_err());
251        // The general parser does not impose the range.
252        assert_eq!(parse_scaled("-2.5", 6).unwrap(), -2_500_000);
253    }
254
255    #[test]
256    fn formats_cleanly_at_any_scale() {
257        assert_eq!(format_micros(80_000), "0.08");
258        assert_eq!(format_micros(-54_000_000), "-54");
259        assert_eq!(format_micros(219_217_767), "219.217767");
260        assert_eq!(format_scaled(123_456_789, 9), "0.123456789");
261        assert_eq!(format_scaled(-1, 9), "-0.000000001");
262        assert_eq!(format_scaled(42, 0), "42");
263    }
264
265    #[test]
266    fn rescales_exactly_or_rejects() {
267        assert_eq!(rescale_exact(12_300, 4, 6).unwrap(), 1_230_000);
268        assert_eq!(rescale_exact(1_230_000, 6, 4).unwrap(), 12_300);
269        assert_eq!(rescale_exact(-1, 6, 6).unwrap(), -1);
270        assert!(
271            rescale_exact(1, 6, 4).is_err(),
272            "1e-6 is not a 4-place value"
273        );
274        assert!(rescale_exact(12_345, 4, 2).is_err());
275        assert!(
276            to_sbe_micros(123_456_789, 9)
277                .unwrap_err()
278                .to_string()
279                .contains("not exactly")
280        );
281        assert_eq!(to_sbe_micros(123_456_000, 9).unwrap(), 123_456);
282    }
283
284    #[test]
285    fn round_trips_scale_conversion() {
286        for value in [-123_456_000_i64, -1_000, 0, 1_000, 123_456_000] {
287            let micros = rescale_exact(value, 9, 6).unwrap();
288            assert_eq!(rescale_exact(micros, 6, 9).unwrap(), value);
289        }
290    }
291
292    #[test]
293    fn rejects_overflow() {
294        assert!(parse_micros("9223372036855").is_err());
295        assert!(rescale_exact(i64::MAX, 0, 6).is_err());
296        assert!(parse_scaled("9223372036854775808", 0).is_err());
297    }
