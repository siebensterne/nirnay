1//! Kill-switch control state with owner-local append-only persistence.
2//!
3//! No new service or database is introduced: the file-backed implementation
4//! is a single fsynced JSON-lines file under the owner's runtime directory.
5//! Fail-closed default: with no readable event the switch is engaged.
6
7use std::{
8    fs::{File, OpenOptions},
9    io::{BufRead, BufReader, Write},
10    path::{Path, PathBuf},
11};
12
13use serde::{
14    Deserialize, //
15    Serialize,
16};
17use thiserror::Error;
18
19/// One authored kill-switch decision. Append-only; never rewritten.
20#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
21pub struct KillSwitchState {
22    pub enabled: bool,
23    pub actor_id: String,
24    pub reason: String,
25    pub revision: u64,
26    pub at_ns: i64,
27}
28
29#[derive(Debug, Clone, PartialEq, Eq, Error)]
30pub enum ControlError {
31    #[error("control store io failure: {0}")]
32    Io(String),
33    #[error("control store serialization failure: {0}")]
34    Serialize(String),
35    #[error("control store record is corrupt: {0}")]
36    Corrupt(String),
37}
38
39/// Minimal persistence contract. Implementations must make `append` durable
40/// before returning, and `load` must fail closed on unreadable data.
41pub trait ControlStore {
42    fn load_kill_switch(&self) -> Result<Option<KillSwitchState>, ControlError>;
43    fn append_kill_switch(&self, state: &KillSwitchState) -> Result<(), ControlError>;
44}
45
46/// In-memory store for callers that do not own a control directory.
47#[derive(Clone, Debug, Default)]
48pub struct NoopControlStore;
49
50impl ControlStore for NoopControlStore {
51    fn load_kill_switch(&self) -> Result<Option<KillSwitchState>, ControlError> {
52        Ok(None)
53    }
54
55    fn append_kill_switch(&self, _state: &KillSwitchState) -> Result<(), ControlError> {
56        Ok(())
57    }
58}
59
60/// Append-only file store. Every record is one JSON line followed by fsync.
61#[derive(Clone, Debug, PartialEq, Eq)]
62pub struct FileControlStore {
63    path: PathBuf,
64}
65
66impl FileControlStore {
67    #[must_use]
68    pub fn new(path: impl AsRef<Path>) -> Self {
69        Self {
70            path: path.as_ref().to_path_buf(),
71        }
72    }
73
74    #[must_use]
75    pub fn path(&self) -> &Path {
76        &self.path
77    }
78}
79
80impl ControlStore for FileControlStore {
81    fn load_kill_switch(&self) -> Result<Option<KillSwitchState>, ControlError> {
82        if !self.path.exists() {
83            return Ok(None);
84        }
85        let file = File::open(&self.path).map_err(|e| ControlError::Io(e.to_string()))?;
86        let mut latest = None;
87        for (index, line) in BufReader::new(file).lines().enumerate() {
88            let line = line.map_err(|e| ControlError::Io(e.to_string()))?;
89            if line.trim().is_empty() {
90                continue;
91            }
92            let state: KillSwitchState = serde_json::from_str(&line)
93                .map_err(|e| ControlError::Corrupt(format!("line {}: {e}", index + 1)))?;
94            if latest
95                .as_ref()
96                .is_none_or(|prior: &KillSwitchState| state.revision >= prior.revision)
97            {
98                latest = Some(state);
99            }
100        }
101        Ok(latest)
102    }
103
104    fn append_kill_switch(&self, state: &KillSwitchState) -> Result<(), ControlError> {
105        let mut file = OpenOptions::new()
106            .create(true)
107            .append(true)
108            .open(&self.path)
109            .map_err(|e| ControlError::Io(e.to_string()))?;
110        let line =
111            serde_json::to_string(state).map_err(|e| ControlError::Serialize(e.to_string()))?;
112        file.write_all(line.as_bytes())
113            .and_then(|()| file.write_all(b"\n"))
114            .and_then(|()| file.sync_all())
115            .map_err(|e| ControlError::Io(e.to_string()))
116    }
117}
