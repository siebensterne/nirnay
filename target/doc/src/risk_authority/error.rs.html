1//! Authority failures. Every rejection is fail-closed: no reservation is made.
2
3use thiserror::Error;
4use trading_core::RiskAssessment;
5
6use crate::observation::ObservationSource;
7
8/// Which measured operational limit was exceeded.
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
10pub enum OpsThresholdKind {
11    #[error("sequence gaps")]
12    SequenceGaps,
13    #[error("journal append latency")]
14    JournalAppendLatency,
15    #[error("journal queue depth")]
16    JournalQueueDepth,
17    #[error("disk capacity")]
18    DiskCapacity,
19    #[error("unknown orders")]
20    UnknownOrders,
21    #[error("reconciliation differences")]
22    ReconciliationDifferences,
23}
24
25#[derive(Debug, Clone, PartialEq, Eq, Error)]
26pub enum AuthorityError {
27    #[error("invalid order intent: {0}")]
28    InvalidIntent(String),
29    #[error("stale account snapshot: expected revision {expected}, current {current}")]
30    StaleSnapshot { expected: u64, current: u64 },
31    #[error("intent id `{0}` was reused with different terms")]
32    ConflictingIntent(String),
33    #[error("intent id `{0}` was already decided and rejected")]
34    PreviouslyRejected(String),
35    #[error("risk assessment rejected the intent: {0:?}")]
36    Rejected(RiskAssessment),
37    #[error("order notional overflowed i64 paise")]
38    NotionalOverflow,
39    #[error("unknown intent id `{0}`")]
40    UnknownIntent(String),
41    #[error("authority state mutex was poisoned")]
42    StatePoisoned,
43    #[error("observation receipt time is in the future relative to the authority clock")]
44    FutureObservation,
45    #[error("accounting overflow in `{0}`")]
46    AccountingOverflow(String),
47    #[error("invalid reservation: {0}")]
48    InvalidReservation(String),
49    #[error("intent id `{0}` was already consumed by a fill or release")]
50    AlreadyConsumed(String),
51    #[error("observation from `{observation_source}` is missing")]
52    MissingObservation {
53        observation_source: ObservationSource,
54    },
55    #[error(
56        "observation from `{observation_source}` is stale: age {age_ns} ns exceeds {max_age_ns} ns"
57    )]
58    StaleObservation {
59        observation_source: ObservationSource,
60        age_ns: u64,
61        max_age_ns: u64,
62    },
63    #[error("observation from `{observation_source}` has unsupported schema version {version}")]
64    UnsupportedObservationVersion {
65        observation_source: ObservationSource,
66        version: u32,
67    },
68    #[error("observation from `{observation_source}` has invalid clock ordering: {detail}")]
69    InvalidObservationTime {
70        observation_source: ObservationSource,
71        detail: String,
72    },
73    #[error("observation from `{observation_source}` declares the wrong source")]
74    ObservationSourceMismatch {
75        observation_source: ObservationSource,
76    },
77    #[error("digest evidence from `{observation_source}` failed verification")]
78    ObservationDigestMismatch {
79        observation_source: ObservationSource,
80    },
81    #[error("producer `{observation_source}` is not registered with this authority")]
82    UnregisteredProducer {
83        observation_source: ObservationSource,
84    },
85    #[error("observation token does not belong to producer `{observation_source}`")]
86    ProducerTokenMismatch {
87        observation_source: ObservationSource,
88    },
89    #[error("authenticated {action} requires a non-empty evidenced reason")]
90    MissingEvidence { action: String },
91    #[error("reconciliation report `{report_id}` blocks new risk: {differences:?}")]
92    ReconciliationMismatch {
93        report_id: String,
94        differences: Vec<String>,
95    },
96    #[error("kill switch engaged by `{actor_id}`: {reason}")]
97    KillSwitchEngaged { reason: String, actor_id: String },
98    #[error("operational limit exceeded: {kind} observed {observed}, limit {limit}")]
99    OpsThresholdExceeded {
100        kind: OpsThresholdKind,
101        observed: u64,
102        limit: u64,
103    },
104    #[error("kill-switch control store failure: {0}")]
105    ControlFailure(String),
106}
