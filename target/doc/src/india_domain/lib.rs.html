22/// Trading-calendar day classification for an exchange (task P1: calendars).
23#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
24#[serde(rename_all = "snake_case")]
25pub enum SessionDay {
26    /// Regular trading session.
27    Trading,
28    /// Holiday: no session.
29    Closed,
30    /// Special session (e.g. live-testing hour announced by the exchange).
31    Special,
32}
33
34/// One calendar day for a venue.
35#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
36#[serde(deny_unknown_fields)]
37pub struct CalendarDay {
38    pub venue: Venue,
39    pub date: jiff::civil::Date,
40    pub session: SessionDay,
41    /// Optional description (holiday name, special-session purpose).
42    pub note: Option<String>,
43}

45/// A venue trading calendar. Point-in-time: lookups are by date.
46#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
47#[serde(deny_unknown_fields)]
48pub struct TradingCalendar {
49    pub venue: Venue,
50    pub days: Vec<CalendarDay>,
51}
52
53impl TradingCalendar {
54    #[must_use]
55    pub fn is_trading_day(&self, date: jiff::civil::Date) -> bool {
56        self.days
57            .iter()
58            .any(|day| day.date == date && day.session != SessionDay::Closed)
59    }
60
61    /// The next trading day strictly after `date`, bounded by the calendar's
62    /// known entries. Returns None when the calendar has no later known
63    /// session; callers must fetch more calendar data rather than guess.
64    #[must_use]
65    pub fn next_trading_day(&self, date: jiff::civil::Date) -> Option<jiff::civil::Date> {
66        self.days
67            .iter()
68            .filter(|day| day.date > date && day.session != SessionDay::Closed)
69            .map(|day| day.date)
70            .min()
71    }
72
73    /// Whether the calendar has an entry for **every** date of the month
74    /// starting at `month_start`.
75    ///
76    /// A holiday file only lists exceptions, so its absence of a date is not
77    /// evidence that the date is a session *or* a holiday. Callers that need
78    /// "the last trading session of the month" must require this coverage
79    /// proof: a sparse calendar can otherwise make an early session look
80    /// like the last one.
81    #[must_use]
82    pub fn covers_month(&self, month_start: jiff::civil::Date) -> bool {
83        let mut expected = month_start;
84        loop {
85            if !self.days.iter().any(|day| day.date == expected) {
86                return false;
87            }
88            let Some(next) = expected.checked_add(jiff::Span::new().days(1)).ok() else {
89                return true;
90            };
91            if next.month() != month_start.month() || next.year() != month_start.year() {
92                return true;
93            }
94            expected = next;
95        }
96    }
97
98    /// Validate: unique dates per venue, consistent venue, chronologically
99    /// ordered entries. Rejects malformed calendars at construction/ingest.
100    pub fn validate(&self) -> Result<(), anyhow::Error> {
101        let mut previous: Option<jiff::civil::Date> = None;
102        let mut seen: std::collections::BTreeSet<jiff::civil::Date> =
103            std::collections::BTreeSet::new();
104        for day in &self.days {
105            anyhow::ensure!(
106                day.venue == self.venue,
107                "calendar for {:?} contains a day for {:?}",
108                self.venue,
109                day.venue
110            );
111            anyhow::ensure!(
112                seen.insert(day.date),
113                "calendar for {:?} contains duplicate date {}",
114                self.venue,
115                day.date
116            );
117            if let Some(last) = previous {
118                anyhow::ensure!(
119                    day.date >= last,
120                    "calendar for {:?} is not chronologically ordered",
121                    self.venue
122                );
123            }
124            previous = Some(day.date);
125        }
126        Ok(())
127    }
128}
