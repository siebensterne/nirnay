1//! Bitemporal provider-to-instrument mapping history.
2//!
3//! Vendor symbols/tokens are not canonical instrument identities. This
4//! history answers a mapping query on both valid time and system knowledge time.
5
6use jiff::{Timestamp, civil::Date};
7use serde::{Deserialize, Serialize};
8
9use crate::{InstrumentId, Segment, SourceProvenance, Venue};
10
11#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
12#[serde(deny_unknown_fields)]
13pub struct ProviderListingMapping {
14    pub mapping_id: String,
15    pub provider: String,
16    pub provider_instrument_id: String,
17    pub instrument_id: InstrumentId,
18    pub venue: Venue,
19    pub segment: Segment,
20    pub venue_symbol: String,
21    pub valid_from: Date,
22    pub valid_to: Option<Date>,
23    pub recorded_at: Timestamp,
24    pub provenance: SourceProvenance,
25}
26
27impl ProviderListingMapping {
28    pub fn validate(&self) -> Result<(), anyhow::Error> {
29        anyhow::ensure!(!self.mapping_id.trim().is_empty(), "mapping_id is required");
30        anyhow::ensure!(!self.provider.trim().is_empty(), "provider is required");
31        anyhow::ensure!(
32            !self.provider_instrument_id.trim().is_empty(),
33            "provider instrument id is required"
34        );
35        anyhow::ensure!(
36            !self.venue_symbol.trim().is_empty(),
37            "venue symbol is required"
38        );
39        anyhow::ensure!(
40            self.valid_to.is_none_or(|end| end >= self.valid_from),
41            "mapping validity interval is inverted"
42        );
43        self.provenance.validate()?;
44        anyhow::ensure!(
45            self.recorded_at >= self.provenance.retrieved_at,
46            "mapping knowledge time predates source retrieval"
47        );
48        anyhow::ensure!(
49            self.provenance.venue == self.venue && self.provenance.segment == self.segment,
50            "mapping source venue/segment does not match the mapping contract"
51        );
52        Ok(())
53    }
54}
55
56#[derive(Debug, Clone, Default)]
57pub struct ProviderListingHistory {
58    mappings: Vec<ProviderListingMapping>,
59}
60
61impl ProviderListingHistory {
62    pub fn append(&mut self, mapping: ProviderListingMapping) -> Result<(), anyhow::Error> {
63        mapping.validate()?;
64        anyhow::ensure!(
65            !self
66                .mappings
67                .iter()
68                .any(|old| old.mapping_id == mapping.mapping_id),
69            "duplicate provider listing mapping id {}",
70            mapping.mapping_id
71        );
72        self.mappings.push(mapping);
73        self.mappings.sort_by(|a, b| {
74            (
75                a.provider.as_str(),
76                a.provider_instrument_id.as_str(),
77                a.valid_from,
78                a.recorded_at,
79            )
80                .cmp(&(
81                    b.provider.as_str(),
82                    b.provider_instrument_id.as_str(),
83                    b.valid_from,
84                    b.recorded_at,
85                ))
86        });
87        Ok(())
88    }
89
90    #[must_use]
91    pub fn mappings(&self) -> &[ProviderListingMapping] {
92        &self.mappings
93    }
94
95    pub fn resolve(
96        &self,
97        provider: &str,
98        provider_instrument_id: &str,
99        venue: Venue,
100        valid_on: Date,
101        known_at: Timestamp,
102    ) -> Result<&ProviderListingMapping, anyhow::Error> {
103        let mut candidates = self.mappings.iter().filter(|mapping| {
104            mapping.provider == provider
105                && mapping.provider_instrument_id == provider_instrument_id
106                && mapping.venue == venue
107                && mapping.valid_from <= valid_on
108                && mapping.valid_to.is_none_or(|end| valid_on <= end)
109                && mapping.recorded_at <= known_at
110        });
111        let first = candidates.next().ok_or_else(|| anyhow::anyhow!(
112            "no point-in-time mapping for {provider}:{provider_instrument_id} on {valid_on} known at {known_at}"))?;
113        anyhow::ensure!(
114            candidates.next().is_none(),
115            "ambiguous point-in-time mapping for {provider}:{provider_instrument_id} on {valid_on}"
116        );
117        Ok(first)
118    }
119}
