1//! Canonical Indian financial instrument and listing domain (task P1-01/P1-02).
2//!
3//! Versioned wire contract: `schemas/india/instrument.v2.json` (NSE/BSE cash
4//! equities and ETFs). v1 is frozen. The canonical ingress path is
5//! `InstrumentWire` (untrusted) -> [`Instrument`] (validated); direct
6//! deserialization into domain types is unsupported. Instrument
7//! identity is platform-internal and never derived from a venue symbol.
8//! Exchanges (NSE/BSE) are venues; Zerodha/Groww are brokers and never part
9//! of instrument identity. Prices are integer paise, quantities integer lots.

61/// Platform-internal stable identity. Never a ticker, MIC, ISIN or token.
62/// The value is private: it can only be created through [`InstrumentId::new`],
63/// and deserialization goes through the same validation, so no caller can smuggle
64/// an invalid identity past the constructors.
65#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
66pub struct InstrumentId {
67    value: String,
68}

70impl InstrumentId {
71    pub fn new(value: impl Into<String>) -> Result<Self, anyhow::Error> {
72        let value = value.into();
73        anyhow::ensure!(
74            !value.is_empty()
75                && value
76                    .chars()
77                    .next()
78                    .is_some_and(|c| c.is_ascii_lowercase() || c.is_ascii_digit())
79                && value.chars().all(|c| c.is_ascii_lowercase()
80                    || c.is_ascii_digit()
81                    || c == '.'
82                    || c == '_'
83                    || c == '-'),
84            "invalid instrument id `{value}`: must be lowercase alphanumeric with . _ - separators"
85        );
86        Ok(Self { value })
87    }
88
89    /// The canonical identity string.
90    #[must_use]
91    pub fn as_str(&self) -> &str {
92        &self.value
93    }
94}

115/// ISIN: cross-venue identity where issued (empty only for indices). Kept as a
116/// private value validated through [`Isin::new`]; deserialization enforces the
117/// same format and ISO 6166 check-digit rules.
118#[derive(Debug, Clone, PartialEq, Eq, Hash)]
119pub struct Isin {
120    value: String,
121}
122
123impl Isin {
124    pub fn new(value: impl Into<String>) -> Result<Self, anyhow::Error> {
125        let value = value.into();
126        let bytes = value.as_bytes();
127        anyhow::ensure!(
128            bytes.len() == 12
129                && bytes[0].is_ascii_uppercase()
130                && bytes[1].is_ascii_uppercase()
131                && bytes[2..11].iter().all(|b| b.is_ascii_alphanumeric())
132                && bytes[11].is_ascii_digit(),
133            "invalid ISIN `{value}`: expected 2 letters + 9 alphanumerics + 1 check digit"
134        );
135        anyhow::ensure!(
136            Self::check_digit_valid(&value),
137            "invalid ISIN `{value}`: ISO 6166 check digit mismatch"
138        );
139        Ok(Self { value })
140    }
141
142    /// The ISIN string.
143    #[must_use]
144    pub fn as_str(&self) -> &str {
145        &self.value
146    }
147
148    /// ISO 6166 check digit: Luhn algorithm over the base-36 digit stream
149    /// produced by mapping letters to 10..=35.
150    #[must_use]
151    fn check_digit_valid(isin: &str) -> bool {
152        let mut digits: Vec<u32> = Vec::with_capacity(24);
153        for byte in isin.bytes().take(11) {
154            if byte.is_ascii_digit() {
155                digits.push(u32::from(byte - b'0'));
156            } else {
157                let letter = u32::from(byte.to_ascii_uppercase() - b'A') + 10;
158                digits.push(letter / 10);
159                digits.push(letter % 10);
160            }
161        }
162        // Luhn with doubling from the rightmost digit of the doubled stream.
163        let mut sum = 0u32;
164        let mut double = true;
165        for digit in digits.iter().rev() {
166            let mut value = *digit;
167            if double {
168                value *= 2;
169                if value > 9 {
170                    value -= 9;
171                }
172            }
173            double = !double;
174            sum += value;
175        }
176        let expected = (10 - (sum % 10)) % 10;
177        expected == u32::from(isin.as_bytes()[11] - b'0')
178    }
179}

285/// An exchange-applied, price-linked tick schedule: an ordered, contiguous
286/// set of bands covering `[0, +∞)`. Schedules are exchange data, not
287/// constants: they are ingested (see `src/venue/india/tick_schedule.rs`) and
288/// versioned by the dates of their source circular.
289#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
290#[serde(deny_unknown_fields)]
291pub struct TickSchedule {
292    /// Bands ordered by `lower_paise`, contiguous from `0` to unbounded.
293    pub bands: Vec<TickBand>,
294}
295
296impl TickSchedule {
297    /// Validate structure: non-empty, ordered, contiguous, positive ticks,
298    /// coverage from `0` to unbounded, unbounded only on the last band.
299    pub fn validate(&self) -> Result<(), anyhow::Error> {
300        anyhow::ensure!(!self.bands.is_empty(), "tick schedule has no bands");
301        for (index, band) in self.bands.iter().enumerate() {
302            anyhow::ensure!(
303                band.tick_paise > 0,
304                "tick schedule band {index} has a non-positive tick"
305            );
306            anyhow::ensure!(
307                band.lower_paise >= 0,
308                "tick schedule band {index} has a negative lower bound"
309            );
310            if let Some(upper) = band.upper_paise {
311                anyhow::ensure!(
312                    upper >= band.lower_paise,
313                    "tick schedule band {index} ends before it starts"
314                );
315                anyhow::ensure!(
316                    index + 1 < self.bands.len(),
317                    "tick schedule band {index} is bounded but follows the last band"
318                );
319            }
320        }
321        anyhow::ensure!(
322            self.bands[0].lower_paise == 0,
323            "tick schedule must start at 0, got {}",
324            self.bands[0].lower_paise
325        );
326        anyhow::ensure!(
327            self.bands
328                .last()
329                .is_some_and(|band| band.upper_paise.is_none()),
330            "tick schedule must end unbounded"
331        );
332        for pair in self.bands.windows(2) {
333            let expected = pair[0]
334                .upper_paise
335                .ok_or_else(|| anyhow::anyhow!("bounded band must be followed"))?
336                + 1;
337            anyhow::ensure!(
338                pair[1].lower_paise == expected,
339                "tick schedule bands are not contiguous at {} (want {expected})",
340                pair[1].lower_paise
341            );
342        }
343        Ok(())
344    }
345
346    /// Tick size (paise) for `price_paise`, or `None` when the price is
347    /// negative or uncovered (callers must fail closed).
348    #[must_use]
349    pub fn tick_for(&self, price_paise: i64) -> Option<i64> {
350        if price_paise < 0 {
351            return None;
352        }
353        self.bands
354            .iter()
355            .find(|band| {
356                band.lower_paise <= price_paise
357                    && band.upper_paise.is_none_or(|upper| price_paise <= upper)
358            })
359            .map(|band| band.tick_paise)
360    }
361
362    /// Whether `price_paise` sits on the grid of its band's tick.
363    #[must_use]
364    pub fn is_aligned(&self, price_paise: i64) -> bool {
365        self.tick_for(price_paise)
366            .is_some_and(|tick| price_paise % tick == 0)
367    }
368}

370/// One instrument's assigned tick for one term interval.
371///
372/// NSE assigns tick sizes *monthly, per instrument*: the tick that applies
373/// for the whole month M is determined by the closing price at the end of
374/// month M-1 (NSE circular CMTR67133, effective 2025-04-15; monthly review
375/// thereafter). The assignment is therefore a record — reference closing
376/// price, the month it governs, the assigned tick, and the schedule
377/// revision that produced it — not a live lookup against the price at
378/// decision time. A venue whose master assigns a per-security tick
379/// directly (BSE `New Tick Size`) leaves the two monthly fields `None`.
380///
381/// Trading engines must use `tick_paise` verbatim for the term interval;
382/// recomputing from the order price would wrongly change the tick
383/// whenever the price crosses a band boundary mid-month.
384#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
385#[serde(deny_unknown_fields)]
386pub struct TickAssignment {
387    /// Assigned tick size, in paise.
388    pub tick_paise: i64,
389    /// Reference closing price (paise) that fixed the assignment; `None`
390    /// for non-monthly assignments.
391    pub reference_close_paise: Option<i64>,
392    /// Trading date of the reference close. For an NSE monthly assignment
393    /// this is the last available close of the month immediately preceding
394    /// `effective_month`; an arbitrary price cannot be assigned.
395    pub reference_close_date: Option<Date>,
396    /// Identity of the dataset the reference close came from (e.g.
397    /// `nse-eod-close`). Never empty for a monthly assignment.
398    pub reference_source: Option<String>,
399    /// SHA-256 of the exact reference-close source bytes, so the price can
400    /// be reproduced from the archive. 64 lowercase hex digits.
401    pub reference_sha256_hex: Option<String>,
402    /// First day of the month this assignment governs; `None` for
403    /// non-monthly assignments.
404    pub effective_month: Option<Date>,
405    /// Exchange tick-schedule revision the assignment was computed under,
406    /// as `{source_id}:{source_format_version}`. Never empty.
407    pub schedule_revision: String,
408    /// Date from which that schedule revision is in force.
409    pub schedule_effective_from: Option<Date>,
410    /// SHA-256 of the exact schedule source bytes.
411    pub schedule_sha256_hex: Option<String>,
412    /// Exchange-calendar revision the assignment was determined under, as
413    /// `{source_id}:{source_format_version}`. Required for a monthly
414    /// assignment; `None` for a non-monthly assignment.
415    #[serde(default)]
416    pub calendar_revision: Option<String>,
417    /// SHA-256 of the exact holiday-calendar source bytes whose sessions
418    /// fixed the reference close date. Required for a monthly assignment;
419    /// `None` for a non-monthly assignment.
420    #[serde(default)]
421    pub calendar_sha256_hex: Option<String>,
422}

424/// `true` when `value` is a 64-digit hex SHA-256.
425#[must_use]
426pub fn is_sha256_hex(value: &str) -> bool {
427    value.len() == 64 && value.bytes().all(|byte| byte.is_ascii_hexdigit())
428}

430/// First day of `date`'s calendar month.
431#[must_use]
432pub fn month_start(date: Date) -> Date {
433    Date::new(date.year(), date.month(), 1).unwrap_or(date)
434}
435
436/// First day of the calendar month before `month_start`.
437fn previous_month_start(month_start: Date) -> Option<Date> {
438    let year = month_start.year();
439    let month = month_start.month();
440    let (year, month) = if month == 1 {
441        (year.checked_sub(1)?, 12)
442    } else {
443        (year, month - 1)
444    };
445    Date::new(year, month, 1).ok()
446}

448impl TickAssignment {
449    /// Validate the assignment record and its price-source lineage.
450    ///
451    /// A monthly assignment is only reproducible when it names the
452    /// reference close date, the dataset, that dataset's hash, the schedule
453    /// revision, the schedule hash, the calendar revision and the calendar
454    /// hash, and the effective month. The reference
455    /// close must fall in the month immediately preceding the effective
456    /// month — the NSE determination rule (previous month's last available
457    /// close) — so a close from the wrong month is rejected rather than
458    /// silently applied.
459    pub fn validate(&self) -> Result<(), anyhow::Error> {
460        anyhow::ensure!(self.tick_paise > 0, "assigned tick must be positive");
461        anyhow::ensure!(
462            !self.schedule_revision.trim().is_empty(),
463            "tick assignment source revision must not be empty"
464        );
465        if let Some(hash) = &self.schedule_sha256_hex {
466            anyhow::ensure!(
467                is_sha256_hex(hash),
468                "tick schedule source hash must be 64 hex digits"
469            );
470        }
471        let monthly = self.effective_month.is_some()
472            || self.reference_close_paise.is_some()
473            || self.reference_close_date.is_some()
474            || self.calendar_revision.is_some()
475            || self.calendar_sha256_hex.is_some();
476        if !monthly {
477            anyhow::ensure!(
478                self.reference_source.is_none() && self.reference_sha256_hex.is_none(),
479                "non-monthly tick assignment cannot carry a reference close source"
480            );
481            anyhow::ensure!(
482                self.calendar_revision.is_none() && self.calendar_sha256_hex.is_none(),
483                "non-monthly tick assignment cannot carry a calendar source"
484            );
485            return Ok(());
486        }
487        let month = self.effective_month.ok_or_else(|| {
488            anyhow::anyhow!("monthly tick assignment requires an effective month")
489        })?;
490        anyhow::ensure!(
491            month_start(month) == month,
492            "effective month must be the first day of a month"
493        );
494        let close = self
495            .reference_close_paise
496            .ok_or_else(|| anyhow::anyhow!("monthly tick assignment requires a reference close"))?;
497        anyhow::ensure!(close >= 0, "reference closing price must be non-negative");
498        let close_date = self.reference_close_date.ok_or_else(|| {
499            anyhow::anyhow!("monthly tick assignment requires a reference close date")
500        })?;
501        let previous = previous_month_start(month)
502            .ok_or_else(|| anyhow::anyhow!("effective month {month} has no preceding month"))?;
503        anyhow::ensure!(
504            close_date >= previous && close_date < month,
505            "reference close {close_date} must fall in {previous}..{month} (the month before the assignment)"
506        );
507        let source = self.reference_source.as_deref().unwrap_or_default();
508        anyhow::ensure!(
509            !source.trim().is_empty(),
510            "monthly tick assignment requires the reference dataset identity"
511        );
512        anyhow::ensure!(
513            is_sha256_hex(self.reference_sha256_hex.as_deref().unwrap_or_default()),
514            "monthly tick assignment requires the reference source hash (64 hex digits)"
515        );
516        anyhow::ensure!(
517            is_sha256_hex(self.schedule_sha256_hex.as_deref().unwrap_or_default()),
518            "monthly tick assignment requires the schedule source hash (64 hex digits)"
519        );
520        let calendar_revision = self.calendar_revision.as_deref().unwrap_or_default();
521        anyhow::ensure!(
522            !calendar_revision.trim().is_empty(),
523            "monthly tick assignment requires the calendar source revision"
524        );
525        anyhow::ensure!(
526            is_sha256_hex(self.calendar_sha256_hex.as_deref().unwrap_or_default()),
527            "monthly tick assignment requires the calendar source hash (64 hex digits)"
528        );
529        let schedule_from = self.schedule_effective_from.ok_or_else(|| {
530            anyhow::anyhow!("monthly tick assignment requires the schedule revision date")
531        })?;
532        anyhow::ensure!(
533            schedule_from <= month,
534            "schedule revision {schedule_from} must be in force by the assignment month {month}"
535        );
536        Ok(())
537    }
538}

619    /// Validate structural invariants: identity rules, derivative linkage,
620    /// option strikes, and chronological corporate actions.
621    pub fn validate(&self) -> Result<(), anyhow::Error> {
622        anyhow::ensure!(
623            !self.name.is_empty(),
624            "instrument {} has an empty name",
625            self.instrument_id
626        );
627        let derivative = matches!(
628            self.instrument_type,
629            InstrumentType::Future | InstrumentType::Option
630        );
631        if derivative {
632            anyhow::ensure!(
633                matches!(&self.underlying_instrument_id, Some(underlying) if *underlying != self.instrument_id),
634                "derivative {} must reference a different underlying instrument",
635                self.instrument_id
636            );
637            anyhow::ensure!(
638                self.expiry_date.is_some(),
639                "derivative {} must have an expiry date",
640                self.instrument_id
641            );
642        }
643        if self.instrument_type == InstrumentType::Option {
644            anyhow::ensure!(
645                self.strike_price_paise.is_some() && self.option_kind.is_some(),
646                "option {} must have a strike price and kind",
647                self.instrument_id
648            );
649            anyhow::ensure!(
650                self.strike_price_paise.is_some_and(|strike| strike.0 > 0),
651                "option {} strike price must be positive",
652                self.instrument_id
653            );
654        }
655        if self.instrument_type != InstrumentType::Index {
656            anyhow::ensure!(
657                self.isin.is_some(),
658                "non-index instrument {} must carry an ISIN",
659                self.instrument_id
660            );
661        }
662        anyhow::ensure!(
663            !self.listings.is_empty(),
664            "instrument {} must have at least one listing",
665            self.instrument_id
666        );
667        let mut previous: Option<Date> = None;
668        for action in &self.corporate_actions {
669            if let Some(last) = previous {
670                anyhow::ensure!(
671                    action.effective_date >= last,
672                    "corporate actions for {} must be chronologically ordered",
673                    self.instrument_id
674                );
675            }
676            anyhow::ensure!(
677                action.numerator > 0 && action.denominator > 0,
678                "corporate action for {} has a non-positive adjustment ratio",
679                self.instrument_id
680            );
681            anyhow::ensure!(
682                action.amount_paise.is_none_or(|amount| amount >= 0),
683                "corporate action for {} has a negative amount",
684                self.instrument_id
685            );
686            previous = Some(action.effective_date);
687        }
688        // Listings: a listing identity (venue, segment, symbol) may repeat
689        // only for historical intervals that do not overlap. Each listing
690        // must carry trading terms that are valid (ordered bounds cover
691        // their effective ranges) and non-overlapping.
692        let mut seen: Vec<(Venue, Segment, &str, Date, Option<Date>)> = Vec::new();
693        for listing in &self.listings {
694            anyhow::ensure!(
695                !listing.venue_symbol.is_empty(),
696                "instrument {} has an empty venue symbol",
697                self.instrument_id
698            );
699            if let Some(end) = listing.effective_to {
700                anyhow::ensure!(
701                    end >= listing.effective_from,
702                    "instrument {} has a listing ending before it starts",
703                    self.instrument_id
704                );
705            }
706            let overlaps = seen.iter().any(|(venue, segment, symbol, from, to)| {
707                *venue == listing.venue
708                    && *segment == listing.segment
709                    && *symbol == listing.venue_symbol
710                    && ranges_overlap(*from, *to, listing.effective_from, listing.effective_to)
711            });
712            anyhow::ensure!(
713                !overlaps,
714                "instrument {} has overlapping listings for {:?} {:?} `{}`",
715                self.instrument_id,
716                listing.venue,
717                listing.segment,
718                listing.venue_symbol
719            );
720            seen.push((
721                listing.venue,
722                listing.segment,
723                listing.venue_symbol.as_str(),
724                listing.effective_from,
725                listing.effective_to,
726            ));
727            if let Some(group) = &listing.settlement_group {
728                anyhow::ensure!(
729                    !group.trim().is_empty(),
730                    "instrument {} has an empty settlement group",
731                    self.instrument_id
732                );
733            }
734            anyhow::ensure!(
735                !listing.terms.is_empty(),
736                "instrument {} has a listing with no trading terms",
737                self.instrument_id
738            );
739            for term in &listing.terms {
740                if let Some(end) = term.effective_to {
741                    anyhow::ensure!(
742                        end >= term.effective_from,
743                        "instrument {} has listing terms ending before they start",
744                        self.instrument_id
745                    );
746                }
747                anyhow::ensure!(
748                    term.tick_assignment.validate().is_ok(),
749                    "instrument {} has an invalid tick assignment",
750                    self.instrument_id
751                );
752                anyhow::ensure!(
753                    term.market_lot > 0 && term.quantity_increment > 0,
754                    "instrument {} has non-positive listing terms",
755                    self.instrument_id
756                );
757            }
758            let terms_overlap = listing.terms.windows(2).any(|pair| {
759                ranges_overlap(
760                    pair[0].effective_from,
761                    pair[0].effective_to,
762                    pair[1].effective_from,
763                    pair[1].effective_to,
764                )
765            });
766            anyhow::ensure!(
767                !terms_overlap,
768                "instrument {} has overlapping listing terms",
769                self.instrument_id
770            );
771        }
772        Ok(())
773    }

775    /// Whether the instrument trades on the venue during the period implied
776    /// by its listings' effective ranges.
777    #[must_use]
778    pub fn lists_on(&self, venue: Venue, on_date: Date) -> bool {
779        self.listings.iter().any(|listing| {
780            listing.venue == venue
781                && listing.effective_from <= on_date
782                && listing.effective_to.is_none_or(|end| on_date <= end)
783                && listing.status == ListingStatus::Active
784        })
785    }
786
787    /// The listing that is active on `date` (status `Active` and interval
788    /// covering the date), if any. Product eligibility, settlement regime
789    /// and Z-category all come from this one state — never from a live
790    /// snapshot of the current master.
791    ///
792    /// When an instrument trades on several venues, callers must use
793    /// [`Instrument::effective_listing_on`] with the intended execution
794    /// venue: the first active listing is not necessarily the one the order
795    /// targets.
796    #[must_use]
797    pub fn effective_listing(&self, on_date: Date) -> Option<&Listing> {
798        self.listings.iter().find(|listing| {
799            listing.status == ListingStatus::Active
800                && listing.effective_from <= on_date
801                && listing.effective_to.is_none_or(|end| on_date <= end)
802        })
803    }
804
805    /// The active listing of `venue` on `date`, if any. This is the
806    /// venue-scoped lookup execution eligibility must use; it never falls
807    /// back to another venue's listing.
808    #[must_use]
809    pub fn effective_listing_on(&self, venue: Venue, on_date: Date) -> Option<&Listing> {
810        self.listings.iter().find(|listing| {
811            listing.venue == venue
812                && listing.status == ListingStatus::Active
813                && listing.effective_from <= on_date
814                && listing.effective_to.is_none_or(|end| on_date <= end)
815        })
816    }
817
818    /// Settlement regime in force on `date`, if the instrument is active.
819    #[must_use]
820    pub fn settlement_regime_on(&self, on_date: Date) -> Option<SettlementRegime> {
821        self.effective_listing(on_date)
822            .map(|listing| listing.settlement_regime)
823    }
824
825    /// Z-category designation in force on `date`, if the instrument is
826    /// active. `Some(false)` is a definite "not Z"; `None` means the
827    /// instrument is not active and the caller must fail closed.
828    #[must_use]
829    pub fn z_category_on(&self, on_date: Date) -> Option<bool> {
830        self.effective_listing(on_date)
831            .map(|listing| listing.z_category)
832    }
833
834    /// The exchange trading terms effective on `date`: the term set of the
835    /// listing currently active on that date whose validity interval
836    /// covers it. `None` when no listing is active or it has no term cover
837    /// on that date — callers must fail closed, never guess.
838    #[must_use]
839    pub fn effective_terms(&self, on_date: Date) -> Option<&ListingTerms> {
840        self.effective_listing(on_date)
841            .and_then(|listing| listing.terms.iter().find(|term| term.is_valid_on(on_date)))
842    }
843
844    /// The exchange trading terms effective on `date` for `venue`: the term
845    /// set of that venue's active listing. `None` when the venue has no
846    /// active listing or no covering term set — callers must fail closed.
847    #[must_use]
848    pub fn effective_terms_on(&self, venue: Venue, on_date: Date) -> Option<&ListingTerms> {
849        self.effective_listing_on(venue, on_date)
850            .and_then(|listing| listing.terms.iter().find(|term| term.is_valid_on(on_date)))
851    }
852}

854/// Whether two effective intervals [from, to] (open-ended when `to` is None)
855/// intersect. Intervals are inclusive at both ends.
856fn ranges_overlap(a_from: Date, a_to: Option<Date>, b_from: Date, b_to: Option<Date>) -> bool {
857    let a_starts_before_b_ends = b_to.is_none_or(|b_end| a_from <= b_end);
858    let b_starts_before_a_ends = a_to.is_none_or(|a_end| b_from <= a_end);
859    a_starts_before_b_ends && b_starts_before_a_ends
860}

1228/// Adjust a raw historical price (paise) across corporate actions so the
1229/// series is comparable through splits/bonuses. Applies actions strictly
1230/// after `as_of` chronologically. Integer-exact: returns None when the
1231/// adjustment is not representable in paise.
1232pub fn adjusted_price_paise(
1233    raw_paise: Paise,
1234    actions: &[CorporateAction],
1235    as_of: Date,
1236) -> anyhow::Result<Paise> {
1237    let mut value = i128::from(raw_paise.0);
1238    for action in actions
1239        .iter()
1240        .filter(|action| action.effective_date > as_of)
1241        .filter(|action| {
1242            matches!(
1243                action.action,
1244                CorporateActionKind::Split
1245                    | CorporateActionKind::Bonus
1246                    | CorporateActionKind::Consolidation
1247            )
1248        })
1249    {
1250        // A 1:2 split (num=1, den=2) doubles share count, halving the
1251        // adjusted price: new = old * numerator / denominator.
1252        anyhow::ensure!(
1253            action.numerator > 0 && action.denominator > 0,
1254            "corporate action has a non-positive adjustment ratio"
1255        );
1256        let scaled = value
1257            .checked_mul(i128::from(action.numerator))
1258            .ok_or_else(|| anyhow::anyhow!("adjustment of {value} paise overflows i128"))?;
1259        // Exact division only: truncation would silently corrupt history.
1260        anyhow::ensure!(
1261            scaled % i128::from(action.denominator) == 0,
1262            "adjustment of {value} paise by {}:{} is not representable in paise",
1263            action.numerator,
1264            action.denominator
1265        );
1266        value = scaled / i128::from(action.denominator);
1267    }
1268    if value > i128::from(i64::MAX) || value < 0 {
1269        anyhow::bail!("adjusted price {value} paise is outside i64 range");
1270    }
1271    Ok(Paise(value as i64))
1272}
