2//! Offline kernel session projector.
3//!
4//! Reads a verified kernel session JSON, projects it into canonical kernel
5//! events, and publishes the result exclusively with a digest sidecar. No
6//! network, broker or order behavior.
7

35#[derive(Debug, Parser)]
36#[command(
37    name = "kernel-session",
38    about = "Project a verified source session into canonical kernel events"
39)]
40struct Args {
41    /// Verified kernel session JSON input.
42    #[arg(long)]
43    input: PathBuf,
44    /// New kernel-events JSONL output.
45    #[arg(long)]
46    output: PathBuf,
47}
48
49fn project(input: &Path, output: &Path) -> anyhow::Result<(u64, String)> {
50    let bytes = std::fs::read(input).context("reading session input")?;
51    let session: KernelSession = serde_json::from_slice(&bytes).context("parsing session input")?;
52    let events = project_session(&session).context("projecting session")?;
53
54    let parent = output
55        .parent()
56        .filter(|path| !path.as_os_str().is_empty())
57        .unwrap_or(Path::new("."));
58    std::fs::create_dir_all(parent).context("creating output directory")?;
59    if output.exists() {
60        bail!("output already exists; kernel publication is exclusive");
61    }
62
63    let file = File::create_new(output).context("creating kernel event output")?;
64    let mut writer = BufWriter::new(file);
65    let mut events_written = 0_u64;
66    let mut hasher = Sha256::new();
67    for event in events {
68        let line = serde_json::to_vec(&event).context("serializing kernel event")?;
69        hasher.update(&line);
70        hasher.update(b"\n");
71        writer.write_all(&line).context("writing kernel event")?;
72        writer.write_all(b"\n").context("writing newline")?;
73        events_written += 1;
74    }
75    writer.flush().context("flushing output")?;
76    writer.get_ref().sync_all().context("syncing output")?;
77    let digest = format!("{:x}", hasher.finalize());
78    Ok((events_written, digest))
79}
80
81fn main() -> anyhow::Result<()> {
82    let args = Args::parse();
83    let (count, digest) = project(&args.input, &args.output)?;
84    eprintln!("events={count} sha256={digest}");
85    Ok(())
86}
