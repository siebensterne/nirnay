96/// Cleanup outcome recorded next to a first refusal.
97#[derive(Debug, Clone, PartialEq, Eq)]
98pub enum CleanupDisposition {
99    /// Cleanup observed an empty directory after a post-removal sync.
100    Completed(CleanupReport),
101    /// Cleanup stopped before an empty, synced directory was observed.
102    Pending(CleanupPending),
103}
104
105impl CleanupDisposition {
106    /// Report success only for a completed report with an empty synced directory.
107    pub fn is_complete(&self) -> bool {
108        matches!(self, Self::Completed(report) if report.is_complete())
109    }
110
111    /// Report pending for every outcome that did not prove cleanup completion.
112    pub fn is_pending(&self) -> bool {
113        matches!(self, Self::Pending(_))
114    }
115}

117/// A first refusal together with its independent cleanup outcome.
118#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
119#[error("{refusal}")]
120pub struct RefusalDiagnostic {
121    /// The original first refusal, which keeps its own stage and code.
122    pub refusal: VerifierError,
123    /// Cleanup attempt made after this refusal, once staging had started.
124    pub cleanup: Option<Box<CleanupDisposition>>,
125}
126
127type PipelineResult<T> = Result<T, Box<(VerifierError, Option<CleanupDisposition>)>>;
128
129fn no_cleanup<T>(result: Result<T, VerifierError>) -> PipelineResult<T> {
130    result.map_err(|error| Box::new((error, None)))
131}
132
133fn with_cleanup(
134    refusal: VerifierError,
135    cleanup: Result<CleanupReport, CleanupPending>,
136) -> (VerifierError, Option<CleanupDisposition>) {
137    let disposition = match cleanup {
138        Ok(report) => CleanupDisposition::Completed(report),
139        Err(pending) => CleanupDisposition::Pending(pending),
140    };
141    (refusal, Some(disposition))
142}

144/// Refusal from a checked in-flight artifact slot transition.
145#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
146pub enum InFlightSlotError {
147    /// No slot is free while the count equals the in-flight limit.
148    #[error("artifact in-flight limit reached")]
149    LimitReached,
150    /// The checked increment would wrap the counter type.
151    #[error("artifact in-flight count overflowed")]
152    Overflow,
153    /// A release was requested with no artifact in flight.
154    #[error("artifact in-flight count underflowed")]
155    Underflow,
156}
157
158impl From<InFlightSlotError> for VerifierError {
159    fn from(error: InFlightSlotError) -> Self {
160        match error {
161            InFlightSlotError::LimitReached
162            | InFlightSlotError::Overflow
163            | InFlightSlotError::Underflow => Self::ResourceLimit {
164                limit: "max_concurrent_artifacts".to_owned(),
165            },
166        }
167    }
168}

170/// Checked count of artifacts being verified at one instant.
171///
172/// The delivered pipeline is sequential: one slot is acquired, one artifact is
173/// verified and the slot is released before the next descriptor opens. A future
174/// scheduler may verify artifacts together, but it must acquire a slot through
175/// [`Self::try_acquire`] before an artifact starts and release it through
176/// [`Self::try_release`] after verification ends. The two transitions refuse
177/// overflow and underflow instead of wrapping the counter.
178#[derive(Debug, Clone, Copy, PartialEq, Eq)]
179pub struct InFlightArtifacts {
180    count: u32,
181    limit: u32,
182}
183
184impl InFlightArtifacts {
185    /// Create an empty counter with the configured in-flight limit.
186    pub fn new(limit: u32) -> Self {
187        Self { count: 0, limit }
188    }
189
190    /// Return the number of artifacts currently in flight.
191    pub fn in_flight(&self) -> u32 {
192        self.count
193    }
194
195    /// Return the configured in-flight limit.
196    pub fn limit(&self) -> u32 {
197        self.limit
198    }
199
200    /// Acquire one slot before an artifact starts verification.
201    ///
202    /// A request made while every slot is occupied is refused. The increment
203    /// is checked before the capacity test so a wrapped count is not stored.
204    pub fn try_acquire(self) -> Result<Self, InFlightSlotError> {
205        let next = self
206            .count
207            .checked_add(1)
208            .ok_or(InFlightSlotError::Overflow)?;
209        if next > self.limit {
210            return Err(InFlightSlotError::LimitReached);
211        }
212        Ok(Self {
213            count: next,
214            limit: self.limit,
215        })
216    }
217
218    /// Release one slot after artifact verification ends.
219    pub fn try_release(self) -> Result<Self, InFlightSlotError> {
220        let next = self
221            .count
222            .checked_sub(1)
223            .ok_or(InFlightSlotError::Underflow)?;
224        Ok(Self {
225            count: next,
226            limit: self.limit,
227        })
228    }
229
230    /// Construct a counter state for checked-transition tests.
231    #[cfg(test)]
232    pub(crate) fn with_count(limit: u32, count: u32) -> Self {
233        Self { count, limit }
234    }
235}

432    fn run_stages_streaming(
433        &self,
434        locator: &ReleaseLocator,
435        request: &AdmissionRequest,
436    ) -> PipelineResult<AdmissionReceipt> {
437        let policy = no_cleanup(
438            self.profile_resolver
439                .streaming_policy()
440                .map_err(ProfileError::into_verifier_error),
441        )?;
442        let verifier_limits = no_cleanup(
443            policy
444                .verifier_limits()
445                .map_err(ProfileError::into_verifier_error),
446        )?;
447        let snapshot_limits = no_cleanup(
448            policy
449                .snapshot_limits()
450                .map_err(ProfileError::into_verifier_error),
451        )?;
452        let mut budget = no_cleanup(FdBudget::with_minimum(
453            verifier_limits.max_open_file_descriptors(),
454            3,
455        ))?;
456        let root = no_cleanup(ReleaseRoot::open(&self.root, &mut budget))?;
457        let release_dir = no_cleanup(root.release_dir(
458            &locator.publisher_id,
459            &locator.dataset_id,
460            &locator.release_id,
461            &mut budget,
462        ))?;
463        let top_level = no_cleanup(enumerate_names(&release_dir))?;
464        let expected_top: HashSet<&str> = ["manifest.json", "signature.json", "artifacts"]
465            .into_iter()
466            .collect();
467        if top_level
468            .iter()
469            .map(String::as_str)
470            .collect::<HashSet<&str>>()
471            != expected_top
472        {
473            return Err(Box::new((
474                VerifierError::FileSet {
475                    detail: "release root file set differs from manifest, sidecar and artifacts"
476                        .to_owned(),
477                },
478                None,
479            )));
480        }
481        let (manifest_file, manifest_facts) = no_cleanup(open_regular_file(
482            &release_dir,
483            "manifest.json",
484            &mut budget,
485        ))?;
486        let manifest_bytes = no_cleanup(read_bounded(
487            &manifest_file,
488            &manifest_facts,
489            manifest_facts.size,
490            verifier_limits.max_preparation_intent_bytes(),
491            "max_preparation_intent_bytes",
492        ))?;
493        let (sidecar_file, sidecar_facts) = no_cleanup(open_regular_file(
494            &release_dir,
495            "signature.json",
496            &mut budget,
497        ))?;
498        let sidecar_bytes = no_cleanup(read_bounded(
499            &sidecar_file,
500            &sidecar_facts,
501            sidecar_facts.size,
502            verifier_limits.max_sealed_intent_bytes(),
503            "max_sealed_intent_bytes",
504        ))?;
505        drop(manifest_file);
506        drop(sidecar_file);
507
508        let staging_limits = crate::files::StagingLimits {
509            max_stored_bytes: snapshot_limits.max_stored_bytes_per_artifact(),
510            max_temp_disk_bytes: snapshot_limits.temp_disk_bytes(),
511            io_chunk_bytes: snapshot_limits.io_chunk_bytes(),
512        };
513        let mut builder = no_cleanup(crate::files::SnapshotBuilder::with_limits(
514            &self.private_staging,
515            &mut budget,
516            staging_limits,
517        ))?;
518        for (name, bytes) in [
519            ("manifest.json", manifest_bytes.as_slice()),
520            ("signature.json", sidecar_bytes.as_slice()),
521        ] {
522            if let Err(error) = builder.add(name, bytes) {
523                return Err(Box::new(with_cleanup(error, builder.discard_partial())));
524            }
525        }
526        let parsed = match self.stages_one_to_four(
527            &manifest_bytes,
528            &sidecar_bytes,
529            request,
530            verifier_limits.max_nesting_depth(),
531        ) {
532            Ok(value) => value,
533            Err(error) => return Err(Box::new(with_cleanup(error, builder.discard_partial()))),
534        };
535        let stage_five = match self.stage_five_streaming(
536            &parsed.manifest,
537            &release_dir,
538            &mut builder,
539            &mut budget,
540            &verifier_limits,
541        ) {
542            Ok(value) => value,
543            Err(error) => return Err(Box::new(with_cleanup(error, builder.discard_partial()))),
544        };
545        if let Err(error) = builder.sync_directory() {
546            return Err(Box::new(with_cleanup(error, builder.discard_partial())));
547        }
548        let snapshot = builder.seal();
549        self.note_stage(5);
550        drop(release_dir);
551        drop(root);
552
553        match self.stages_six_to_twelve(
554            &parsed.manifest,
555            &parsed.sidecar,
556            &parsed.manifest_digest,
557            &stage_five,
558            &snapshot,
559            request,
560        ) {
561            Ok(receipt) => Ok(receipt),
562            Err(error) => Err(Box::new(with_cleanup(error, snapshot.discard()))),
563        }
564    }

595            )));
596        }
597        let (manifest_file, manifest_facts) = no_cleanup(open_regular_file(

600            &mut budget,
601        ))?;

608        ))?;

615            &sidecar_file,

619            "max_sidecar_bytes",
620        ))?;
621        drop(manifest_file);
622        drop(sidecar_file);
623
624        let mut builder = no_cleanup(SnapshotBuilder::new(
625            &self.private_staging,
626            &mut budget,
627            &self.policy,
628        ))?;
629        for (name, bytes) in [
630            ("manifest.json", manifest_bytes.as_slice()),
631            ("signature.json", sidecar_bytes.as_slice()),
632        ] {
633            if let Err(error) = builder.add(name, bytes) {
634                return Err(Box::new(with_cleanup(error, builder.discard_partial())));
635            }
636        }
637
638        let parsed = match self.stages_one_to_four(
639            &manifest_bytes,
640            &sidecar_bytes,
641            request,
642            self.policy.max_json_nesting,
643        ) {
644            Ok(value) => value,
645            Err(error) => return Err(Box::new(with_cleanup(error, builder.discard_partial()))),
646        };
647        let stage_five =
648            match self.stage_five(&parsed.manifest, &release_dir, &mut builder, &mut budget) {
649                Ok(value) => value,
650                Err(error) => return Err(Box::new(with_cleanup(error, builder.discard_partial()))),
651            };
652        if let Err(error) = builder.sync_directory() {
653            return Err(Box::new(with_cleanup(error, builder.discard_partial())));
654        }
655        let snapshot = builder.seal();
656        self.note_stage(5);
657        drop(release_dir);
658        drop(root);
659
660        match self.stages_six_to_twelve(
661            &parsed.manifest,
662            &parsed.sidecar,
663            &parsed.manifest_digest,
664            &stage_five,
665            &snapshot,
666            request,
667        ) {
668            Ok(receipt) => Ok(receipt),
669            Err(error) => Err(Box::new(with_cleanup(error, snapshot.discard()))),
670        }
671    }

673    fn stages_one_to_four(
674        &self,
675        manifest_bytes: &[u8],
676        sidecar_bytes: &[u8],
677        request: &AdmissionRequest,
678        max_depth: u32,
679    ) -> Result<ParsedStages, VerifierError> {
680        let manifest_value =
681            parse_strict(manifest_bytes).map_err(|error| VerifierError::ManifestParse {
682                detail: error.to_string(),
683            })?;
684        check_depth(&manifest_value, max_depth)?;
685        let manifest = if self.require_canonical_wire {
686            let parsed = canonical_wire::parse_manifest(&manifest_value)
687                .map_err(canonical_wire::CanonicalError::into_manifest_error)?;
688            *self.canonical_profiles.borrow_mut() = parsed.profiles.clone();
689            *self.canonical_payload_schemas.borrow_mut() = parsed.declared_payload_schemas.clone();
690            canonical_wire::into_internal_manifest(&parsed)?
691        } else {
692            serde_json::from_value(manifest_value.clone()).map_err(classify_decode_error)?
693        };
694        self.check_manifest_schema(&manifest)?;
695
696        let sidecar_value =
697            parse_strict(sidecar_bytes).map_err(|error| VerifierError::SidecarSchema {
698                detail: error.to_string(),
699            })?;
700        check_depth(&sidecar_value, max_depth)?;
701        let sidecar: SignatureSidecar =
702            serde_json::from_value(sidecar_value).map_err(classify_decode_error)?;
703        self.note_stage(1);
704
705        let canonical = canonical_bytes(&manifest_value);
706        let manifest_digest = hex_lower(&sha256(&canonical));
707        let sidecar_digest = parse_hex_lower(&sidecar.manifest_sha256, "manifest_sha256")
708            .map_err(|_| VerifierError::MalformedManifestDigest)?;
709        if sidecar_digest != sha256(&canonical) {
710            return Err(VerifierError::ManifestDigestMismatch);
711        }
712        self.note_stage(2);
713
714        if sidecar.schema_version != 1
715            || sidecar.algorithm != "Ed25519"
716            || sidecar.key_id.is_empty()
717        {
718            return Err(VerifierError::SidecarSchema {
719                detail: "sidecar version, algorithm or key id is invalid".to_owned(),
720            });
721        }
722        let signature_bytes = STANDARD
723            .decode(&sidecar.signature_b64)
724            .map_err(|_| VerifierError::MalformedSignature)?;
725        if signature_bytes.len() != 64 {
726            return Err(VerifierError::MalformedSignature);
727        }
728        let signature = Signature::from_slice(&signature_bytes)
729            .map_err(|_| VerifierError::MalformedSignature)?;
730
731        let entry = self.registry.lookup(
732            &manifest.publication_identity.publisher_identity,
733            &sidecar.key_id,
734        )?;
735        let verifying_key = VerifyingKey::from_bytes(&entry.public_key_bytes)
736            .map_err(|_| VerifierError::KeyAlgorithm)?;
737        let mut message = Vec::with_capacity(SIGNING_DOMAIN.len() + 32);
738        message.extend_from_slice(SIGNING_DOMAIN);
739        message.extend_from_slice(&sidecar_digest);
740        verifying_key
741            .verify(&message, &signature)
742            .map_err(|_| VerifierError::SignatureInvalid)?;
743        self.note_stage(3);
744
745        self.registry
746            .check_policy(entry, manifest.release_identity.as_of_ns)?;
747        if sidecar.key_id != manifest.publication_identity.signing_key_id {
748            return Err(VerifierError::KeyBinding);
749        }
750        if request.publisher_identity != manifest.publication_identity.publisher_identity {
751            return Err(VerifierError::KeyBinding);
752        }
753        self.note_stage(4);
754        Ok(ParsedStages {
755            manifest,
756            sidecar,
757            manifest_digest,
758        })
759    }

761    fn check_manifest_schema(&self, manifest: &ReleaseManifest) -> Result<(), VerifierError> {
762        let identity = &manifest.release_identity;
763        if manifest.schema_version != 1 {
764            return Err(VerifierError::ManifestSchema {
765                detail: "schema_version must be one".to_owned(),
766            });
767        }
768        if manifest.contract_version != "v2" || manifest.contract_revision.is_empty() {
769            return Err(VerifierError::ManifestSchema {
770                detail: "contract version or revision is invalid".to_owned(),
771            });
772        }
773        if identity.as_of_ns == 0
774            || identity.created_at_utc_ns == 0
775            || identity.release_id.is_empty()
776            || identity.publication_state != "AVAILABLE"
777        {
778            return Err(VerifierError::ManifestSchema {
779                detail: "release identity is incomplete or not AVAILABLE".to_owned(),
780            });
781        }
782        if manifest.digests.digest_algorithm != "sha256"
783            || parse_hex_lower(&manifest.digests.data_sha256, "data_sha256").is_err()
784        {
785            return Err(VerifierError::ManifestSchema {
786                detail: "dataset digest declaration is invalid".to_owned(),
787            });
788        }
789        if manifest.publication_identity.publisher_identity.is_empty()
790            || manifest.publication_identity.signing_key_id.is_empty()
791            || manifest.publication_identity.signature_reference != "signature.json"
792            || manifest.publication_identity.archive_reference.is_empty()
793        {
794            return Err(VerifierError::ManifestSchema {
795                detail: "publication identity is incomplete".to_owned(),
796            });
797        }
798        Ok(())
799    }

816    fn stage_five(
817        &self,
818        manifest: &ReleaseManifest,
819        release_dir: &std::fs::File,
820        builder: &mut SnapshotBuilder,
821        budget: &mut FdBudget,
822    ) -> Result<StageFive, VerifierError> {
823        let payload_registry = if self.require_canonical_wire {
824            Some(self.load_payload_registry()?)
825        } else {
826            None
827        };
828        let descriptors = self.check_artifact_declarations(manifest)?;
829        let artifacts_dir = open_dir_at(release_dir, "artifacts", budget)?;
830        let observed = enumerate_names(&artifacts_dir)?;
831        let components: Vec<String> = descriptors
832            .iter()
833            .map(|descriptor| artifact_component(&descriptor.path))
834            .collect::<Result<_, _>>()?;
835        let expected: HashSet<&str> = components.iter().map(String::as_str).collect();
836        if observed
837            .iter()
838            .map(String::as_str)
839            .collect::<HashSet<&str>>()
840            != expected
841        {
842            return Err(VerifierError::FileSet {
843                detail: "artifact directory file set differs from artifacts[]".to_owned(),
844            });
845        }
846        let mut dataset_hasher = sha2::Sha256::new();
847        let mut all_records = Vec::new();
848        let mut bindings = Vec::new();
849        for descriptor in descriptors {
850            let component = artifact_component(&descriptor.path)?;
851            let (file, facts) = open_regular_file(&artifacts_dir, &component, budget)?;
852            let stored = read_bounded(
853                &file,
854                &facts,
855                descriptor.stored_bytes,
856                self.policy.max_stored_bytes,
857                "max_stored_bytes",
858            )?;
859            let stored_digest = sha256(&stored);
860            if stored_digest != parse_hex_lower(&descriptor.stored_sha256, "stored_sha256")? {
861                return Err(VerifierError::DigestMismatch {
862                    detail: "stored_sha256 differs from opened bytes".to_owned(),
863                });
864            }
865            let expanded = decompress(&stored, descriptor, &self.policy)?;
866            if expanded.len() as u64 != descriptor.uncompressed_bytes {
867                return Err(VerifierError::ByteCount {
868                    detail: "uncompressed_bytes differs from decoded bytes".to_owned(),
869                });
870            }
871            if expanded.len() as u64 > self.policy.max_expanded_bytes {
872                return Err(VerifierError::ResourceLimit {
873                    limit: "max_expanded_bytes".to_owned(),
874                });
875            }
876            let expanded_digest = sha256(&expanded);
877            if expanded_digest
878                != parse_hex_lower(&descriptor.uncompressed_sha256, "uncompressed_sha256")?
879            {
880                return Err(VerifierError::DigestMismatch {
881                    detail: "uncompressed_sha256 differs from decoded bytes".to_owned(),
882                });
883            }
884            dataset_hasher.update(&expanded);
885            let records = if self.require_canonical_wire {
886                verify_canonical_record_stream(
887                    &expanded,
888                    descriptor,
889                    &self.canonical_profiles.borrow(),
890                    &self.canonical_payload_schemas.borrow(),
891                    payload_registry
892                        .as_ref()
893                        .ok_or_else(|| PayloadRegistryError::NotInstalled.into_verifier_error())?,
894                    &self.policy,
895                )?
896            } else {
897                verify_record_stream(&expanded, descriptor, &self.policy)?
898            };
899            drop(file);
900            builder.add(&component, &stored)?;
901            all_records.push(records);
902            bindings.push(ArtifactBinding {
903                path: descriptor.path.clone(),
904                logical_partition_id: descriptor.logical_partition_id.clone(),
905                shard_ordinal: descriptor.shard_ordinal,
906                stored_sha256: descriptor.stored_sha256.clone(),
907                uncompressed_sha256: descriptor.uncompressed_sha256.clone(),
908                stored_bytes: descriptor.stored_bytes,
909                uncompressed_bytes: descriptor.uncompressed_bytes,
910                record_count: descriptor.record_count,
911            });
912        }
913        let mut total = 0_u64;
914        for binding in &bindings {
915            total = total.saturating_add(binding.uncompressed_bytes);
916        }
917        if total > self.policy.max_total_expanded_bytes {
918            return Err(VerifierError::ResourceLimit {
919                limit: "max_total_expanded_bytes".to_owned(),
920            });
921        }
922        let dataset_bytes = dataset_hasher.finalize();
923        let mut raw_digest = [0_u8; 32];
924        raw_digest.copy_from_slice(&dataset_bytes);
925        let data_digest = hex_lower(&raw_digest);
926        if data_digest != manifest.digests.data_sha256 {
927            return Err(VerifierError::DigestMismatch {
928                detail: "data_sha256 differs from ordered uncompressed streams".to_owned(),
929            });
930        }
931        Ok(StageFive {
932            bindings,
933            data_digest,
934            records: all_records,
935            envelope_agrees: true,
936        })
937    }

939    fn stage_five_streaming(
940        &self,
941        manifest: &ReleaseManifest,
942        release_dir: &std::fs::File,
943        builder: &mut SnapshotBuilder,
944        budget: &mut FdBudget,
945        limits: &EffectiveLimits,
946    ) -> Result<StageFive, VerifierError> {
947        let payload_registry = if self.require_canonical_wire {
948            Some(self.load_payload_registry()?)
949        } else {
950            None
951        };
952        let descriptors = self.check_artifact_declarations_streaming(manifest, limits)?;
953        let artifacts_dir = open_dir_at(release_dir, "artifacts", budget)?;
954        let observed = enumerate_names(&artifacts_dir)?;
955        let components: Vec<String> = descriptors
956            .iter()
957            .map(|descriptor| artifact_component(&descriptor.path))
958            .collect::<Result<_, _>>()?;
959        let expected: HashSet<&str> = components.iter().map(String::as_str).collect();
960        if observed
961            .iter()
962            .map(String::as_str)
963            .collect::<HashSet<&str>>()
964            != expected
965        {
966            return Err(VerifierError::FileSet {
967                detail: "artifact directory file set differs from artifacts[]".to_owned(),
968            });
969        }
970        let mut dataset_hasher = Sha256::new();
971        let mut bindings = Vec::new();
972        let mut total_expanded = 0_u64;
973        let mut total_stored = 0_u64;
974        let mut total_records = 0_u64;
975        let mut in_flight = InFlightArtifacts::new(limits.max_concurrent_artifacts());
976        let started = std::time::Instant::now();
977        for (descriptor, component) in descriptors.iter().zip(components) {
978            if started.elapsed().as_millis() > u128::from(limits.max_processing_duration_ms()) {
979                return Err(VerifierError::ResourceLimit {
980                    limit: "max_processing_duration_ms".to_owned(),
981                });
982            }
983            let (file, facts) = open_regular_file(&artifacts_dir, &component, budget)?;
984            let profiles = self.canonical_profiles.borrow();
985            let declared_schemas = self.canonical_payload_schemas.borrow();
986            let mode = if self.require_canonical_wire {
987                RecordMode::Canonical {
988                    profiles: &profiles,
989                    declared_schemas: &declared_schemas,
990                    registry: payload_registry
991                        .as_ref()
992                        .ok_or_else(|| PayloadRegistryError::NotInstalled.into_verifier_error())?,
993                }
994            } else {
995                RecordMode::Internal
996            };
997            assert_eq!(
998                in_flight.in_flight(),
999                0,
1000                "no slot may remain in flight between artifacts"
1001            );
1002            in_flight = in_flight.try_acquire()?;
1003            assert_eq!(
1004                in_flight.in_flight(),
1005                1,
1006                "the delivered pipeline verifies one artifact at a time"
1007            );
1008            let state = streaming::ArtifactStream {
1009                file: &file,
1010                facts: &facts,
1011                descriptor,
1012                component: &component,
1013                limits,
1014                dataset_hasher: &mut dataset_hasher,
1015                builder,
1016                mode,
1017                expected_envelope: &manifest.schema_versions.envelope_version,
1018            }
1019            .verify();
1020            in_flight = in_flight.try_release()?;
1021            let state = state?;
1022            drop(file);
1023            total_expanded = total_expanded
1024                .checked_add(state.expanded_bytes())
1025                .ok_or(VerifierError::RecordCount)?;
1026            total_stored =
1027                total_stored
1028                    .checked_add(state.stored_bytes())
1029                    .ok_or(VerifierError::ByteCount {
1030                        detail: "release stored byte total overflowed".to_owned(),
1031                    })?;
1032            total_records = total_records
1033                .checked_add(state.record_count())
1034                .ok_or(VerifierError::RecordCount)?;
1035            if total_expanded > limits.max_expanded_bytes_per_release() {
1036                return Err(VerifierError::ResourceLimit {
1037                    limit: "max_expanded_bytes_per_release".to_owned(),
1038                });
1039            }
1040            if total_stored > limits.max_stored_bytes_per_release() {
1041                return Err(VerifierError::ResourceLimit {
1042                    limit: "max_stored_bytes_per_release".to_owned(),
1043                });
1044            }
1045            if total_records > limits.max_records_per_release() {
1046                return Err(VerifierError::ResourceLimit {
1047                    limit: "max_records_per_release".to_owned(),
1048                });
1049            }
1050            bindings.push(ArtifactBinding {
1051                path: descriptor.path.clone(),
1052                logical_partition_id: descriptor.logical_partition_id.clone(),
1053                shard_ordinal: descriptor.shard_ordinal,
1054                stored_sha256: descriptor.stored_sha256.clone(),
1055                uncompressed_sha256: descriptor.uncompressed_sha256.clone(),
1056                stored_bytes: descriptor.stored_bytes,
1057                uncompressed_bytes: descriptor.uncompressed_bytes,
1058                record_count: descriptor.record_count,
1059            });
1060        }
1061        assert_eq!(
1062            in_flight.in_flight(),
1063            0,
1064            "every acquired slot is released before stage five completes"
1065        );
1066        let dataset_bytes = dataset_hasher.finalize();
1067        let mut raw_digest = [0_u8; 32];
1068        raw_digest.copy_from_slice(&dataset_bytes);
1069        let data_digest = hex_lower(&raw_digest);
1070        if data_digest != manifest.digests.data_sha256 {
1071            return Err(VerifierError::DigestMismatch {
1072                detail: "data_sha256 differs from ordered uncompressed streams".to_owned(),
1073            });
1074        }
1075        Ok(StageFive {
1076            bindings,
1077            data_digest,
1078            records: Vec::new(),
1079            envelope_agrees: true,
1080        })
1081    }

1101    fn check_artifact_declarations_streaming<'a>(
1102        &self,
1103        manifest: &'a ReleaseManifest,
1104        limits: &EffectiveLimits,
1105    ) -> Result<Vec<&'a ArtifactDescriptor>, VerifierError> {
1106        Self::check_artifact_count(manifest.artifacts.len(), limits.max_artifacts())?;
1107        let mut descriptors: Vec<&ArtifactDescriptor> = manifest.artifacts.iter().collect();
1108        descriptors.sort_by(|left, right| {
1109            left.logical_partition_id
1110                .cmp(&right.logical_partition_id)
1111                .then(left.shard_ordinal.cmp(&right.shard_ordinal))
1112        });
1113        let mut paths = HashSet::new();
1114        let mut identities = HashSet::new();
1115        let mut previous: Option<(&str, u32)> = None;
1116        for descriptor in &descriptors {
1117            artifact_component(&descriptor.path)?;
1118            if !paths.insert(&descriptor.path) {
1119                return Err(VerifierError::FileSet {
1120                    detail: "duplicate normalized artifact path".to_owned(),
1121                });
1122            }
1123            let identity = (
1124                descriptor.logical_partition_id.as_str(),
1125                descriptor.shard_ordinal,
1126            );
1127            if !identities.insert(identity) {
1128                return Err(VerifierError::FileSet {
1129                    detail: "duplicate logical partition and shard ordinal".to_owned(),
1130                });
1131            }
1132            if let Some(earlier) = previous
1133                && earlier >= (identity.0, identity.1)
1134            {
1135                return Err(VerifierError::ManifestSchema {
1136                    detail: "artifacts are not in canonical ascending order".to_owned(),
1137                });
1138            }
1139            previous = Some(identity);
1140            if descriptor.encoding != RECORD_ENCODING
1141                || !matches!(
1142                    descriptor.record_order_profile.as_str(),
1143                    SUPERSEDES_STRICT | SUPERSEDES_GAPS | REFERENCE_KEY
1144                )
1145            {
1146                return Err(VerifierError::ManifestSchema {
1147                    detail: "artifact encoding or record order profile is unknown".to_owned(),
1148                });
1149            }
1150            let compression_ok = match descriptor.compression.as_str() {
1151                "none" | "gzip" => descriptor.compression_version == 1,
1152                _ => false,
1153            };
1154            if !compression_ok {
1155                return Err(VerifierError::Decompression {
1156                    detail: "compression declaration is unknown".to_owned(),
1157                });
1158            }
1159            if descriptor.stored_bytes == 0
1160                || descriptor.stored_bytes > limits.max_stored_bytes_per_artifact()
1161                || descriptor.uncompressed_bytes == 0
1162                || descriptor.uncompressed_bytes > limits.max_expanded_bytes_per_artifact()
1163                || descriptor.record_count == 0
1164                || descriptor.record_count > limits.max_records_per_artifact()
1165            {
1166                return Err(VerifierError::ResourceLimit {
1167                    limit: "artifact declared size or record count exceeds the profile".to_owned(),
1168                });
1169            }
1170            parse_hex_lower(&descriptor.stored_sha256, "stored_sha256")?;
1171            parse_hex_lower(&descriptor.uncompressed_sha256, "uncompressed_sha256")?;
1172        }
1173        Ok(descriptors)
1174    }

1256    fn stages_six_to_twelve(
1257        &self,
1258        manifest: &ReleaseManifest,
1259        sidecar: &SignatureSidecar,
1260        manifest_digest: &str,
1261        stage_five: &StageFive,
1262        snapshot: &PrivateSnapshot,
1263        request: &AdmissionRequest,
1264    ) -> Result<AdmissionReceipt, VerifierError> {
1265        verify_snapshot_binding(snapshot, stage_five)?;
1266        let rights = self.rights.evaluate(manifest, request);
1267        if rights.decision != Decision::Permit {
1268            return Err(VerifierError::RightsNotPermitted);
1269        }
1270        if rights.matching_grant_ids.len() != 1 {
1271            return Err(VerifierError::AmbiguousGrants);
1272        }
1273        self.note_stage(6);
1274        self.stage_seven(manifest, request, stage_five)?;
1275        self.note_stage(7);
1276        self.stage_eight(manifest)?;
1277        self.note_stage(8);
1278        self.stage_nine(manifest, stage_five)?;
1279        self.note_stage(9);
1280        self.stage_ten(manifest, request)?;
1281        self.note_stage(10);
1282        self.stage_eleven(manifest, sidecar, manifest_digest, stage_five, request)?;
1283        self.note_stage(11);
1284        let receipt = AdmissionReceipt::new(crate::receipt::ReceiptInput {
1285            release_id: manifest.release_identity.release_id.clone(),
1286            manifest_sha256: manifest_digest.to_owned(),
1287            data_sha256: stage_five.data_digest.clone(),
1288            publisher_id: manifest.publication_identity.publisher_identity.clone(),

1298            consumer_use_profile: request.use_profile.clone(),
1299            verifier_commit: self.verifier_commit.clone(),
1300        })?;
1301        self.note_stage(12);
1302        Ok(receipt)
1303    }

1305    fn stage_seven(
1306        &self,
1307        manifest: &ReleaseManifest,
1308        request: &AdmissionRequest,
1309        stage_five: &StageFive,
1310    ) -> Result<(), VerifierError> {
1311        let coverage = self.coverage.evaluate(manifest, request);
1312        if coverage.used_implicit_all {
1313            return Err(VerifierError::Coverage {
1314                detail: "coverage evaluator relied on implicit ALL".to_owned(),
1315            });
1316        }
1317        let declared: HashSet<&str> = manifest
1318            .coverage
1319            .dimensions
1320            .iter()
1321            .map(|dimension| dimension.dimension_id.as_str())
1322            .collect();
1323        for required in COVERAGE_DIMENSIONS {
1324            if !declared.contains(required) {
1325                return Err(VerifierError::Coverage {
1326                    detail: format!("coverage dimension {} is absent", required),
1327                });
1328            }
1329        }
1330        for dimension in &manifest.coverage.dimensions {
1331            let missing_denominator = match dimension.denominator {
1332                Some(value) => value == 0,
1333                None => true,
1334            };
1335            if missing_denominator || dimension.numerator > dimension.denominator.unwrap_or(0) {
1336                return Err(VerifierError::Coverage {
1337                    detail: format!(
1338                        "dimension {} has no usable denominator",
1339                        dimension.dimension_id
1340                    ),
1341                });
1342            }
1343            if dimension.denominator_identity.is_empty()
1344                || dimension.universe_name.is_empty()
1345                || dimension.evidence_reference.is_empty()
1346                || dimension.as_of_ns != manifest.release_identity.as_of_ns
1347            {
1348                return Err(VerifierError::Coverage {
1349                    detail: format!("dimension {} is incomplete", dimension.dimension_id),
1350                });
1351            }
1352            if dimension.verdict == "PASS"
1353                && !coverage
1354                    .approved_dimensions
1355                    .contains(&dimension.dimension_id)
1356            {
1357                return Err(VerifierError::Coverage {
1358                    detail: format!(
1359                        "dimension {} PASS is not established",
1360                        dimension.dimension_id
1361                    ),
1362                });
1363            }
1364        }
1365        if stage_five.bindings.is_empty() {
1366            return Err(VerifierError::Coverage {
1367                detail: "no verified artifact supports coverage".to_owned(),
1368            });
1369        }
1370        Ok(())
1371    }

1437    fn stage_ten(
1438        &self,
1439        manifest: &ReleaseManifest,
1440        request: &AdmissionRequest,
1441    ) -> Result<(), VerifierError> {
1442        let lineage = &manifest.lineage;
1443        if lineage.raw_archive_reference.is_empty()
1444            || lineage.transformation_record.is_empty()
1445            || lineage.was_derived_from.is_empty()
1446            || lineage.had_primary_source.is_empty()
1447            || lineage.knowledge_time_ns == 0
1448            || lineage.revision == 0
1449        {
1450            return Err(VerifierError::Lineage {
1451                detail: "source lineage declaration is incomplete".to_owned(),
1452            });
1453        }
1454        parse_hex_lower(&lineage.raw_archive_sha256, "raw_archive_sha256")?;
1455        if lineage.knowledge_time_ns > manifest.release_identity.as_of_ns {
1456            return Err(VerifierError::PointInTime {
1457                detail: "knowledge time is after release as_of_ns".to_owned(),
1458            });
1459        }
1460        if request.as_of_ns < lineage.knowledge_time_ns
1461            || request.as_of_ns > manifest.release_identity.as_of_ns
1462        {
1463            return Err(VerifierError::PointInTime {
1464                detail: "request as_of_ns is outside the knowledge basis".to_owned(),
1465            });
1466        }
1467        let supersedes = match &lineage.supersedes {
1468            Some(value) => value,
1469            None => {
1470                if lineage.invalidated_at_time_ns != 0
1471                    && lineage.invalidated_at_time_ns < lineage.knowledge_time_ns
1472                {
1473                    return Err(VerifierError::PointInTime {
1474                        detail: "invalidation precedes knowledge".to_owned(),
1475                    });
1476                }
1477                return Ok(());
1478            }
1479        };
1480        if supersedes.supersedes_release_id == manifest.release_identity.release_id
1481            || supersedes.supersedes_release_id.is_empty()
1482            || supersedes.revision == 0
1483            || supersedes.revision >= lineage.revision
1484            || supersedes.correction_reason.is_empty()
1485        {
1486            return Err(VerifierError::Lineage {
1487                detail: "supersedes relation is self-referential or regresses revision".to_owned(),
1488            });
1489        }
1490        if supersedes.knowledge_time_ns == 0
1491            || supersedes.knowledge_time_ns > lineage.knowledge_time_ns
1492            || supersedes.knowledge_time_ns > manifest.release_identity.as_of_ns
1493            || supersedes.invalidated_at_time_ns < supersedes.knowledge_time_ns
1494        {
1495            return Err(VerifierError::PointInTime {
1496                detail: "correction knowledge or invalidation time is inconsistent".to_owned(),
1497            });
1498        }
1499        parse_hex_lower(
1500            &supersedes.supersedes_manifest_sha256,
1501            "supersedes manifest digest",
1502        )?;
1503        parse_hex_lower(
1504            &supersedes.supersedes_data_sha256,
1505            "supersedes dataset digest",
1506        )?;
1507        self.walk_lineage(supersedes, manifest.release_identity.release_id.as_str())
1508    }

1510    fn walk_lineage(
1511        &self,
1512        relation: &crate::schema::SupersedesRelation,
1513        current_release_id: &str,
1514    ) -> Result<(), VerifierError> {
1515        let mut visited = HashSet::new();
1516        visited.insert(current_release_id.to_owned());
1517        let mut next = Some(relation.supersedes_release_id.clone());
1518        let mut expected_revision = relation.revision;
1519        let mut expected_manifest = relation.supersedes_manifest_sha256.clone();
1520        let mut expected_data = relation.supersedes_data_sha256.clone();
1521        while let Some(release_id) = next {
1522            if !visited.insert(release_id.clone()) {
1523                return Err(VerifierError::Lineage {
1524                    detail: "correction lineage contains a cycle".to_owned(),
1525                });
1526            }
1527            let evidence =
1528                self.lineage
1529                    .predecessor(&release_id)
1530                    .ok_or_else(|| VerifierError::Lineage {
1531                        detail: "predecessor is absent from the trusted archive".to_owned(),
1532                    })?;
1533            if evidence.release_id != release_id
1534                || evidence.manifest_sha256 != expected_manifest
1535                || evidence.data_sha256 != expected_data
1536                || evidence.revision != expected_revision
1537            {
1538                return Err(VerifierError::Lineage {
1539                    detail: "predecessor evidence disagrees with the declared relation".to_owned(),
1540                });
1541            }
1542            if evidence.knowledge_time_ns > relation.knowledge_time_ns {
1543                return Err(VerifierError::PointInTime {
1544                    detail: "predecessor knowledge time is after the correction".to_owned(),
1545                });
1546            }
1547            next = evidence.supersedes_release_id.clone();
1548            expected_revision = evidence.revision.saturating_sub(1);
1549            expected_manifest = evidence.manifest_sha256;
1550            expected_data = evidence.data_sha256;
1551        }
1552        Ok(())
1553    }

1555    fn stage_eleven(
1556        &self,
1557        manifest: &ReleaseManifest,
1558        sidecar: &SignatureSidecar,
1559        manifest_digest: &str,
1560        stage_five: &StageFive,
1561        request: &AdmissionRequest,
1562    ) -> Result<(), VerifierError> {

1564            || request.manifest_sha256 != manifest_digest
1565            || request.data_sha256 != stage_five.data_digest
1566            || request.publisher_identity != manifest.publication_identity.publisher_identity
1567        {
1568            return Err(VerifierError::UseScope {
1569                detail: "request release identity or digests differ from verified values"
1570                    .to_owned(),
1571            });
1572        }
1573        let required: &[&str] = match request.use_profile.as_str() {
1574            "reference_lookup" => &["read_release"],
1575            "historical_research" | "intraday_replay_research" => &[
1576                "read_release",
1577                "store_permitted_research_copy",
1578                "replay",
1579                "internal_analytics",
1580            ],
1581            "derived_research_dataset" => &[
1582                "read_release",
1583                "store_permitted_research_copy",
1584                "internal_analytics",
1585                "retain_permitted_derived_data",
1586                "create_derived",
1587            ],
1588            "portfolio_simulation" => &["read_release", "replay", "internal_analytics"],
1589            _ => {
1590                return Err(VerifierError::UseScope {
1591                    detail: "use profile is not declared".to_owned(),
1592                });
1593            }
1594        };
1595        let requested: HashSet<&str> = request
1596            .requested_actions
1597            .iter()
1598            .map(String::as_str)
1599            .collect();
1600        let expected: HashSet<&str> = required.iter().copied().collect();
1601        if requested != expected {
1602            return Err(VerifierError::UseScope {
1603                detail: "requested actions are not exactly the profile actions".to_owned(),
1604            });
1605        }
1606        let needed_capability = if request.use_profile == "intraday_replay_research" {
1607            Some("intraday_capability")
1608        } else {
1609            None
1610        };
1611        if let Some(capability) = needed_capability {
1612            let present = request
1613                .declared_capabilities
1614                .iter()
1615                .any(|claim| claim.capability_id == capability);
1616            if !present {
1617                return Err(VerifierError::Capability {
1618                    detail: format!("capability {} is not declared", capability),
1619                });
1620            }
1621        }
1622        if request.purpose != DECLARED_PURPOSE
1623            || request.actor.is_empty()
1624            || request.licensee.is_empty()
1625            || request.recipient_identity.is_empty()
1626        {
1627            return Err(VerifierError::UseScope {
1628                detail: "request purpose or identity does not match the reviewed terms".to_owned(),
1629            });
1630        }
1631        if request.scope.subject_state != "NAMED" || request.scope.named_subjects.is_empty() {
1632            return Err(VerifierError::UseScope {
1633                detail: "runtime request must use a non-empty NAMED subject set".to_owned(),
1634            });
1635        }
1636        match (
1637            request.scope.data_time_from_ns,
1638            request.scope.data_time_to_ns,
1639        ) {
1640            (Some(from), Some(to)) if from <= to => {}
1641            _ => {
1642                return Err(VerifierError::UseScope {
1643                    detail: "request data-time window is unbounded or inverted".to_owned(),
1644                });
1645            }
1646        }
1647        if request.requested_retention.max_duration_ns == 0
1648            || request.requested_retention.purge_rule.is_empty()
1649        {
1650            return Err(VerifierError::Retention);
1651        }
1652        if sidecar.algorithm != "Ed25519" {
1653            return Err(VerifierError::SidecarSchema {
1654                detail: "sidecar algorithm is not Ed25519".to_owned(),
1655            });
1656        }
1657        Ok(())
1658    }

1674fn verify_canonical_record_stream(
1675    bytes: &[u8],
1676    descriptor: &ArtifactDescriptor,
1677    profiles: &HashMap<String, ArtifactRecordOrderProfile>,
1678    declared_schemas: &HashMap<String, DeclaredPayloadSchema>,
1679    registry: &PayloadSchemaRegistry,
1680    policy: &VerifierResourcePolicy,
1681) -> Result<Vec<ReleaseRecord>, VerifierError> {
1682    if bytes.is_empty() || bytes.last() != Some(&b'\n') {
1683        return Err(VerifierError::RecordStream {
1684            detail: "canonical record stream lacks a trailing LF boundary".to_owned(),
1685        });
1686    }
1687    if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) || bytes.contains(&0x0D) {
1688        return Err(VerifierError::RecordStream {
1689            detail: "canonical record stream contains a BOM or CR byte".to_owned(),
1690        });
1691    }
1692    let profile = profiles
1693        .get(&descriptor.path)
1694        .ok_or_else(|| VerifierError::RecordOrder {
1695            detail: "artifact has no canonical order profile".to_owned(),
1696        })?;
1697    if descriptor.encoding != RECORD_ENCODING {
1698        return Err(VerifierError::ManifestSchema {
1699            detail: "canonical record encoding is not registered".to_owned(),
1700        });
1701    }
1702    let mut records = Vec::new();
1703    let mut keys = Vec::new();
1704    for line in bytes[..bytes.len() - 1].split(|byte| *byte == b'\n') {
1705        if line.is_empty() {
1706            return Err(VerifierError::RecordStream {
1707                detail: "canonical record stream contains a blank line".to_owned(),
1708            });
1709        }
1710        if line.len() as u64 > policy.max_record_bytes {
1711            return Err(VerifierError::ResourceLimit {
1712                limit: "max_record_bytes".to_owned(),
1713            });
1714        }
1715        let key = canonical_wire::derive_key_from_bytes(line, profile)
1716            .map_err(canonical_wire::CanonicalError::into_record_error)?;
1717        let value = parse_strict(line).map_err(|error| VerifierError::RecordStream {
1718            detail: error.to_string(),
1719        })?;
1720        check_depth(&value, policy.max_json_nesting)?;
1721        let record = canonical_wire::parse_record(&value)
1722            .map_err(canonical_wire::CanonicalError::into_record_error)?;
1723        let declared = declared_schemas
1724            .get(record.event_type.as_str())
1725            .ok_or_else(|| {
1726                PayloadRegistryError::UnknownEvent {
1727                    event_type: record.event_type.as_str().to_owned(),
1728                }
1729                .into_verifier_error()
1730            })?;
1731        crate::payload_registry::validate_record(
1732            registry,
1733            record.event_type.as_str(),
1734            declared,
1735            &record.payload,
1736        )
1737        .map_err(PayloadRegistryError::into_verifier_error)?;
1738        keys.push(key);
1739        records.push(canonical_wire::into_internal_record(&record));
1740    }
1741    canonical_wire::validate_order_sequence(&keys)
1742        .map_err(canonical_wire::CanonicalError::into_record_error)?;
1743    if records.len() as u64 != descriptor.record_count {
1744        return Err(VerifierError::RecordCount);
1745    }
1746    Ok(records)
1747}

1749fn verify_snapshot_binding(
1750    snapshot: &PrivateSnapshot,
1751    stage_five: &StageFive,
1752) -> Result<(), VerifierError> {
1753    let expected_count = stage_five.bindings.len() + 2;
1754    if snapshot.entries().len() != expected_count {
1755        return Err(VerifierError::SnapshotMismatch);
1756    }
1757    for entry in snapshot.entries() {
1758        if !entry.verified {
1759            return Err(VerifierError::SnapshotMismatch);
1760        }
1761        let proof = snapshot.verify_entry(&entry.name)?;
1762        if proof.stored_sha256 != entry.stored_sha256 || proof.stored_bytes != entry.stored_bytes {
1763            return Err(VerifierError::SnapshotMismatch);
1764        }
1765    }
1766    for binding in &stage_five.bindings {
1767        let component = artifact_component(&binding.path)?;
1768        let entry = snapshot.entry(&component)?;
1769        if hex_lower(entry.digest()) != binding.stored_sha256
1770            || entry.stored_bytes != binding.stored_bytes
1771        {
1772            return Err(VerifierError::SnapshotMismatch);
1773        }
1774    }
1775    Ok(())
1776}

1798fn decompress(
1799    stored: &[u8],
1800    descriptor: &ArtifactDescriptor,
1801    policy: &VerifierResourcePolicy,
1802) -> Result<Vec<u8>, VerifierError> {
1803    if descriptor.compression == "none" {
1804        if stored.len() as u64 > policy.max_decompression_budget {
1805            return Err(VerifierError::ResourceLimit {
1806                limit: "max_decompression_budget".to_owned(),
1807            });
1808        }
1809        return Ok(stored.to_vec());
1810    }
1811    let budget = policy
1812        .max_decompression_budget
1813        .min(policy.max_expanded_bytes)
1814        .min(descriptor.uncompressed_bytes);
1815    let decoder = GzDecoder::new(stored);
1816    let mut expanded = Vec::new();
1817    let read = decoder
1818        .take(budget.saturating_add(1))
1819        .read_to_end(&mut expanded)
1820        .map_err(|error| VerifierError::Decompression {
1821            detail: error.to_string(),
1822        })?;
1823    if read as u64 > budget {
1824        return Err(VerifierError::ResourceLimit {
1825            limit: "max_decompression_budget".to_owned(),
1826        });
1827    }
1828    Ok(expanded)
1829}

1831fn verify_record_stream(
1832    bytes: &[u8],
1833    descriptor: &ArtifactDescriptor,
1834    policy: &VerifierResourcePolicy,
1835) -> Result<Vec<ReleaseRecord>, VerifierError> {
1836    if bytes.is_empty() || bytes.last() != Some(&b'\n') {
1837        return Err(VerifierError::RecordStream {
1838            detail: "record stream lacks a trailing LF boundary".to_owned(),
1839        });
1840    }
1841    if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) || bytes.contains(&0x0D) {
1842        return Err(VerifierError::RecordStream {
1843            detail: "record stream contains a BOM or CR byte".to_owned(),
1844        });
1845    }
1846    let profiles = [SUPERSEDES_STRICT, SUPERSEDES_GAPS, REFERENCE_KEY];
1847    if !profiles.contains(&descriptor.record_order_profile.as_str()) {
1848        return Err(VerifierError::ManifestSchema {
1849            detail: "record order profile is unknown".to_owned(),
1850        });
1851    }
1852    let mut records = Vec::new();
1853    let mut previous_sequence: Option<(String, String, u64)> = None;
1854    let mut previous_reference: Option<(String, u64, u64, String)> = None;
1855    for line in bytes[..bytes.len() - 1].split(|byte| *byte == b'\n') {
1856        if line.is_empty() {
1857            return Err(VerifierError::RecordStream {
1858                detail: "record stream contains a blank line".to_owned(),
1859            });
1860        }
1861        if line.len() as u64 > policy.max_record_bytes {
1862            return Err(VerifierError::ResourceLimit {
1863                limit: "max_record_bytes".to_owned(),
1864            });
1865        }
1866        let text = std::str::from_utf8(line).map_err(|_| VerifierError::RecordStream {
1867            detail: "record line is not UTF-8".to_owned(),
1868        })?;
1869        let value = parse_strict(line).map_err(|_| VerifierError::RecordStream {
1870            detail: "record line is not a strict JSON object".to_owned(),
1871        })?;
1872        check_depth(&value, policy.max_json_nesting)?;
1873        if canonical_bytes(&value) != line {
1874            return Err(VerifierError::RecordStream {
1875                detail: "record line is not RFC 8785 canonical bytes".to_owned(),
1876            });
1877        }
1878        let record: ReleaseRecord =
1879            serde_json::from_str(text).map_err(|_| VerifierError::RecordStream {
1880                detail: "record line does not match the record schema".to_owned(),
1881            })?;
1882        check_record_order(
1883            &record,
1884            descriptor.record_order_profile.as_str(),
1885            &mut previous_sequence,
1886            &mut previous_reference,
1887        )?;
1888        records.push(record);
1889    }
1890    if records.len() as u64 != descriptor.record_count {
1891        return Err(VerifierError::RecordCount);
1892    }
1893    Ok(records)
1894}
1895
1896fn check_record_order(
1897    record: &ReleaseRecord,
1898    profile: &str,
1899    previous_sequence: &mut Option<(String, String, u64)>,
1900    previous_reference: &mut Option<(String, u64, u64, String)>,
1901) -> Result<(), VerifierError> {
1902    if profile == REFERENCE_KEY {
1903        let key = (
1904            record.object_id.clone(),
1905            record.revision,
1906            record.knowledge_time_ns,
1907            record.assertion_id.clone(),
1908        );
1909        if let Some(earlier) = previous_reference
1910            && *earlier >= key
1911        {
1912            return Err(VerifierError::RecordOrder {
1913                detail: "reference key ordering violation or duplicate position".to_owned(),
1914            });
1915        }
1916        *previous_reference = Some(key);
1917        return Ok(());
1918    }
1919    if record.sequence_domain_id.is_empty() || record.sequence_epoch.is_empty() {
1920        return Err(VerifierError::RecordOrder {
1921            detail: "sequence domain or epoch is empty".to_owned(),
1922        });
1923    }
1924    let key = (
1925        record.sequence_domain_id.clone(),
1926        record.sequence_epoch.clone(),
1927        record.sequence_number,
1928    );
1929    if let Some(earlier) = previous_sequence {
1930        if earlier.0 == key.0 && earlier.1 == key.1 {
1931            if earlier.2 == key.2 {
1932                return Err(VerifierError::RecordOrder {
1933                    detail: "duplicate sequence position".to_owned(),
1934                });
1935            }
1936            if profile == SUPERSEDES_STRICT && key.2 != earlier.2 + 1 {
1937                return Err(VerifierError::RecordOrder {
1938                    detail: "sequence gap under a strict gap policy".to_owned(),
1939                });
1940            }
1941        } else if *earlier >= key {
1942            return Err(VerifierError::RecordOrder {
1943                detail: "records are not in ascending sequence-domain order".to_owned(),
1944            });
1945        }
1946    }
1947    *previous_sequence = Some(key);
1948    Ok(())
1949}
