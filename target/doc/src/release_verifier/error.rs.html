2//! Typed refusals for the independent NDC release verifier.
3//!
4//! One variant covers one failure class. The `code` method maps a variant to the
5//! stable D2 rejection code reported in an admission decision.
6
7use std::path::PathBuf;
8
9/// A private snapshot cleanup that did not reach an empty, synced directory.
10///
11/// The staging path is retained verbatim so an operator can identify the
12/// directory that still needs attention. The entry names are the tracked names
13/// cleanup had not confirmed as absent or removed when it stopped.
14#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
15#[error("CLEANUP_PENDING: private snapshot cleanup incomplete at {staging_path:?}: {detail}")]
16pub struct CleanupPending {
17    /// Exact caller-provided private staging path.
18    pub staging_path: PathBuf,
19    /// Tracked entry names not confirmed absent or removed.
20    pub remaining_entries: Vec<String>,
21    /// Operation that left cleanup incomplete, without untrusted file bytes.
22    pub detail: String,
23}
24
25/// A refusal raised by one of the twelve verification stages.
26///
27/// The variant identifies the first failing stage and failure class. No variant
28/// carries a recovery suggestion: a rejected release is not retried under the
29/// same identity.
30#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
31pub enum VerifierError {
32    /// The manifest bytes are not a single strict JSON object.
33    #[error("REJECT_SCHEMA_ENVELOPE: stage 1 manifest parse: {detail}")]
34    ManifestParse {
35        /// Parser detail without raw release bytes.
36        detail: String,
37    },
38    /// A JSON object in the manifest or sidecar repeats a member name.
39    #[error("REJECT_SCHEMA_ENVELOPE: stage 1 duplicate JSON member: {member}")]
40    DuplicateMember {
41        /// Repeating member name.
42        member: String,
43    },
44    /// A manifest or sidecar object carries a member outside its schema.
45    #[error("REJECT_SCHEMA_ENVELOPE: stage 1 unknown member: {member}")]
46    UnknownMember {
47        /// Unknown member name.
48        member: String,
49    },
50    /// A required manifest member is absent or has the wrong value.
51    #[error("REJECT_SCHEMA_ENVELOPE: stage 1 invalid manifest: {detail}")]
52    ManifestSchema {
53        /// Field-level detail.
54        detail: String,
55    },
56    /// Recomputed RFC 8785 bytes do not match the sidecar digest.
57    #[error("REJECT_SIGNATURE_INTEGRITY: stage 2 manifest digest mismatch")]
58    ManifestDigestMismatch,
59    /// The sidecar digest is not a lowercase 64-hex SHA-256 value.
60    #[error("REJECT_SIGNATURE_INTEGRITY: stage 2 malformed manifest digest")]
61    MalformedManifestDigest,
62    /// The sidecar does not match its exact five-member schema.
63    #[error("REJECT_SIGNATURE_INTEGRITY: stage 3 sidecar schema: {detail}")]
64    SidecarSchema {
65        /// Field-level detail.
66        detail: String,
67    },
68    /// The sidecar signature is not standard padded base64 of 64 bytes.
69    #[error("REJECT_SIGNATURE_INTEGRITY: stage 3 malformed signature encoding")]
70    MalformedSignature,
71    /// The detached Ed25519 signature does not verify over the domain digest.
72    #[error("REJECT_SIGNATURE_INTEGRITY: stage 3 signature verification failed")]
73    SignatureInvalid,
74    /// The publisher key is absent from the out-of-band registry.
75    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 unknown publisher key")]
76    UnknownKey,
77    /// The publisher key is not in a state that permits verification.
78    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 publisher key state {state} is not active")]
79    KeyState {
80        /// Registry state text.
81        state: String,
82    },
83    /// The sidecar key id, manifest signing key id and registry entry disagree.
84    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 publisher key binding mismatch")]
85    KeyBinding,
86    /// The registry entry declares an algorithm other than Ed25519.
87    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 key algorithm mismatch")]
88    KeyAlgorithm,
89    /// The registry entry declares a purpose other than release signing.
90    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 key purpose mismatch")]
91    KeyPurpose,
92    /// The publisher-supplied validity interval does not cover the release time.
93    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 key validity does not cover as_of_ns")]
94    KeyValidity,
95    /// The trusted registry itself is malformed.
96    #[error("REJECT_PUBLISHER_IDENTITY: stage 4 registry invalid: {detail}")]
97    RegistryInvalid {
98        /// Registry-level detail.
99        detail: String,
100    },
101    /// A declared artifact path violates the path-safety rules.
102    #[error("REJECT_TRUSTED_PATH: stage 5 unsafe artifact path: {detail}")]
103    UnsafePath {
104        /// Path-level detail.
105        detail: String,
106    },
107    /// A path component is a symlink or resolves outside the release directory.
108    #[error("REJECT_TRUSTED_PATH: stage 5 symlink component: {detail}")]
109    SymlinkEscape {
110        /// Component detail.
111        detail: String,
112    },
113    /// A published file has more than one hard link.
114    #[error("REJECT_TRUSTED_PATH: stage 5 hard-linked published file: {detail}")]
115    HardLink {
116        /// File detail.
117        detail: String,
118    },
119    /// A required path names a directory, socket, device or other special file.
120    #[error("REJECT_TRUSTED_PATH: stage 5 special file where regular file required")]
121    SpecialFile,
122    /// The published directory file set differs from the declared set.
123    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 file set mismatch: {detail}")]
124    FileSet {
125        /// Missing, extra or duplicate entry detail.
126        detail: String,
127    },
128    /// Actual bytes, declared bytes and post-read metadata disagree.
129    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 byte-count mismatch: {detail}")]
130    ByteCount {
131        /// Size detail.
132        detail: String,
133    },
134    /// A stored, uncompressed or dataset digest differs from the manifest.
135    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 digest mismatch: {detail}")]
136    DigestMismatch {
137        /// Digest family detail.
138        detail: String,
139    },
140    /// A declared digest is not lowercase 64-hex.
141    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 malformed digest: {detail}")]
142    MalformedDigest {
143        /// Field detail.
144        detail: String,
145    },
146    /// A record stream is not canonical UTF-8 JSONL with LF boundaries.
147    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 non-canonical record stream: {detail}")]
148    RecordStream {
149        /// Record-level detail.
150        detail: String,
151    },
152    /// Declared record count differs from observed boundaries.
153    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 record count mismatch")]
154    RecordCount,
155    /// Records are reordered, duplicated or contain a disallowed gap.
156    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 record order policy violation: {detail}")]
157    RecordOrder {
158        /// Ordering detail.
159        detail: String,
160    },
161    /// Decompression fails or the declared compression profile is unknown.
162    #[error("REJECT_ARTIFACT_DIGESTS: stage 5 decompression failure: {detail}")]
163    Decompression {
164        /// Codec detail.
165        detail: String,
166    },
167    /// A declared or actual quantity exceeds the effective resource profile.
168    #[error("REJECT_RESOURCE_LIMIT: verifier resource policy exceeded: {limit}")]
169    ResourceLimit {
170        /// Effective limit field name.
171        limit: String,
172    },
173    /// Profile selection, digest, role or ceiling validation refused the run.
174    #[error("REJECT_RESOURCE_LIMIT: stage 5 resource profile refused: {detail}")]
175    ResourceProfile {
176        /// Profile-level detail without raw limits or caller declarations.
177        detail: String,
178    },
179    /// The trusted local registry approved no profile for the required role.
180    #[error("REJECT_RESOURCE_LIMIT: no approved resource profile for role {role}")]
181    NoApprovedProfile {
182        /// Role with no approved binding.
183        role: String,
184    },
185    /// A record envelope version differs from the manifest at detection.
186    #[error("REJECT_SCHEMA_ENVELOPE: stage 5 record envelope version mismatch: {detail}")]
187    EnvelopeVersion {
188        /// Envelope-level detail.
189        detail: String,
190    },
191    /// A record event type missed the ratified payload schema registry.
192    #[error("REJECT_PAYLOAD_REGISTRY: stage 5 payload registry miss: {detail}")]
193    PayloadRegistry {
194        /// Registry-level detail without raw record bytes.
195        detail: String,
196    },
197    /// The private snapshot copy does not reproduce the opened descriptor bytes.
198    #[error("REJECT_TRUSTED_PATH: stage 5 private snapshot verification failed")]
199    SnapshotMismatch,
200    /// A filesystem operation failed.
201    #[error("REJECT_IO: trusted filesystem operation failed: {detail}")]
202    Io {
203        /// Operation and error text without untrusted file bytes.
204        detail: String,
205    },
206    /// Private snapshot cleanup did not prove an empty, synced directory.
207    #[error(transparent)]
208    CleanupPending(#[from] CleanupPending),
209    /// Injected rights evaluation returned UNKNOWN or DENY.
210    #[error("REJECT_RIGHTS_EVIDENCE: stage 6 rights decision is not PERMIT")]
211    RightsNotPermitted,
212    /// Coverage is absent, lacks a denominator or relies on implicit ALL.
213    #[error("REJECT_COVERAGE: stage 7 coverage decision fails closed: {detail}")]
214    Coverage {
215        /// Coverage detail.
216        detail: String,
217    },
218    /// Producer activity evidence references are incomplete or inconsistent.
219    #[error("REJECT_RIGHTS_EVIDENCE: stage 8 producer activity evidence missing: {detail}")]
220    ProducerActivity {
221        /// Evidence detail.
222        detail: String,
223    },
224    /// A declared envelope or payload schema version is missing or unknown.
225    #[error("REJECT_SCHEMA_ENVELOPE: stage 9 schema version violation: {detail}")]
226    Envelope {
227        /// Envelope detail.
228        detail: String,
229    },
230    /// Correction lineage has a cycle, self-reference or missing predecessor.
231    #[error("REJECT_SOURCE_LINEAGE: stage 10 correction lineage invalid: {detail}")]
232    Lineage {
233        /// Lineage detail.
234        detail: String,
235    },
236    /// Knowledge time is after the as-of instant or invalidates before knowledge.
237    #[error("REJECT_PIT_CONSISTENCY: stage 10 point-in-time violation: {detail}")]
238    PointInTime {
239        /// Time detail.
240        detail: String,
241    },
242    /// A requested consumer action is not covered by the declared profile.
243    #[error("REJECT_USE_SCOPE: stage 11 consumer use violation: {detail}")]
244    UseScope {
245        /// Use detail.
246        detail: String,
247    },
248    /// A requested capability is absent from the admission declaration.
249    #[error("REJECT_USE_SCOPE: stage 11 declared capability missing: {detail}")]
250    Capability {
251        /// Capability detail.
252        detail: String,
253    },
254    /// Retention is unbounded or lacks an explicit purge rule.
255    #[error("REJECT_RETENTION: stage 11 retention declaration fails closed")]
256    Retention,
257    /// More than one grant matches a requested action.
258    #[error("REJECT_AMBIGUOUS_GRANTS: stage 11 non-unique matching grant")]
259    AmbiguousGrants,
260    /// Receipt construction could not bind every required field.
261    #[error("REJECT_RECEIPT: stage 12 receipt binding failed: {detail}")]
262    Receipt {
263        /// Receipt detail.
264        detail: String,
265    },
266}
267
268impl VerifierError {
269    /// Return the stable rejection code for this refusal class.
270    pub fn code(&self) -> &'static str {
271        match self {
272            Self::ManifestParse { .. }
273            | Self::DuplicateMember { .. }
274            | Self::UnknownMember { .. }
275            | Self::ManifestSchema { .. }
276            | Self::Envelope { .. }
277            | Self::EnvelopeVersion { .. } => "REJECT_SCHEMA_ENVELOPE",
278            Self::PayloadRegistry { .. } => "REJECT_PAYLOAD_REGISTRY",
279            Self::ManifestDigestMismatch
280            | Self::MalformedManifestDigest
281            | Self::SidecarSchema { .. }
282            | Self::MalformedSignature
283            | Self::SignatureInvalid => "REJECT_SIGNATURE_INTEGRITY",
284            Self::UnknownKey
285            | Self::KeyState { .. }
286            | Self::KeyBinding
287            | Self::KeyAlgorithm
288            | Self::KeyPurpose
289            | Self::KeyValidity
290            | Self::RegistryInvalid { .. } => "REJECT_PUBLISHER_IDENTITY",
291            Self::UnsafePath { .. }
292            | Self::SymlinkEscape { .. }
293            | Self::HardLink { .. }
294            | Self::SpecialFile
295            | Self::SnapshotMismatch => "REJECT_TRUSTED_PATH",
296            Self::FileSet { .. }
297            | Self::ByteCount { .. }
298            | Self::DigestMismatch { .. }
299            | Self::MalformedDigest { .. }
300            | Self::RecordStream { .. }
301            | Self::RecordCount
302            | Self::RecordOrder { .. }
303            | Self::Decompression { .. } => "REJECT_ARTIFACT_DIGESTS",
304            Self::ResourceLimit { .. }
305            | Self::ResourceProfile { .. }
306            | Self::NoApprovedProfile { .. } => "REJECT_RESOURCE_LIMIT",
307            Self::Io { .. } => "REJECT_IO",
308            Self::CleanupPending(..) => "CLEANUP_PENDING",
309            Self::RightsNotPermitted | Self::ProducerActivity { .. } => "REJECT_RIGHTS_EVIDENCE",
310            Self::Coverage { .. } => "REJECT_COVERAGE",
311            Self::Lineage { .. } => "REJECT_SOURCE_LINEAGE",
312            Self::PointInTime { .. } => "REJECT_PIT_CONSISTENCY",
313            Self::UseScope { .. } | Self::Capability { .. } => "REJECT_USE_SCOPE",
314            Self::Retention => "REJECT_RETENTION",
315            Self::AmbiguousGrants => "REJECT_AMBIGUOUS_GRANTS",
316            Self::Receipt { .. } => "REJECT_RECEIPT",
317        }
318    }
319
320    /// Return the one-based stage number that raised this refusal.
321    ///
322    /// Cross-cutting policy and filesystem refusals report the stage that first
323    /// encountered them.
324    pub fn stage(&self) -> u8 {
325        match self {
326            Self::ManifestParse { .. }
327            | Self::DuplicateMember { .. }
328            | Self::UnknownMember { .. }
329            | Self::ManifestSchema { .. } => 1,
330            Self::ManifestDigestMismatch | Self::MalformedManifestDigest => 2,
331            Self::SidecarSchema { .. } | Self::MalformedSignature | Self::SignatureInvalid => 3,
332            Self::UnknownKey
333            | Self::KeyState { .. }
334            | Self::KeyBinding
335            | Self::KeyAlgorithm
336            | Self::KeyPurpose
337            | Self::KeyValidity
338            | Self::RegistryInvalid { .. } => 4,
339            Self::UnsafePath { .. }
340            | Self::SymlinkEscape { .. }
341            | Self::HardLink { .. }
342            | Self::SpecialFile
343            | Self::FileSet { .. }
344            | Self::ByteCount { .. }
345            | Self::DigestMismatch { .. }
346            | Self::MalformedDigest { .. }
347            | Self::RecordStream { .. }
348            | Self::RecordCount
349            | Self::RecordOrder { .. }
350            | Self::Decompression { .. }
351            | Self::SnapshotMismatch
352            | Self::Io { .. }
353            | Self::ResourceLimit { .. }
354            | Self::ResourceProfile { .. }
355            | Self::NoApprovedProfile { .. }
356            | Self::EnvelopeVersion { .. }
357            | Self::PayloadRegistry { .. }
358            | Self::CleanupPending(..) => 5,
359            Self::RightsNotPermitted => 6,
360            Self::Coverage { .. } => 7,
361            Self::ProducerActivity { .. } => 8,
362            Self::Envelope { .. } => 9,
363            Self::Lineage { .. } | Self::PointInTime { .. } => 10,
364            Self::UseScope { .. }
365            | Self::Capability { .. }
366            | Self::Retention
367            | Self::AmbiguousGrants => 11,
368            Self::Receipt { .. } => 12,
369        }
370    }
